//! Screen reader support for the stack: an MSAA `IAccessible` for the
//! flyout with one list item per tile, so Narrator and NVDA can read names,
//! kinds, selection and focus, and open items.

use std::ffi::c_void;

use windows::Win32::Foundation::*;
use windows::Win32::System::Com::*;
use windows::Win32::UI::Accessibility::*;
use windows::Win32::UI::Input::KeyboardAndMouse::GetFocus;
use windows::Win32::UI::WindowsAndMessaging::*;
use windows::core::*;

use crate::shelf::{self, Kind};
use crate::stackview;

const CHILDID_SELF: i32 = 0;
// oleacc.h values; this crate only exposes them under another feature.
const STATE_FOCUSABLE: u32 = 0x0010_0000;
const STATE_OFFSCREEN: u32 = 0x0001_0000;
const STATE_INVISIBLE: u32 = 0x0000_8000;

#[implement(IAccessible)]
struct StackAccessible {
    hwnd: HWND,
}

/// The item a child id names: child ids are 1-based, 0 is the list itself.
fn item(v: &VARIANT) -> Result<Option<usize>> {
    let id = i32::try_from(v).map_err(|_| Error::from(E_INVALIDARG))?;
    let n = crate::app_ref().map_or(0, |a| a.stack.items.len()) as i32;
    match id {
        CHILDID_SELF => Ok(None),
        k if (1..=n).contains(&k) => Ok(Some(k as usize - 1)),
        _ => Err(E_INVALIDARG.into()),
    }
}

fn not_here<T>() -> Result<T> {
    Err(Error::from(S_FALSE))
}

impl IDispatch_Impl for StackAccessible_Impl {
    fn GetTypeInfoCount(&self) -> Result<u32> {
        Ok(0)
    }
    fn GetTypeInfo(&self, _: u32, _: u32) -> Result<ITypeInfo> {
        Err(E_NOTIMPL.into())
    }
    fn GetIDsOfNames(&self, _: *const GUID, _: *const PCWSTR, _: u32, _: u32, _: *mut i32) -> Result<()> {
        Err(E_NOTIMPL.into())
    }
    fn Invoke(&self, _: i32, _: *const GUID, _: u32, _: DISPATCH_FLAGS, _: *const DISPPARAMS, _: *mut VARIANT, _: *mut EXCEPINFO, _: *mut u32) -> Result<()> {
        Err(E_NOTIMPL.into())
    }
}

impl IAccessible_Impl for StackAccessible_Impl {
    fn accParent(&self) -> Result<IDispatch> {
        let mut out: *mut c_void = std::ptr::null_mut();
        unsafe {
            CreateStdAccessibleObject(self.hwnd, OBJID_WINDOW.0, &IDispatch::IID, &mut out)?;
            Ok(IDispatch::from_raw(out))
        }
    }

    fn accChildCount(&self) -> Result<i32> {
        Ok(crate::app_ref().map_or(0, |a| a.stack.items.len()) as i32)
    }

    fn get_accChild(&self, _: &VARIANT) -> Result<IDispatch> {
        // Tiles are simple elements, not objects of their own.
        not_here()
    }

    fn get_accName(&self, v: &VARIANT) -> Result<BSTR> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        Ok(match item(v)? {
            None => {
                let (n, name) = (app.stack.items.len(), &app.stack.names[app.stack.current]);
                BSTR::from(if n == 1 { format!("{name}, 1 item") } else { format!("{name}, {n} items") })
            }
            Some(i) => {
                let kind = &app.stack.items[i].kind;
                BSTR::from(format!("{}, {}", shelf::name(kind), shelf::kind_label(kind)))
            }
        })
    }

    fn get_accValue(&self, v: &VARIANT) -> Result<BSTR> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        match item(v)?.map(|i| &app.stack.items[i].kind) {
            Some(Kind::File(p)) => Ok(BSTR::from(p.to_string_lossy().as_ref())),
            Some(Kind::Link(l)) => Ok(BSTR::from(l.as_str())),
            _ => not_here(),
        }
    }

    fn get_accDescription(&self, _: &VARIANT) -> Result<BSTR> {
        not_here()
    }

    fn get_accRole(&self, v: &VARIANT) -> Result<VARIANT> {
        let role = if item(v)?.is_some() { ROLE_SYSTEM_LISTITEM } else { ROLE_SYSTEM_LIST };
        Ok(VARIANT::from(role as i32))
    }

    fn get_accState(&self, v: &VARIANT) -> Result<VARIANT> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        let focused = unsafe { GetFocus() } == self.hwnd;
        let state = match item(v)? {
            None => STATE_FOCUSABLE | if focused && app.stack.items.is_empty() { STATE_SYSTEM_FOCUSED } else { 0 },
            Some(i) => {
                let mut s = STATE_FOCUSABLE | STATE_SYSTEM_SELECTABLE | STATE_SYSTEM_MULTISELECTABLE;
                if stackview::is_selected(app, i) {
                    s |= STATE_SYSTEM_SELECTED;
                }
                if focused && stackview::focus_index(app) == i {
                    s |= STATE_SYSTEM_FOCUSED;
                }
                if stackview::tile_screen_rect(app, i).is_none() {
                    s |= STATE_OFFSCREEN | STATE_INVISIBLE;
                }
                s
            }
        };
        Ok(VARIANT::from(state as i32))
    }

    fn get_accHelp(&self, _: &VARIANT) -> Result<BSTR> {
        not_here()
    }

    fn get_accHelpTopic(&self, _: *mut BSTR, _: &VARIANT) -> Result<i32> {
        Err(E_NOTIMPL.into())
    }

    fn get_accKeyboardShortcut(&self, _: &VARIANT) -> Result<BSTR> {
        not_here()
    }

    fn accFocus(&self) -> Result<VARIANT> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        if unsafe { GetFocus() } != self.hwnd {
            return Ok(VARIANT::default());
        }
        if app.stack.items.is_empty() {
            return Ok(VARIANT::from(CHILDID_SELF));
        }
        Ok(VARIANT::from(stackview::focus_index(app) as i32 + 1))
    }

    fn accSelection(&self) -> Result<VARIANT> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        // One selected tile is reported directly; for several, readers fall
        // back to each tile's selected state.
        let selected: Vec<usize> = (0..app.stack.items.len()).filter(|&i| stackview::is_selected(app, i)).collect();
        Ok(match selected[..] {
            [one] => VARIANT::from(one as i32 + 1),
            _ => VARIANT::default(),
        })
    }

    fn get_accDefaultAction(&self, v: &VARIANT) -> Result<BSTR> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        match item(v)?.map(|i| &app.stack.items[i].kind) {
            Some(Kind::Text(_)) => Ok(BSTR::from("Paste")),
            Some(_) => Ok(BSTR::from("Open")),
            None => not_here(),
        }
    }

    fn accSelect(&self, flags: i32, v: &VARIANT) -> Result<()> {
        let app = crate::app_mut().ok_or(Error::from(E_FAIL))?;
        let Some(i) = item(v)? else { return Err(E_INVALIDARG.into()) };
        if flags & (SELFLAG_TAKEFOCUS | SELFLAG_TAKESELECTION) as i32 != 0 {
            stackview::take_focus(app, i);
        }
        Ok(())
    }

    fn accLocation(&self, left: *mut i32, top: *mut i32, width: *mut i32, height: *mut i32, v: &VARIANT) -> Result<()> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        let r = match item(v)? {
            Some(i) => stackview::tile_screen_rect(app, i),
            None => {
                let mut r = RECT::default();
                unsafe { GetWindowRect(self.hwnd, &mut r).ok().map(|_| r) }
            }
        };
        let Some(r) = r else { return not_here() };
        unsafe {
            *left = r.left;
            *top = r.top;
            *width = r.right - r.left;
            *height = r.bottom - r.top;
        }
        Ok(())
    }

    fn accNavigate(&self, _: i32, _: &VARIANT) -> Result<VARIANT> {
        Err(E_NOTIMPL.into())
    }

    fn accHitTest(&self, x: i32, y: i32) -> Result<VARIANT> {
        let app = crate::app_ref().ok_or(Error::from(E_FAIL))?;
        Ok(match stackview::hit_screen(app, POINT { x, y }) {
            Some(i) => VARIANT::from(i as i32 + 1),
            None => VARIANT::from(CHILDID_SELF),
        })
    }

    fn accDoDefaultAction(&self, v: &VARIANT) -> Result<()> {
        let app = crate::app_mut().ok_or(Error::from(E_FAIL))?;
        let Some(i) = item(v)? else { return Err(E_INVALIDARG.into()) };
        stackview::activate(app, i);
        Ok(())
    }

    fn put_accName(&self, _: &VARIANT, _: &BSTR) -> Result<()> {
        Err(E_NOTIMPL.into())
    }

    fn put_accValue(&self, _: &VARIANT, _: &BSTR) -> Result<()> {
        Err(E_NOTIMPL.into())
    }
}

/// Answers WM_GETOBJECT for the stack window's client area.
pub fn get_object(hwnd: HWND, wparam: WPARAM, lparam: LPARAM) -> Option<LRESULT> {
    if lparam.0 as i32 != OBJID_CLIENT.0 {
        return None;
    }
    let acc: IAccessible = StackAccessible { hwnd }.into();
    Some(unsafe { LresultFromObject(&IAccessible::IID, wparam, &acc) })
}

/// Tells screen readers the keyboard focus moved to tile `focus`.
pub fn focus_changed(hwnd: HWND, focus: Option<usize>) {
    let child = focus.map_or(CHILDID_SELF, |i| i as i32 + 1);
    unsafe {
        NotifyWinEvent(EVENT_OBJECT_FOCUS, hwnd, OBJID_CLIENT.0, child);
    }
}
//! Quick actions on a selection: copy paths, copy as Markdown links, zip,
//! and copy to a folder. Zipping runs in-process on a worker thread and
//! posts the result back to the host window.

use std::fs::File;
use std::io;
use std::os::windows::fs::MetadataExt;
use std::path::{Path, PathBuf};

use windows::Win32::Foundation::*;
use windows::Win32::Storage::FileSystem::FileTimeToLocalFileTime;
use windows::Win32::System::Com::*;
use windows::Win32::System::Time::FileTimeToSystemTime;
use windows::Win32::UI::Shell::Common::COMDLG_FILTERSPEC;
use windows::Win32::UI::Shell::*;
use windows::Win32::UI::WindowsAndMessaging::PostMessageW;
use windows::core::*;
use zip::write::SimpleFileOptions;
use zip::{DateTime, ZipWriter};

use crate::shelf::{self, Kind};

/// Posted to the host window when a zip finishes: wParam is 1 on success.
pub const WM_ZIP_DONE: u32 = windows::Win32::UI::WindowsAndMessaging::WM_APP + 2;

/// One line per file path or link; text items are left out.
pub fn paths_text(kinds: &[&Kind]) -> String {
    kinds
        .iter()
        .filter_map(|k| match k {
            Kind::File(p) => Some(p.to_string_lossy().into_owned()),
            Kind::Link(l) => Some(l.clone()),
            Kind::Text(_) => None,
        })
        .collect::<Vec<_>>()
        .join("\r\n")
}

/// `file:///` URL for a path, with anything outside the safe set escaped.
pub fn file_url(path: &Path) -> String {
    let raw = path.to_string_lossy().replace('\\', "/");
    let mut out = String::from("file:///");
    for b in raw.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~' | b'/' | b':' => out.push(b as char),
            _ => out.push_str(&format!("%{b:02X}")),
        }
    }
    out
}

/// One Markdown link per file or link, as a list when there are several.
pub fn markdown(kinds: &[&Kind]) -> String {
    let links: Vec<String> = kinds
        .iter()
        .filter_map(|k| {
            let label = shelf::name(k).replace('[', "\\[").replace(']', "\\]");
            match k {
                Kind::File(p) => Some(format!("[{label}]({})", file_url(p))),
                Kind::Link(l) if l.starts_with("www.") => Some(format!("[{label}](https://{l})")),
                Kind::Link(l) => Some(format!("[{label}]({l})")),
                Kind::Text(_) => None,
            }
        })
        .collect();
    match links.len() {
        1 => links.into_iter().next().unwrap(),
        _ => links.iter().map(|l| format!("- {l}")).collect::<Vec<_>>().join("\r\n"),
    }
}

/// Files and folders in a selection that still exist.
pub fn existing_files(kinds: &[&Kind]) -> Vec<PathBuf> {
    kinds
        .iter()
        .filter_map(|k| if let Kind::File(p) = k { Some(p.clone()) } else { None })
        .filter(|p| p.exists())
        .collect()
}

/// Asks for a folder with the standard picker.
pub fn pick_folder(owner: HWND) -> Option<PathBuf> {
    unsafe {
        let dialog: IFileOpenDialog = CoCreateInstance(&FileOpenDialog, None, CLSCTX_INPROC_SERVER).ok()?;
        let opts = dialog.GetOptions().ok()?;
        dialog.SetOptions(opts | FOS_PICKFOLDERS | FOS_FORCEFILESYSTEM).ok()?;
        dialog.SetTitle(w!("Copy to folder")).ok()?;
        dialog.SetOkButtonLabel(w!("Copy here")).ok()?;
        dialog.Show(owner).ok()?;
        item_path(&dialog.GetResult().ok()?)
    }
}

/// Asks where a missing file went, starting in its old folder, or the
/// nearest one above it that still exists.
pub fn pick_file(owner: HWND, old: &Path) -> Option<PathBuf> {
    unsafe {
        let dialog: IFileOpenDialog = CoCreateInstance(&FileOpenDialog, None, CLSCTX_INPROC_SERVER).ok()?;
        dialog.SetTitle(w!("Find file")).ok()?;
        if let Some(name) = old.file_name() {
            let _ = dialog.SetFileName(&HSTRING::from(name));
        }
        if let Some(dir) = old.ancestors().skip(1).find(|d| d.is_dir())
            && let Ok(folder) = SHCreateItemFromParsingName::<_, _, IShellItem>(&HSTRING::from(dir.as_os_str()), None)
        {
            let _ = dialog.SetFolder(&folder);
        }
        dialog.Show(owner).ok()?;
        item_path(&dialog.GetResult().ok()?)
    }
}

fn item_path(item: &IShellItem) -> Option<PathBuf> {
    unsafe {
        let p = item.GetDisplayName(SIGDN_FILESYSPATH).ok()?;
        let path = PathBuf::from(p.to_string().ok()?);
        CoTaskMemFree(Some(p.0 as *const _));
        Some(path)
    }
}

/// Copies files into a folder with the shell's own progress and conflict
/// dialogs. Returns false if the user cancelled or it failed.
pub fn copy_to(owner: HWND, files: &[PathBuf], folder: &Path) -> bool {
    unsafe {
        let Ok(op) = CoCreateInstance::<_, IFileOperation>(&FileOperation, None, CLSCTX_ALL) else { return false };
        let _ = op.SetOwnerWindow(owner);
        let _ = op.SetOperationFlags(FOF_NOCONFIRMMKDIR | FOFX_ADDUNDORECORD);
        let Ok(dest) = SHCreateItemFromParsingName::<_, _, IShellItem>(&HSTRING::from(folder.as_os_str()), None) else { return false };
        for f in files {
            if let Ok(src) = SHCreateItemFromParsingName::<_, _, IShellItem>(&HSTRING::from(f.as_os_str()), None) {
                let _ = op.CopyItem(&src, &dest, None, None);
            }
        }
        op.PerformOperations().is_ok() && !op.GetAnyOperationsAborted().is_ok_and(|a| a.as_bool())
    }
}

/// Asks where to save a zip, starting in the first file's folder.
pub fn pick_zip(owner: HWND, near: &Path) -> Option<PathBuf> {
    unsafe {
        let dialog: IFileSaveDialog = CoCreateInstance(&FileSaveDialog, None, CLSCTX_INPROC_SERVER).ok()?;
        let filter = [COMDLG_FILTERSPEC { pszName: w!("Zip archive"), pszSpec: w!("*.zip") }];
        dialog.SetFileTypes(&filter).ok()?;
        dialog.SetDefaultExtension(w!("zip")).ok()?;
        dialog.SetFileName(w!("Cairn items.zip")).ok()?;
        dialog.SetTitle(w!("Save as zip")).ok()?;
        if let Some(dir) = near.parent()
            && let Ok(folder) = SHCreateItemFromParsingName::<_, _, IShellItem>(&HSTRING::from(dir.as_os_str()), None)
        {
            let _ = dialog.SetFolder(&folder);
        }
        dialog.Show(owner).ok()?;
        item_path(&dialog.GetResult().ok()?)
    }
}

/// Zips files and folders into `out` on a worker thread, then posts
/// WM_ZIP_DONE to `notify`.
pub fn zip(files: Vec<PathBuf>, out: PathBuf, notify: HWND) {
    let target = notify.0 as isize;
    std::thread::spawn(move || {
        let ok = write_zip(&files, &out).is_ok();
        if !ok {
            let _ = std::fs::remove_file(&out);
        }
        unsafe {
            let _ = PostMessageW(HWND(target as *mut _), WM_ZIP_DONE, WPARAM(ok as usize), LPARAM(0));
        }
    });
}

/// Each item is added under its own name, so the zip holds names, not full
/// paths. Folders bring everything inside them.
fn write_zip(files: &[PathBuf], out: &Path) -> zip::result::ZipResult<()> {
    let mut zip = ZipWriter::new(File::create(out)?);
    for f in files {
        if let Some(name) = f.file_name() {
            add(&mut zip, f, &name.to_string_lossy())?;
        }
    }
    zip.finish()?;
    Ok(())
}

fn add(zip: &mut ZipWriter<File>, path: &Path, name: &str) -> zip::result::ZipResult<()> {
    let meta = std::fs::metadata(path)?;
    let mut options = SimpleFileOptions::default().large_file(meta.len() >= u32::MAX as u64);
    if let Some(time) = zip_time(meta.last_write_time()) {
        options = options.last_modified_time(time);
    }
    if meta.is_dir() {
        zip.add_directory(format!("{name}/"), options)?;
        for entry in std::fs::read_dir(path)? {
            let entry = entry?;
            add(zip, &entry.path(), &format!("{name}/{}", entry.file_name().to_string_lossy()))?;
        }
    } else {
        zip.start_file(name, options)?;
        io::copy(&mut File::open(path)?, zip)?;
    }
    Ok(())
}

/// Zips store local time. None outside the years a zip can hold (1980 to 2107).
fn zip_time(filetime: u64) -> Option<DateTime> {
    let utc = FILETIME { dwLowDateTime: filetime as u32, dwHighDateTime: (filetime >> 32) as u32 };
    let (mut local, mut t) = (FILETIME::default(), SYSTEMTIME::default());
    unsafe {
        FileTimeToLocalFileTime(&utc, &mut local).ok()?;
        FileTimeToSystemTime(&local, &mut t).ok()?;
    }
    DateTime::from_date_and_time(t.wYear, t.wMonth as u8, t.wDay as u8, t.wHour as u8, t.wMinute as u8, t.wSecond as u8).ok()
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn copies_paths_and_links_but_not_text() {
        let (f, t, l) = (Kind::File("C:\\a b\\x.txt".into()), Kind::Text("note".into()), Kind::Link("https://e.com".into()));
        assert_eq!(paths_text(&[&f, &t, &l]), "C:\\a b\\x.txt\r\nhttps://e.com");
    }

    #[test]
    fn markdown_links_escape_paths_and_labels() {
        let f = Kind::File("C:\\My Docs\\[draft] #1.md".into());
        assert_eq!(markdown(&[&f]), "[\\[draft\\] #1.md](file:///C:/My%20Docs/%5Bdraft%5D%20%231.md)");
        let (a, b) = (Kind::Link("www.e.com/x".into()), Kind::Link("https://e.com".into()));
        assert_eq!(markdown(&[&a, &b]), "- [e.com/x](https://www.e.com/x)\r\n- [e.com](https://e.com)");
    }

    #[test]
    fn zips_files_and_folders_from_different_folders() {
        let dir = std::env::temp_dir().join("cairn-zip-test");
        let _ = std::fs::remove_dir_all(&dir);
        std::fs::create_dir_all(dir.join("one")).unwrap();
        std::fs::create_dir_all(dir.join("two").join("inner")).unwrap();
        let (a, b) = (dir.join("one").join("a.txt"), dir.join("two").join("inner").join("b.txt"));
        std::fs::write(&a, "a").unwrap();
        std::fs::write(&b, "b").unwrap();
        let out = dir.join("out.zip");
        write_zip(&[a, dir.join("two")], &out).unwrap();
        let mut zip = zip::ZipArchive::new(File::open(&out).unwrap()).unwrap();
        let mut names: Vec<&str> = zip.file_names().collect();
        names.sort();
        assert_eq!(names, ["a.txt", "two/", "two/inner/", "two/inner/b.txt"]);
        let mut text = String::new();
        io::Read::read_to_string(&mut zip.by_name("two/inner/b.txt").unwrap(), &mut text).unwrap();
        assert_eq!(text, "b");
        // Keeps the file's own date, not the 1980 default.
        assert!(zip.by_name("a.txt").unwrap().last_modified().is_some_and(|t| t.year() >= 2026));
        std::fs::remove_dir_all(&dir).unwrap();
    }
}
//! Pasting a text item: put it on the clipboard, then send Ctrl+V to the
//! app the stack was opened from.

use windows::Win32::Foundation::*;
use windows::Win32::System::DataExchange::*;
use windows::Win32::System::Memory::*;
use windows::Win32::System::Ole::CF_UNICODETEXT;
use windows::Win32::UI::Input::KeyboardAndMouse::*;

/// Replaces the clipboard with plain text.
pub fn write_text(owner: HWND, text: &str) -> bool {
    let units: Vec<u16> = text.encode_utf16().chain(Some(0)).collect();
    unsafe {
        if OpenClipboard(owner).is_err() {
            return false;
        }
        let _ = EmptyClipboard();
        let mut ok = false;
        if let Ok(global) = GlobalAlloc(GMEM_MOVEABLE, units.len() * 2) {
            let ptr = GlobalLock(global) as *mut u16;
            if !ptr.is_null() {
                std::ptr::copy_nonoverlapping(units.as_ptr(), ptr, units.len());
                let _ = GlobalUnlock(global);
                ok = SetClipboardData(CF_UNICODETEXT.0 as u32, HANDLE(global.0)).is_ok();
            }
            if !ok {
                let _ = GlobalFree(global);
            }
        }
        let _ = CloseClipboard();
        ok
    }
}

/// Sends Ctrl+V to whatever window has focus.
pub fn send_paste() {
    let key = |vk: VIRTUAL_KEY, up: bool| INPUT {
        r#type: INPUT_KEYBOARD,
        Anonymous: INPUT_0 {
            ki: KEYBDINPUT { wVk: vk, dwFlags: if up { KEYEVENTF_KEYUP } else { KEYBD_EVENT_FLAGS(0) }, ..Default::default() },
        },
    };
    let inputs = [key(VK_CONTROL, false), key(VK_V, false), key(VK_V, true), key(VK_CONTROL, true)];
    unsafe {
        SendInput(&inputs, size_of::<INPUT>() as i32);
    }
}
//! OLE drag and drop: the edge tab and the stack accept files, text and
//! links; items can be dragged out into any app.

use std::cell::{Cell, RefCell};

#[derive(Clone, Copy, PartialEq)]
pub enum Zone {
    /// The edge tab window that took the drag.
    Tab(HWND),
    Stack,
    /// The shake-to-show zone near the cursor.
    Near,
}

fn registered(name: PCWSTR) -> u16 {
    unsafe { windows::Win32::System::DataExchange::RegisterClipboardFormatW(name) as u16 }
}

fn url_format() -> u16 {
    registered(w!("UniformResourceLocatorW"))
}

fn format(cf: u16) -> FORMATETC {
    FORMATETC { cfFormat: cf, ptd: std::ptr::null_mut(), dwAspect: DVASPECT_CONTENT.0, lindex: -1, tymed: TYMED_HGLOBAL.0 as u32 }
}

fn has(data: &IDataObject, cf: u16) -> bool {
    unsafe { data.QueryGetData(&format(cf)) == S_OK }
}

fn png_format() -> u16 {
    registered(w!("PNG"))
}

fn descriptor_format() -> u16 {
    registered(w!("FileGroupDescriptorW"))
}

fn contents_format() -> u16 {
    registered(w!("FileContents"))
}

fn supported(data: &IDataObject) -> bool {
    [CF_HDROP.0, descriptor_format(), url_format(), CF_UNICODETEXT.0, png_format(), CF_DIB.0].into_iter().any(|cf| has(data, cf))
}

/// A file or folder a source offers with no path on disk, such as an email
/// attachment or a file inside a zip.
struct Virtual {
    /// May hold a relative path ("folder\file.txt") for items inside a folder.
    name: String,
    dir: bool,
    written: Option<FILETIME>,
}

/// The FILEGROUPDESCRIPTORW a source puts on the data object.
fn parse_descriptors(bytes: &[u8]) -> Option<Vec<Virtual>> {
    let count = u32::from_le_bytes(bytes.get(..4)?.try_into().ok()?) as usize;
    let size = size_of::<FILEDESCRIPTORW>();
    (0..count)
        .map(|i| {
            let raw = bytes.get(4 + i * size..4 + (i + 1) * size)?;
            let d: FILEDESCRIPTORW = unsafe { std::ptr::read_unaligned(raw.as_ptr() as *const _) };
            let (flags, attrs, units) = (d.dwFlags, d.dwFileAttributes, d.cFileName);
            let end = units.iter().position(|&u| u == 0).unwrap_or(units.len());
            Some(Virtual {
                name: String::from_utf16_lossy(&units[..end]),
                dir: flags & FD_ATTRIBUTES.0 as u32 != 0 && attrs & FILE_ATTRIBUTE_DIRECTORY.0 != 0,
                written: (flags & FD_WRITESTIME.0 as u32 != 0).then_some(d.ftLastWriteTime),
            })
        })
        .collect()
}

/// The parts of a name a source gave, safe to use as a path under Cairn's
/// folder: no "..", no drive or root, no characters Windows refuses.
fn safe_parts(name: &str) -> Option<Vec<String>> {
    const RESERVED: [&str; 22] = [
        "CON", "PRN", "AUX", "NUL", "COM1", "COM2", "COM3", "COM4", "COM5", "COM6", "COM7", "COM8", "COM9", "LPT1", "LPT2", "LPT3", "LPT4", "LPT5", "LPT6",
        "LPT7", "LPT8", "LPT9",
    ];
    let parts: Vec<String> = name
        .split(['\\', '/'])
        .map(|part| {
            let clean: String = part.chars().map(|c| if c < ' ' || "<>:\"|?*".contains(c) { '_' } else { c }).collect();
            let clean = clean.trim().trim_end_matches('.').to_string();
            let stem = clean.split('.').next().unwrap_or_default().to_ascii_uppercase();
            if RESERVED.contains(&stem.as_str()) { format!("_{clean}") } else { clean }
        })
        .filter(|p| !p.is_empty())
        .collect();
    (!parts.is_empty()).then_some(parts)
}

/// Copies a stream to a file in 64 KB pieces, so large files never sit in
/// memory whole.
fn write_stream(stream: &IStream, target: &std::path::Path) -> Option<()> {
    use std::io::Write;
    let mut file = std::fs::File::create(target).ok()?;
    let mut buf = vec![0u8; 64 * 1024];
    unsafe {
        let _ = stream.Seek(0, STREAM_SEEK_SET, None);
        loop {
            let mut read = 0u32;
            stream.Read(buf.as_mut_ptr() as *mut _, buf.len() as u32, Some(&mut read)).ok().ok()?;
            if read == 0 {
                return Some(());
            }
            file.write_all(&buf[..read as usize]).ok()?;
        }
    }
}

/// Writes the contents of virtual item `index` to `target`. Sources hand it
/// over as memory, a stream, or (Outlook messages) a structured storage.
fn save_contents(data: &IDataObject, index: usize, target: &std::path::Path) -> Option<()> {
    let fmt = FORMATETC {
        cfFormat: contents_format(),
        ptd: std::ptr::null_mut(),
        dwAspect: DVASPECT_CONTENT.0,
        lindex: index as i32,
        tymed: (TYMED_HGLOBAL.0 | TYMED_ISTREAM.0 | TYMED_ISTORAGE.0) as u32,
    };
    unsafe {
        let mut medium = data.GetData(&fmt).ok()?;
        let saved = match TYMED(medium.tymed as i32) {
            TYMED_ISTREAM => medium.u.pstm.as_ref().and_then(|s| write_stream(s, target)),
            TYMED_ISTORAGE => medium.u.pstg.as_ref().and_then(|src| {
                let dest = StgCreateDocfile(&HSTRING::from(target.as_os_str()), STGM_CREATE | STGM_READWRITE | STGM_SHARE_EXCLUSIVE, 0).ok()?;
                src.CopyTo(None, None, &dest).ok()?;
                dest.Commit(0).ok()
            }),
            TYMED_HGLOBAL => {
                let global = medium.u.hGlobal;
                let ptr = GlobalLock(global) as *const u8;
                let ok = !ptr.is_null() && std::fs::write(target, std::slice::from_raw_parts(ptr, GlobalSize(global))).is_ok();
                let _ = GlobalUnlock(global);
                ok.then_some(())
            }
            _ => None,
        };
        ReleaseStgMedium(&mut medium);
        saved
    }
}

/// FILETIME (100 ns steps since 1601) as a SystemTime.
fn system_time(ft: FILETIME) -> Option<std::time::SystemTime> {
    let ticks = ((ft.dwHighDateTime as u64) << 32 | ft.dwLowDateTime as u64).checked_sub(116_444_736_000_000_000)?;
    std::time::UNIX_EPOCH.checked_add(std::time::Duration::from_nanos(ticks.checked_mul(100)?))
}

/// Saves the files and folders a source offers without paths into Cairn's
/// folder and returns the top-level ones. None when there are none, or when
/// they are only link shortcuts (browsers offer a .url file with every
/// dragged link), so the link itself is kept instead.
fn read_virtual(data: &IDataObject) -> Option<Vec<PathBuf>> {
    let items = parse_descriptors(&read_bytes(data, descriptor_format())?)?;
    if items.iter().all(|v| v.name.to_ascii_lowercase().ends_with(".url")) {
        return None;
    }
    let dir = crate::shelf::owned_dir()?;
    // Each top-level name, and where it was saved.
    let mut tops: Vec<(String, PathBuf)> = Vec::new();
    for (i, v) in items.iter().enumerate() {
        let Some(parts) = safe_parts(&v.name) else { continue };
        let top = match tops.iter().find(|(n, _)| *n == parts[0]) {
            Some((_, p)) => p.clone(),
            None => {
                let p = crate::shelf::free_path(&dir, &parts[0]);
                tops.push((parts[0].clone(), p.clone()));
                p
            }
        };
        let target = parts[1..].iter().fold(top, |p, part| p.join(part));
        if v.dir {
            let _ = std::fs::create_dir_all(&target);
            continue;
        }
        if let Some(parent) = target.parent() {
            let _ = std::fs::create_dir_all(parent);
        }
        if save_contents(data, i, &target).is_none() {
            let _ = std::fs::remove_file(&target);
            continue;
        }
        if let Some(t) = v.written.and_then(system_time)
            && let Ok(f) = std::fs::File::options().write(true).open(&target)
        {
            let _ = f.set_modified(t);
        }
    }
    let saved: Vec<PathBuf> = tops.into_iter().map(|(_, p)| p).filter(|p| p.exists()).collect();
    (!saved.is_empty()).then_some(saved)
}

fn read_bytes(data: &IDataObject, cf: u16) -> Option<Vec<u8>> {
    unsafe {
        let mut medium = data.GetData(&format(cf)).ok()?;
        let global = medium.u.hGlobal;
        let size = GlobalSize(global);
        let ptr = GlobalLock(global) as *const u8;
        let bytes = (!ptr.is_null() && size > 0).then(|| std::slice::from_raw_parts(ptr, size).to_vec());
        let _ = GlobalUnlock(global);
        ReleaseStgMedium(&mut medium);
        bytes
    }
}

/// A .bmp file for CF_DIB data: the 14-byte file header, then the DIB as is.
pub fn bmp_from_dib(dib: &[u8]) -> Option<Vec<u8>> {
    let u32_at = |at: usize| Some(u32::from_le_bytes(dib.get(at..at + 4)?.try_into().ok()?));
    let header = u32_at(0)? as usize;
    let bits = u16::from_le_bytes(dib.get(14..16)?.try_into().ok()?) as u32;
    let compression = u32_at(16)?;
    let used = u32_at(32)?;
    let colors = if used > 0 { used } else if bits <= 8 { 1 << bits } else { 0 };
    // A 40-byte header with BI_BITFIELDS is followed by three colour masks.
    let masks = if compression == 3 && header == 40 { 12 } else { 0 };
    let offset = 14 + header + masks + colors as usize * 4;
    if header < 40 || offset > 14 + dib.len() {
        return None;
    }
    let mut out = Vec::with_capacity(14 + dib.len());
    out.extend_from_slice(b"BM");
    out.extend_from_slice(&((14 + dib.len()) as u32).to_le_bytes());
    out.extend_from_slice(&[0; 4]);
    out.extend_from_slice(&(offset as u32).to_le_bytes());
    out.extend_from_slice(dib);
    Some(out)
}

fn read_utf16(data: &IDataObject, cf: u16) -> Option<String> {
    unsafe {
        let mut medium = data.GetData(&format(cf)).ok()?;
        let global = medium.u.hGlobal;
        let size = GlobalSize(global) / 2;
        let ptr = GlobalLock(global) as *const u16;
        let text = (!ptr.is_null()).then(|| {
            let units = std::slice::from_raw_parts(ptr, size);
            let end = units.iter().position(|&u| u == 0).unwrap_or(units.len());
            String::from_utf16_lossy(&units[..end])
        });
        let _ = GlobalUnlock(global);
        ReleaseStgMedium(&mut medium);
        text
    }
}

fn read_paths(data: &IDataObject) -> Option<Vec<PathBuf>> {
    unsafe {
        let mut medium = data.GetData(&format(CF_HDROP.0)).ok()?;
        let drop = HDROP(medium.u.hGlobal.0);
        let count = DragQueryFileW(drop, u32::MAX, None);
        let paths = (0..count)
            .map(|i| {
                let mut buf = vec![0u16; DragQueryFileW(drop, i, None) as usize + 1];
                let n = DragQueryFileW(drop, i, Some(&mut buf)) as usize;
                PathBuf::from(String::from_utf16_lossy(&buf[..n]))
            })
            .collect();
        ReleaseStgMedium(&mut medium);
        Some(paths)
    }
}

/// Files first, then files with no path (saved into Cairn), then a link,
/// then plain text, then an image. Images come last because apps like Excel
/// offer a picture of copied cells next to the text.
pub fn read(data: &IDataObject) -> Option<Dropped> {
    if has(data, CF_HDROP.0) {
        return read_paths(data).map(Dropped::Paths);
    }
    if has(data, descriptor_format())
        && let Some(paths) = read_virtual(data)
    {
        return Some(Dropped::Paths(paths));
    }
    if let Some(text) = read_utf16(data, url_format()).or_else(|| read_utf16(data, CF_UNICODETEXT.0)) {
        // An image copied or dragged from a browser comes with its address.
        // Keep the picture: the address may stop working once the page closes.
        if crate::shelf::looks_like_link(&text)
            && let Some(image) = read_image(data)
        {
            return Some(image);
        }
        return Some(Dropped::Text(text));
    }
    read_image(data)
}

fn read_image(data: &IDataObject) -> Option<Dropped> {
    if let Some(png) = read_bytes(data, png_format()) {
        return Some(Dropped::Image { bytes: png, ext: "png" });
    }
    read_bytes(data, CF_DIB.0).and_then(|dib| bmp_from_dib(&dib)).map(|bmp| Dropped::Image { bytes: bmp, ext: "bmp" })
}

/// Cairn keeps references, so ask the source to copy, never move.
fn choose_effect(allowed: DROPEFFECT) -> DROPEFFECT {
    if allowed.0 & DROPEFFECT_COPY.0 != 0 {
        DROPEFFECT_COPY
    } else if allowed.0 & DROPEFFECT_LINK.0 != 0 {
        DROPEFFECT_LINK
    } else {
        DROPEFFECT_NONE
    }
}

/// The label under the source's drag image ("Add to Stack"). Shown only by
/// sources that allow it, such as Explorer and Cairn itself.
fn describe(data: &IDataObject, kind: DROPIMAGETYPE, message: &str, insert: &str) {
    let text = |s: &str| {
        let mut out = [0u16; 260];
        for (o, c) in out.iter_mut().zip(s.encode_utf16().take(259)) {
            *o = c;
        }
        out
    };
    let d = DROPDESCRIPTION { r#type: kind, szMessage: text(message), szInsert: text(insert) };
    let bytes = unsafe { std::slice::from_raw_parts(&d as *const _ as *const u8, size_of::<DROPDESCRIPTION>()) };
    let _ = set_global(data, registered(w!("DropDescription")), bytes);
}

/// Whether the shell is drawing the drag image and its effect badge, in
/// which case the source shows a plain arrow.
fn showing_layered(data: &IDataObject) -> bool {
    let cf = registered(w!("IsShowingLayered"));
    has(data, cf) && read_bytes(data, cf).is_some_and(|b| b.iter().take(4).any(|&x| x != 0))
}

#[implement(IDropTarget)]
struct DropTarget {
    zone: Zone,
    hwnd: HWND,
    accept: Cell<bool>,
    /// Keeps the source's drag image drawn over Cairn's windows.
    helper: Option<IDropTargetHelper>,
    data: RefCell<Option<IDataObject>>,
    /// The stack name last put in the drop label.
    label: RefCell<String>,
}

impl DropTarget {
    /// Names the stack the drop will land in, when that changed.
    fn update_label(&self) {
        let name = crate::drop_label();
        if *self.label.borrow() == name {
            return;
        }
        if let Some(data) = &*self.data.borrow() {
            describe(data, DROPIMAGE_COPY, "Add to %1", &name);
        }
        *self.label.borrow_mut() = name;
    }

    fn clear_label(&self) {
        if let Some(data) = &*self.data.borrow()
            && !self.label.borrow().is_empty()
        {
            describe(data, DROPIMAGE_INVALID, "", "");
        }
        self.label.borrow_mut().clear();
    }
}

impl IDropTarget_Impl for DropTarget_Impl {
    fn DragEnter(&self, data: Option<&IDataObject>, _: MODIFIERKEYS_FLAGS, pt: &POINTL, effect: *mut DROPEFFECT) -> Result<()> {
        let ok = data.is_some_and(supported);
        self.accept.set(ok);
        *self.data.borrow_mut() = data.cloned();
        self.label.borrow_mut().clear();
        unsafe {
            *effect = if ok { choose_effect(*effect) } else { DROPEFFECT_NONE };
        }
        crate::on_drag_hover(self.zone, ok);
        if ok {
            self.update_label();
        }
        if let (Some(h), Some(d)) = (&self.helper, data) {
            unsafe {
                let _ = h.DragEnter(self.hwnd, d, &POINT { x: pt.x, y: pt.y }, *effect);
            }
        }
        Ok(())
    }

    fn DragOver(&self, _: MODIFIERKEYS_FLAGS, pt: &POINTL, effect: *mut DROPEFFECT) -> Result<()> {
        unsafe {
            *effect = if self.accept.get() { choose_effect(*effect) } else { DROPEFFECT_NONE };
        }
        if self.accept.get() {
            crate::on_drag_move(self.zone, POINT { x: pt.x, y: pt.y });
            // Pointing at another stack's name switches to it.
            self.update_label();
        }
        if let Some(h) = &self.helper {
            unsafe {
                let _ = h.DragOver(&POINT { x: pt.x, y: pt.y }, *effect);
            }
        }
        Ok(())
    }

    fn DragLeave(&self) -> Result<()> {
        self.clear_label();
        if let Some(h) = &self.helper {
            unsafe {
                let _ = h.DragLeave();
            }
        }
        *self.data.borrow_mut() = None;
        crate::on_drag_hover(self.zone, false);
        Ok(())
    }

    fn Drop(&self, data: Option<&IDataObject>, _: MODIFIERKEYS_FLAGS, pt: &POINTL, effect: *mut DROPEFFECT) -> Result<()> {
        self.clear_label();
        *self.data.borrow_mut() = None;
        crate::on_drag_hover(self.zone, false);
        let dropped = data.and_then(read);
        unsafe {
            *effect = if dropped.is_some() { choose_effect(*effect) } else { DROPEFFECT_NONE };
        }
        if let (Some(h), Some(d)) = (&self.helper, data) {
            unsafe {
                let _ = h.Drop(d, &POINT { x: pt.x, y: pt.y }, *effect);
            }
        }
        if let Some(d) = dropped {
            crate::on_drop(self.zone, d);
        }
        Ok(())
    }
}

pub fn register(hwnd: HWND, zone: Zone) -> Result<()> {
    let helper = unsafe { CoCreateInstance(&CLSID_DragDropHelper, None, CLSCTX_INPROC_SERVER).ok() };
    let target: IDropTarget = DropTarget {
        zone,
        hwnd,
        accept: Cell::new(false),
        helper,
        data: RefCell::new(None),
        label: RefCell::new(String::new()),
    }
    .into();
    unsafe { RegisterDragDrop(hwnd, &target) }
}

#[implement(IDropSource)]
struct DropSource {
    data: IDataObject,
}

impl IDropSource_Impl for DropSource_Impl {
    fn QueryContinueDrag(&self, escape: BOOL, keys: MODIFIERKEYS_FLAGS) -> HRESULT {
        if escape.as_bool() {
            DRAGDROP_S_CANCEL
        } else if keys.0 & MK_LBUTTON.0 == 0 {
            DRAGDROP_S_DROP
        } else {
            S_OK
        }
    }

    fn GiveFeedback(&self, _: DROPEFFECT) -> HRESULT {
        // The shell draws the effect beside the drag image; a "+" on the
        // cursor as well would show it twice.
        if showing_layered(&self.data) {
            unsafe {
                if let Ok(arrow) = LoadCursorW(None, IDC_ARROW) {
                    SetCursor(arrow);
                }
            }
            return S_OK;
        }
        DRAGDROP_S_USEDEFAULTCURSORS
    }
}

fn set_utf16(obj: &IDataObject, cf: u16, text: &str) -> Option<()> {
    let units: Vec<u16> = text.encode_utf16().chain(Some(0)).collect();
    let bytes = unsafe { std::slice::from_raw_parts(units.as_ptr() as *const u8, units.len() * 2) };
    set_global(obj, cf, bytes)
}

fn set_global(obj: &IDataObject, cf: u16, bytes: &[u8]) -> Option<()> {
    unsafe {
        let global = GlobalAlloc(GMEM_MOVEABLE, bytes.len()).ok()?;
        let ptr = GlobalLock(global) as *mut u8;
        if ptr.is_null() {
            let _ = GlobalFree(global);
            return None;
        }
        std::ptr::copy_nonoverlapping(bytes.as_ptr(), ptr, bytes.len());
        let _ = GlobalUnlock(global);
        let medium = STGMEDIUM {
            tymed: TYMED_HGLOBAL.0 as u32,
            u: STGMEDIUM_0 { hGlobal: global },
            pUnkForRelease: ManuallyDrop::new(None),
        };
        // With fRelease = TRUE the data object owns the memory on success.
        if obj.SetData(&format(cf), &medium, true).is_err() {
            let _ = GlobalFree(global);
            return None;
        }
    }
    Some(())
}

/// A data object other apps understand: the shell's own for files, plain
/// text (plus a URL format for links) otherwise.
pub fn data_object(kind: &Kind) -> Option<IDataObject> {
    unsafe {
        match kind {
            Kind::File(p) => {
                let item: IShellItem = SHCreateItemFromParsingName(&HSTRING::from(p.to_string_lossy().as_ref()), None).ok()?;
                item.BindToHandler(None, &BHID_DataObject).ok()
            }
            Kind::Text(t) => {
                let obj: IDataObject = SHCreateDataObject(None, None, None).ok()?;
                set_utf16(&obj, CF_UNICODETEXT.0, t)?;
                Some(obj)
            }
            Kind::Link(l) => {
                let obj: IDataObject = SHCreateDataObject(None, None, None).ok()?;
                set_utf16(&obj, CF_UNICODETEXT.0, l)?;
                set_utf16(&obj, url_format(), l)?;
                Some(obj)
            }
        }
    }
}

/// One data object for several items. Files win: if any are selected, only
/// the files go, as one shell item array, so Explorer and mail clients see a
/// normal multi-file drag. Otherwise texts and links go as lines of text.
pub fn data_object_many(kinds: &[&Kind]) -> Option<IDataObject> {
    if let [one] = kinds {
        return data_object(one);
    }
    let paths: Vec<&PathBuf> = kinds.iter().filter_map(|k| if let Kind::File(p) = k { Some(p) } else { None }).collect();
    if paths.is_empty() {
        let lines: Vec<&str> = kinds
            .iter()
            .filter_map(|k| match k {
                Kind::Text(t) | Kind::Link(t) => Some(t.as_str()),
                Kind::File(_) => None,
            })
            .collect();
        let obj: IDataObject = unsafe { SHCreateDataObject(None, None, None).ok()? };
        set_utf16(&obj, CF_UNICODETEXT.0, &lines.join("\r\n"))?;
        return Some(obj);
    }
    unsafe {
        let pidls: Vec<*const Common::ITEMIDLIST> = paths
            .iter()
            .map(|p| ILCreateFromPathW(&HSTRING::from(p.to_string_lossy().as_ref())) as *const Common::ITEMIDLIST)
            .filter(|p| !p.is_null())
            .collect();
        let data = (!pidls.is_empty())
            .then(|| SHCreateShellItemArrayFromIDLists(&pidls).ok())
            .flatten()
            .and_then(|array| array.BindToHandler(None, &BHID_DataObject).ok());
        for p in pidls {
            ILFree(Some(p));
        }
        data
    }
}

/// Runs the modal drag loop with `image` under the cursor. Callers must not
/// hold app state across it, since window messages keep flowing while it
/// runs. The shell owns the image's bitmap once it takes it.
pub fn drag_out(data: &IDataObject, image: Option<SHDRAGIMAGE>) -> DROPEFFECT {
    if let Some(img) = image {
        let helper: Option<IDragSourceHelper2> = unsafe { CoCreateInstance(&CLSID_DragDropHelper, None, CLSCTX_INPROC_SERVER).ok() };
        let taken = helper.is_some_and(|h| unsafe {
            let _ = h.SetFlags(DSH_ALLOWDROPDESCRIPTIONTEXT.0 as u32);
            h.InitializeFromBitmap(&img, data).is_ok()
        });
        if !taken {
            unsafe {
                let _ = DeleteObject(img.hbmpDragImage);
            }
        }
    }
    let source: IDropSource = DropSource { data: data.clone() }.into();
    let mut effect = DROPEFFECT_NONE;
    unsafe {
        let _ = DoDragDrop(data, &source, DROPEFFECT_COPY | DROPEFFECT_LINK, &mut effect);
    }
    effect
}

        let (t, l) = (Kind::Text("one".into()), Kind::Link("https://example.com".into()));
        let obj = data_object_many(&[&t, &l]).expect("data object");
        assert!(matches!(read(&obj), Some(Dropped::Text(s)) if s == "one\r\nhttps://example.com"));
    }

    #[test]
    fn a_browser_image_keeps_the_picture_not_its_address() {
        let with = |text: &str| {
            let obj: IDataObject = unsafe { SHCreateDataObject(None, None, None) }.unwrap();
            set_utf16(&obj, CF_UNICODETEXT.0, text).unwrap();
            set_global(&obj, png_format(), b"\x89PNG fake").unwrap();
            read(&obj)
        };
        assert!(matches!(with("https://example.com/cat.png"), Some(Dropped::Image { ext: "png", .. })));
        // Copied cells from Excel come with a picture too; the text wins.
        assert!(matches!(with("Q3\t1200"), Some(Dropped::Text(t)) if t == "Q3\t1200"));
    }

    #[test]
    fn reads_file_descriptors() {
        let entry = |name: &str, flags: u32, attrs: u32| {
            let mut d: FILEDESCRIPTORW = unsafe { std::mem::zeroed() };
            d.dwFlags = flags;
            d.dwFileAttributes = attrs;
            d.ftLastWriteTime = FILETIME { dwLowDateTime: 1, dwHighDateTime: 2 };
            let mut units = [0u16; 260];
            for (u, c) in units.iter_mut().zip(name.encode_utf16()) {
                *u = c;
            }
            d.cFileName = units;
            unsafe { std::slice::from_raw_parts(&d as *const _ as *const u8, size_of::<FILEDESCRIPTORW>()) }.to_vec()
        };
        let mut bytes = 2u32.to_le_bytes().to_vec();
        bytes.extend(entry("Photos", FD_ATTRIBUTES.0 as u32, FILE_ATTRIBUTE_DIRECTORY.0));
        bytes.extend(entry("Photos\\ü.jpg", FD_WRITESTIME.0 as u32, 0));
        let items = parse_descriptors(&bytes).unwrap();
        assert_eq!((items[0].name.as_str(), items[0].dir), ("Photos", true));
        assert_eq!((items[1].name.as_str(), items[1].dir), ("Photos\\ü.jpg", false));
        assert!(items[0].written.is_none() && items[1].written.is_some());
        // A count larger than the data is refused.
        bytes[0] = 3;
        assert!(parse_descriptors(&bytes).is_none());
    }

    #[test]
    fn names_from_sources_stay_inside_cairns_folder() {
        assert_eq!(safe_parts("a\\b/c.txt").unwrap(), ["a", "b", "c.txt"]);
        assert_eq!(safe_parts("..\\..\\Windows\\x.dll").unwrap(), ["Windows", "x.dll"]);
        assert_eq!(safe_parts("C:\\x").unwrap(), ["C_", "x"]);
        assert_eq!(safe_parts("re: why?.msg").unwrap(), ["re_ why_.msg"]);
        assert_eq!(safe_parts("con.txt").unwrap(), ["_con.txt"]);
        assert!(safe_parts("\\..\\.").is_none());
    }

    #[test]
    fn streams_are_copied_to_files() {
        let dir = std::env::temp_dir().join("cairn-stream-test");
        let _ = std::fs::create_dir_all(&dir);
        let data: Vec<u8> = (0..200_000u32).map(|i| i as u8).collect();
        let stream = unsafe { SHCreateMemStream(Some(&data)) }.unwrap();
        let target = dir.join("big.bin");
        write_stream(&stream, &target).unwrap();
        assert_eq!(std::fs::read(&target).unwrap(), data);
    }

    #[test]
    fn dib_becomes_a_bmp_file() {
        // A 2×1 24-bit DIB: 40-byte header, no palette, one padded row.
        let mut dib = Vec::new();
        for v in [40u32, 2, 1] {
            dib.extend_from_slice(&v.to_le_bytes());
        }
        dib.extend_from_slice(&1u16.to_le_bytes());
        dib.extend_from_slice(&24u16.to_le_bytes());
        dib.extend_from_slice(&[0; 24]);
        dib.extend_from_slice(&[255, 0, 0, 0, 255, 0, 0, 0]);
        let bmp = bmp_from_dib(&dib).unwrap();
        assert_eq!(&bmp[..2], b"BM");
        assert_eq!(u32::from_le_bytes(bmp[2..6].try_into().unwrap()) as usize, bmp.len());
        // Pixels start right after the two headers.
        assert_eq!(u32::from_le_bytes(bmp[10..14].try_into().unwrap()), 54);
        // An 8-bit DIB has a 256-colour palette in between.
        dib[14] = 8;
        dib.extend_from_slice(&[0; 1024]);
        assert_eq!(u32::from_le_bytes(bmp_from_dib(&dib).unwrap()[10..14].try_into().unwrap()), 54 + 1024);
    }
}

//! The edge tab: a slim pill on a screen edge of each enabled monitor, each a
//! real OLE drop target. When a supported drag reaches one, Windows calls
//! DragEnter and the pill grows into the drop zone, so Cairn never watches the
//! mouse globally.
//!
//! Each window keeps one size. While idle, only a thin strip along the edge
//! has non-zero alpha; everywhere else clicks and drops fall through to the
//! windows underneath. Dragging the pill moves it along the edge.

use std::time::Instant;

use windows::Win32::Foundation::*;
use windows::Win32::Graphics::DirectWrite::*;
use windows::Win32::Graphics::Gdi::*;
use windows::Win32::System::Ole::RevokeDragDrop;
use windows::Win32::UI::Controls::WM_MOUSELEAVE;
use windows::Win32::UI::Input::KeyboardAndMouse::*;
use windows::Win32::UI::WindowsAndMessaging::*;
use windows::core::*;

use crate::gfx::{Color, Surface, Tween, bezier};
use crate::settings::Edge;
use crate::{App, dnd, dpi_at};

const WIN_W: f32 = 120.0;
const WIN_H: f32 = 200.0;
const CARD_W: f32 = 88.0;
const CARD_H: f32 = 160.0;
const CARD_GAP: f32 = 6.0;
const HIT_W: f32 = 10.0;
const HIT_H: f32 = 120.0;
const PILL_W: f32 = 4.0;
const PILL_H: f32 = 96.0;
const PILL_GAP: f32 = 3.0;
const OPEN_MS: f32 = 140.0;
const CLOSE_MS: f32 = 120.0;
const FLASH_MS: u32 = 700;
/// The lichen pulse a drop sets off.
const PULSE_MS: f32 = 600.0;
const ANIM_TIMER: usize = 1;
const FLASH_TIMER: usize = 2;

/// A monitor as Cairn sees it: its device name (the key settings use), work
/// area and whether it is the main one.
pub struct Monitor {
    pub handle: HMONITOR,
    pub device: String,
    pub work: RECT,
    pub primary: bool,
}

/// Every connected monitor, main one first, then left to right.
pub fn monitors() -> Vec<Monitor> {
    unsafe extern "system" fn each(mon: HMONITOR, _: HDC, _: *mut RECT, data: LPARAM) -> BOOL {
        unsafe {
            let list = &mut *(data.0 as *mut Vec<Monitor>);
            let mut info = MONITORINFOEXW::default();
            info.monitorInfo.cbSize = size_of::<MONITORINFOEXW>() as u32;
            if GetMonitorInfoW(mon, &mut info.monitorInfo).as_bool() {
                let end = info.szDevice.iter().position(|&c| c == 0).unwrap_or(info.szDevice.len());
                list.push(Monitor {
                    handle: mon,
                    device: String::from_utf16_lossy(&info.szDevice[..end]),
                    work: info.monitorInfo.rcWork,
                    primary: info.monitorInfo.dwFlags & MONITORINFOF_PRIMARY != 0,
                });
            }
        }
        TRUE
    }
    let mut list: Vec<Monitor> = Vec::new();
    unsafe {
        let _ = EnumDisplayMonitors(None, None, Some(each), LPARAM(&mut list as *mut _ as isize));
    }
    list.sort_by_key(|m| (!m.primary, m.work.left, m.work.top));
    list
}

pub struct EdgeTab {
    pub hwnd: HWND,
    pub monitor: HMONITOR,
    pub device: String,
    work: RECT,
    pub fullscreen_hidden: bool,
    surface: Option<Surface>,
    dpi: u32,
    origin: (i32, i32),
    drag_over: bool,
    /// Caption shown briefly after a drop ("Added 2").
    flash: Option<String>,
    hover: bool,
    tracked: bool,
    /// Pressed on the pill: cursor y and window y at the press, and whether
    /// it has moved far enough to count as a drag along the edge.
    press: Option<(i32, i32, bool)>,
    /// Expansion: 0 is the idle pill, 1 the full drop zone.
    progress: f32,
    from: f32,
    start: Instant,
    /// Fades from 1 to 0 after a drop.
    pulse: Tween,
}

impl EdgeTab {
    fn expanded(&self) -> bool {
        self.drag_over || self.flash.is_some()
    }
}

/// Where the tab's center sits, as a fraction of the work area's height,
/// kept inside it.
fn center_y(work: &RECT, pos: f32, h: i32) -> i32 {
    let span = work.bottom - work.top;
    let y = work.top + (span as f32 * pos) as i32 - h / 2;
    y.clamp(work.top, (work.bottom - h).max(work.top))
}

/// The fraction `center_y` would turn back into this window top.
fn pos_of(work: &RECT, top: i32, h: i32) -> f32 {
    let span = (work.bottom - work.top).max(1);
    ((top + h / 2 - work.top) as f32 / span as f32).clamp(0.0, 1.0)
}

/// The monitors that get a tab: the ones not turned off in settings, or the
/// main one if that leaves none.
fn enabled(app: &App) -> Vec<Monitor> {
    let all = monitors();
    let on: Vec<Monitor> = all.iter().filter(|m| !app.settings.monitors_off.contains(&m.device)).map(clone).collect();
    if on.is_empty() { all.into_iter().filter(|m| m.primary).collect() } else { on }
}

fn clone(m: &Monitor) -> Monitor {
    Monitor { handle: m.handle, device: m.device.clone(), work: m.work, primary: m.primary }
}

/// Makes one tab per enabled monitor, reusing windows where it can, and
/// puts each against the chosen edge.
pub fn sync(app: &mut App) {
    let want = enabled(app);
    while app.tabs.len() > want.len() {
        let t = app.tabs.pop().unwrap();
        unsafe {
            let _ = RevokeDragDrop(t.hwnd);
            let _ = DestroyWindow(t.hwnd);
        }
    }
    let ex = WS_EX_LAYERED | WS_EX_TOPMOST | WS_EX_TOOLWINDOW | WS_EX_NOACTIVATE;
    while app.tabs.len() < want.len() {
        let Ok(hwnd) = crate::create_window(w!("CairnTab"), ex) else { break };
        let _ = dnd::register(hwnd, dnd::Zone::Tab(hwnd));
        app.tabs.push(EdgeTab {
            hwnd,
            monitor: HMONITOR::default(),
            device: String::new(),
            work: RECT::default(),
            fullscreen_hidden: false,
            surface: None,
            dpi: 96,
            origin: (0, 0),
            drag_over: false,
            flash: None,
            hover: false,
            tracked: false,
            press: None,
            progress: 0.0,
            from: 0.0,
            start: Instant::now(),
            pulse: Tween::new(0.0),
        });
    }
    for (t, m) in app.tabs.iter_mut().zip(want) {
        t.monitor = m.handle;
        t.device = m.device;
        t.work = m.work;
    }
    app.active_tab = app.active_tab.min(app.tabs.len().saturating_sub(1));
    for i in 0..app.tabs.len() {
        position(app, i);
        show(app, i);
    }
}

/// The tab to open the stack from: the one on the monitor under the cursor,
/// or the first.
pub fn under_cursor(app: &App) -> usize {
    let mut pt = POINT::default();
    unsafe {
        let _ = GetCursorPos(&mut pt);
        let mon = MonitorFromPoint(pt, MONITOR_DEFAULTTONEAREST);
        app.tabs.iter().position(|t| t.monitor == mon).unwrap_or(0)
    }
}

/// The active tab's inner edge at its vertical center, plus its work area:
/// where the stack flyout opens from.
pub fn anchor(app: &App) -> (POINT, RECT) {
    let Some(t) = app.tabs.get(app.active_tab) else {
        let wa = monitors().into_iter().next().map(|m| m.work).unwrap_or_default();
        return (POINT { x: wa.right, y: (wa.top + wa.bottom) / 2 }, wa);
    };
    let s = t.dpi as f32 / 96.0;
    let hit = (HIT_W * s) as i32;
    let x = match app.settings.edge {
        Edge::Left => t.work.left + hit,
        Edge::Right => t.work.right - hit,
    };
    let h = (WIN_H * s).ceil() as i32;
    (POINT { x, y: t.origin.1 + h / 2 }, t.work)
}

/// Puts a tab against the chosen edge, at the saved height.
pub fn position(app: &mut App, i: usize) {
    let (edge, pos) = (app.settings.edge, app.settings.tab_pos);
    let Some(t) = app.tabs.get_mut(i) else { return };
    let wa = t.work;
    let dpi = dpi_at(POINT { x: (wa.left + wa.right) / 2, y: (wa.top + wa.bottom) / 2 });
    let s = dpi as f32 / 96.0;
    let (w, h) = ((WIN_W * s).ceil() as i32, (WIN_H * s).ceil() as i32);
    let x = match edge {
        Edge::Left => wa.left,
        Edge::Right => wa.right - w,
    };
    t.dpi = dpi;
    t.origin = (x, center_y(&wa, pos, h));
    if t.surface.as_ref().is_none_or(|surf| surf.width != w || surf.height != h) {
        t.surface = Surface::new(&app.gfx, w, h).ok();
    }
    render(app, i);
}

/// Left x of a box `w` wide that sits `gap` DIPs from the screen edge.
fn from_edge(edge: Edge, gap: f32, w: f32) -> f32 {
    match edge {
        Edge::Left => gap,
        Edge::Right => WIN_W - gap - w,
    }
}

pub fn render(app: &mut App, i: usize) {
    let Some(t) = app.tabs.get(i) else { return };
    let Some(surf) = &t.surface else { return };
    let (g, th) = (&app.gfx, &app.theme);
    let edge = app.settings.edge;
    let p = t.progress;
    let lerp = |a: f32, b: f32| a + (b - a) * p;
    let caption = g.text_format(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_NORMAL).ok();
    let flash = t.flash.clone();
    let pulse = t.pulse.value();
    let hover = t.hover || t.press.is_some_and(|(_, _, moved)| moved);
    let _ = surf.draw(t.dpi, |d| {
        // The hit strip: alpha 1/255, invisible but it catches drags.
        let cy = WIN_H / 2.0;
        d.fill_rect(from_edge(edge, 0.0, HIT_W), cy - HIT_H / 2.0, HIT_W, HIT_H, Color(0.0, 0.0, 0.0, 1.0 / 255.0));
        if p < 0.02 {
            let alpha = if hover { 0.9 } else { 0.4 };
            let x = from_edge(edge, PILL_GAP, PILL_W);
            d.fill_rounded(x, cy - PILL_H / 2.0, PILL_W, PILL_H, PILL_W / 2.0, th.outline_control.alpha(alpha));
            return;
        }
        let (w, h) = (lerp(PILL_W, CARD_W), lerp(PILL_H, CARD_H));
        let x = from_edge(edge, lerp(PILL_GAP, CARD_GAP), w);
        let y = cy - h / 2.0;
        let r = lerp(PILL_W / 2.0, 8.0);
        d.rect_shadow(x, y + 6.0, w, h, r, 12.0 * p, 0.45 * th.shadow * p);
        d.fill_rounded(x, y, w, h, r, th.surface_flyout.alpha(p));
        d.fill_rounded(x, y, w, h, r, th.primary_subtle.alpha(p));
        if pulse > 0.0 {
            d.fill_rounded(x, y, w, h, r, th.primary.alpha(0.28 * pulse * p));
        }
        d.stroke_rounded_dashed(x + 0.75, y + 0.75, w - 1.5, h - 1.5, r, 1.5, th.primary.alpha(p));
        if p > 0.6 {
            let a = (p - 0.6) / 0.4;
            d.cairn_mark(x + (w - 32.0) / 2.0, y + 44.0, 32.0, th.stone.alpha(a), th.primary.alpha(a));
            if let Some(f) = &caption {
                let (text, color) = match &flash {
                    Some(s) => (s.as_str(), th.text),
                    None => ("Drop here", th.text_secondary),
                };
                d.text(text, f, x, y + 92.0, w, 20.0, color.alpha(a));
            }
        }
    });
    let _ = surf.present(t.hwnd, t.origin.0, t.origin.1, 1.0);
}

pub fn show(app: &App, i: usize) {
    let Some(t) = app.tabs.get(i) else { return };
    unsafe {
        let _ = ShowWindow(t.hwnd, if t.fullscreen_hidden { SW_HIDE } else { SW_SHOWNOACTIVATE });
    }
}

/// Re-asserts topmost, since other topmost windows can cover the tabs.
pub fn raise(app: &App) {
    for t in app.tabs.iter().filter(|t| !t.fullscreen_hidden) {
        unsafe {
            let _ = SetWindowPos(t.hwnd, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE);
        }
    }
}

fn animate(app: &mut App, i: usize) {
    let Some(t) = app.tabs.get_mut(i) else { return };
    t.from = t.progress;
    t.start = Instant::now();
    unsafe {
        SetTimer(t.hwnd, ANIM_TIMER, 10, None);
    }
    tick(app, i);
}

fn tick(app: &mut App, i: usize) {
    let animations = app.animations;
    let Some(t) = app.tabs.get_mut(i) else { return };
    let target = if t.expanded() { 1.0 } else { 0.0 };
    let dur = if target > t.from { OPEN_MS } else { CLOSE_MS };
    let k = if animations { (t.start.elapsed().as_secs_f32() * 1000.0 / dur).min(1.0) } else { 1.0 };
    t.progress = t.from + (target - t.from) * bezier(0.0, 0.0, 0.0, 1.0, k);
    if k >= 1.0 && t.pulse.done() {
        unsafe {
            let _ = KillTimer(t.hwnd, ANIM_TIMER);
        }
    }
    render(app, i);
}

pub fn is_drag_over(app: &App, i: usize) -> bool {
    app.tabs.get(i).is_some_and(|t| t.drag_over)
}

pub fn set_drag_over(app: &mut App, i: usize, over: bool) {
    if let Some(t) = app.tabs.get_mut(i)
        && t.drag_over != over
    {
        t.drag_over = over;
        animate(app, i);
    }
}

/// Shows what the drop did, then folds back to the pill.
pub fn dropped(app: &mut App, i: usize, added: usize) {
    let Some(t) = app.tabs.get_mut(i) else { return };
    t.flash = Some(match added {
        0 => "Already here".to_string(),
        n => format!("Added {n}"),
    });
    if added > 0 && app.animations {
        t.pulse.run(1.0, 0.0, PULSE_MS);
    }
    unsafe {
        SetTimer(t.hwnd, FLASH_TIMER, FLASH_MS, None);
    }
    animate(app, i);
}

fn cursor_y() -> i32 {
    let mut pt = POINT::default();
    unsafe {
        let _ = GetCursorPos(&mut pt);
    }
    pt.y
}

/// Saves where a dragged tab ended up and moves the other tabs to match.
fn finish_move(app: &mut App, i: usize) {
    let t = &app.tabs[i];
    let h = (WIN_H * t.dpi as f32 / 96.0).ceil() as i32;
    app.settings.tab_pos = pos_of(&t.work, t.origin.1, h);
    app.settings.save();
    for j in (0..app.tabs.len()).filter(|&j| j != i) {
        position(app, j);
    }
}

pub fn handle(app: &mut App, i: usize, hwnd: HWND, msg: u32, wparam: WPARAM, _lparam: LPARAM) -> Option<LRESULT> {
    match msg {
        WM_TIMER if wparam.0 == ANIM_TIMER => tick(app, i),
        WM_TIMER if wparam.0 == FLASH_TIMER => {
            unsafe {
                let _ = KillTimer(hwnd, FLASH_TIMER);
            }
            app.tabs[i].flash = None;
            animate(app, i);
        }
        WM_LBUTTONDOWN => {
            let t = &mut app.tabs[i];
            t.press = Some((cursor_y(), t.origin.1, false));
            unsafe {
                SetCapture(hwnd);
            }
        }
        WM_MOUSEMOVE => {
            let t = &mut app.tabs[i];
            if let Some((from_y, from_top, moved)) = t.press {
                let dy = cursor_y() - from_y;
                let moved = moved || dy.abs() > unsafe { GetSystemMetrics(SM_CYDRAG) };
                if moved {
                    let h = (WIN_H * t.dpi as f32 / 96.0).ceil() as i32;
                    t.origin.1 = (from_top + dy).clamp(t.work.top, (t.work.bottom - h).max(t.work.top));
                    t.press = Some((from_y, from_top, true));
                    render(app, i);
                }
            } else if !t.tracked {
                t.tracked = true;
                t.hover = true;
                let mut tme = TRACKMOUSEEVENT { cbSize: size_of::<TRACKMOUSEEVENT>() as u32, dwFlags: TME_LEAVE, hwndTrack: hwnd, dwHoverTime: 0 };
                unsafe {
                    let _ = TrackMouseEvent(&mut tme);
                }
                render(app, i);
            }
        }
        WM_MOUSELEAVE => {
            let t = &mut app.tabs[i];
            t.tracked = false;
            t.hover = false;
            render(app, i);
        }
        WM_LBUTTONUP => {
            let press = app.tabs[i].press.take();
            unsafe {
                let _ = ReleaseCapture();
            }
            match press {
                Some((_, _, true)) => finish_move(app, i),
                Some(_) => {
                    app.active_tab = i;
                    crate::toggle_stack(app);
                }
                None => {}
            }
        }
        WM_CAPTURECHANGED => {
            if app.tabs[i].press.take().is_some_and(|(_, _, moved)| moved) {
                finish_move(app, i);
            }
        }
        WM_RBUTTONUP => crate::show_context_menu(app),
        WM_MOUSEACTIVATE => return Some(LRESULT(MA_NOACTIVATE as isize)),
        WM_DPICHANGED => position(app, i),
        _ => return None,
    }
    Some(LRESULT(0))
}

//! Direct2D/DirectWrite rendering into per-pixel-alpha layered windows,
//! plus the brand color tokens.

use std::cell::Cell;
use std::ffi::c_void;
use std::time::Instant;

use windows::Foundation::Numerics::Matrix3x2;
use windows::Win32::Foundation::*;
use windows::Win32::Graphics::Direct2D::Common::*;
use windows::Win32::Graphics::Direct2D::*;
use windows::Win32::Graphics::DirectWrite::*;
use windows::Win32::Graphics::Dxgi::Common::*;
use windows::Win32::Graphics::Gdi::*;
use windows::Win32::System::Registry::*;
use windows::Win32::UI::WindowsAndMessaging::*;
use windows::core::*;

#[derive(Clone, Copy)]
pub struct Color(pub f32, pub f32, pub f32, pub f32);

impl Color {
    pub const fn hex(rgb: u32, a: u8) -> Color {
        Color(
            ((rgb >> 16) & 0xff) as f32 / 255.0,
            ((rgb >> 8) & 0xff) as f32 / 255.0,
            (rgb & 0xff) as f32 / 255.0,
            a as f32 / 255.0,
        )
    }
    pub fn alpha(self, a: f32) -> Color {
        Color(self.0, self.1, self.2, self.3 * a)
    }
    fn d2d(self) -> D2D1_COLOR_F {
        D2D1_COLOR_F { r: self.0, g: self.1, b: self.2, a: self.3 }
    }
}

/// True when the taskbar (and tray) is light, so the tray icon must be dark.
pub fn taskbar_is_light() -> bool {
    personalize_flag(w!("SystemUsesLightTheme"))
}

fn personalize_flag(name: PCWSTR) -> bool {
    let mut value: u32 = 0;
    let mut size = size_of::<u32>() as u32;
    let err = unsafe {
        RegGetValueW(
            HKEY_CURRENT_USER,
            w!("Software\\Microsoft\\Windows\\CurrentVersion\\Themes\\Personalize"),
            name,
            RRF_RT_REG_DWORD,
            None,
            Some(&mut value as *mut u32 as *mut c_void),
            Some(&mut size),
        )
    };
    err == ERROR_SUCCESS && value == 1
}

pub struct Gfx {
    pub d2d: ID2D1Factory,
    pub dwrite: IDWriteFactory,
    pub ui_font: HSTRING,
    pub display_font: HSTRING,
    pub icon_font: HSTRING,
    pub code_font: HSTRING,
}

impl Gfx {
    pub fn new() -> Result<Gfx> {
        unsafe {
            let d2d: ID2D1Factory = D2D1CreateFactory(D2D1_FACTORY_TYPE_SINGLE_THREADED, None)?;
            let dwrite: IDWriteFactory = DWriteCreateFactory(DWRITE_FACTORY_TYPE_SHARED)?;
            let mut fonts: Option<IDWriteFontCollection> = None;
            dwrite.GetSystemFontCollection(&mut fonts, false)?;
            let fonts = fonts.unwrap();
            let pick = |names: &[&str]| -> HSTRING {
                for name in names {
                    let h = HSTRING::from(*name);
                    let (mut index, mut exists) = (0u32, BOOL(0));
                    if fonts.FindFamilyName(&h, &mut index, &mut exists).is_ok() && exists.as_bool() {
                        return h;
                    }
                }
                HSTRING::from(*names.last().unwrap())
            };
            Ok(Gfx {
                ui_font: pick(&["Segoe UI Variable Text", "Segoe UI"]),
                display_font: pick(&["Segoe UI Variable Display", "Segoe UI"]),
                icon_font: pick(&["Segoe Fluent Icons", "Segoe MDL2 Assets"]),
                code_font: pick(&["Cascadia Mono", "Consolas"]),
                d2d,
                dwrite,
            })
        }
    }

    /// A one-line format aligned to the leading or trailing edge that
    /// ends overflowing text with an ellipsis.
    pub fn text_format_aligned(
        &self,
        family: &HSTRING,
        size: f32,
        weight: DWRITE_FONT_WEIGHT,
        align: DWRITE_TEXT_ALIGNMENT,
    ) -> Result<IDWriteTextFormat> {
        let f = self.text_format(family, size, weight)?;
        unsafe {
            f.SetTextAlignment(align)?;
            let sign = self.dwrite.CreateEllipsisTrimmingSign(&f)?;
            let trimming = DWRITE_TRIMMING {
                granularity: DWRITE_TRIMMING_GRANULARITY_CHARACTER,
                delimiter: 0,
                delimiterCount: 0,
            };
            f.SetTrimming(&trimming, &sign)?;
        }
        Ok(f)
    }

    pub fn text_format(&self, family: &HSTRING, size: f32, weight: DWRITE_FONT_WEIGHT) -> Result<IDWriteTextFormat> {
        unsafe {
            let f = self.dwrite.CreateTextFormat(
                family,
                None,
                weight,
                DWRITE_FONT_STYLE_NORMAL,
                DWRITE_FONT_STRETCH_NORMAL,
                size,
                w!("en-us"),
            )?;
            f.SetTextAlignment(DWRITE_TEXT_ALIGNMENT_CENTER)?;
            f.SetParagraphAlignment(DWRITE_PARAGRAPH_ALIGNMENT_CENTER)?;
            f.SetWordWrapping(DWRITE_WORD_WRAPPING_NO_WRAP)?;
            Ok(f)
        }
    }

    /// Width in DIPs of a single line of text.
    pub fn text_width(&self, text: &str, format: &IDWriteTextFormat) -> f32 {
        let wide: Vec<u16> = text.encode_utf16().collect();
        unsafe {
            let Ok(layout) = self.dwrite.CreateTextLayout(&wide, format, 10_000.0, 100.0) else { return 0.0 };
            let mut m = DWRITE_TEXT_METRICS::default();
            if layout.GetMetrics(&mut m).is_ok() { m.width } else { 0.0 }
        }
    }
}

/// A 32bpp premultiplied DIB that Direct2D draws into and
/// `UpdateLayeredWindow` presents.
pub struct Surface {
    dc: HDC,
    bitmap: HBITMAP,
    old: HGDIOBJ,
    pub width: i32,
    pub height: i32,
    target: ID2D1DCRenderTarget,
}

impl Surface {
    pub fn new(gfx: &Gfx, width: i32, height: i32) -> Result<Surface> {
        unsafe {
            let dc = CreateCompatibleDC(None);
            let info = BITMAPINFO {
                bmiHeader: BITMAPINFOHEADER {
                    biSize: size_of::<BITMAPINFOHEADER>() as u32,
                    biWidth: width,
                    biHeight: -height,
                    biPlanes: 1,
                    biBitCount: 32,
                    biCompression: BI_RGB.0,
                    ..Default::default()
                },
                ..Default::default()
            };
            let mut bits = std::ptr::null_mut();
            let bitmap = CreateDIBSection(dc, &info, DIB_RGB_COLORS, &mut bits, None, 0)?;
            let old = SelectObject(dc, bitmap);
            let props = D2D1_RENDER_TARGET_PROPERTIES {
                r#type: D2D1_RENDER_TARGET_TYPE_DEFAULT,
                pixelFormat: D2D1_PIXEL_FORMAT {
                    format: DXGI_FORMAT_B8G8R8A8_UNORM,
                    alphaMode: D2D1_ALPHA_MODE_PREMULTIPLIED,
                },
                ..Default::default()
            };
            let target = gfx.d2d.CreateDCRenderTarget(&props)?;
            target.SetTextAntialiasMode(D2D1_TEXT_ANTIALIAS_MODE_GRAYSCALE);
            Ok(Surface { dc, bitmap, old, width, height, target })
        }
    }

    /// Draws with coordinates in DIPs at the given DPI.
    pub fn draw(&self, dpi: u32, f: impl FnOnce(&Painter)) -> Result<()> {
        unsafe {
            let rect = RECT { left: 0, top: 0, right: self.width, bottom: self.height };
            self.target.BindDC(self.dc, &rect)?;
            self.target.SetDpi(dpi as f32, dpi as f32);
            self.target.BeginDraw();
            self.target.Clear(Some(&Color(0.0, 0.0, 0.0, 0.0).d2d()));
            self.target.SetTransform(&Matrix3x2::identity());
            f(&Painter { rt: &self.target, opacity: Cell::new(1.0), offset: Cell::new((0.0, 0.0)) });
            self.target.EndDraw(None, None)
        }
    }

    pub fn present(&self, hwnd: HWND, x: i32, y: i32, opacity: f32) -> Result<()> {
        unsafe {
            let screen = GetDC(None);
            let blend = BLENDFUNCTION {
                BlendOp: AC_SRC_OVER as u8,
                BlendFlags: 0,
                SourceConstantAlpha: (opacity.clamp(0.0, 1.0) * 255.0).round() as u8,
                AlphaFormat: AC_SRC_ALPHA as u8,
            };
            let r = UpdateLayeredWindow(
                hwnd,
                screen,
                Some(&POINT { x, y }),
                Some(&SIZE { cx: self.width, cy: self.height }),
                self.dc,
                Some(&POINT { x: 0, y: 0 }),
                COLORREF(0),
                Some(&blend),
                ULW_ALPHA,
            );
            ReleaseDC(None, screen);
            r
        }
    }

    /// Hands the pixels over as a bitmap the caller owns (for a drag image).
    pub fn into_bitmap(self) -> HBITMAP {
        let me = std::mem::ManuallyDrop::new(self);
        unsafe {
            SelectObject(me.dc, me.old);
            let _ = DeleteDC(me.dc);
            // Releases the render target; the bitmap is not touched.
            drop(std::ptr::read(&me.target));
        }
        me.bitmap
    }

    /// Builds an icon (for the tray) from the current pixels.
    pub fn to_icon(&self) -> Result<HICON> {
        unsafe {
            let mask = CreateBitmap(self.width, self.height, 1, 1, None);
            let info = ICONINFO { fIcon: TRUE, hbmColor: self.bitmap, hbmMask: mask, ..Default::default() };
            let icon = CreateIconIndirect(&info);
            let _ = DeleteObject(mask);
            icon
        }
    }
}

impl Drop for Surface {
    fn drop(&mut self) {
        unsafe {
            SelectObject(self.dc, self.old);
            let _ = DeleteObject(self.bitmap);
            let _ = DeleteDC(self.dc);
        }
    }
}

pub struct Painter<'a> {
    rt: &'a ID2D1RenderTarget,
    /// Multiplies every color and image drawn, for fades.
    opacity: Cell<f32>,
    offset: Cell<(f32, f32)>,
}

fn ellipse(x: f32, y: f32, r: f32) -> D2D1_ELLIPSE {
    D2D1_ELLIPSE { point: D2D_POINT_2F { x, y }, radiusX: r, radiusY: r }
}

impl Painter<'_> {
    fn brush(&self, c: Color) -> Option<ID2D1SolidColorBrush> {
        unsafe { self.rt.CreateSolidColorBrush(&c.alpha(self.opacity.get()).d2d(), None).ok() }
    }

    /// Draws `f` faded to `opacity` and shifted by dx, dy DIPs.
    pub fn layer(&self, opacity: f32, dx: f32, dy: f32, f: impl FnOnce(&Self)) {
        let (o, (ox, oy)) = (self.opacity.get(), self.offset.get());
        self.opacity.set(o * opacity);
        self.offset.set((ox + dx, oy + dy));
        unsafe { self.rt.SetTransform(&Matrix3x2::translation(ox + dx, oy + dy)) };
        f(self);
        self.opacity.set(o);
        self.offset.set((ox, oy));
        unsafe { self.rt.SetTransform(&Matrix3x2::translation(ox, oy)) };
    }

    pub fn fill_circle(&self, x: f32, y: f32, r: f32, c: Color) {
        if let Some(b) = self.brush(c) {
            unsafe { self.rt.FillEllipse(&ellipse(x, y, r), &b) }
        }
    }

    pub fn stroke_circle(&self, x: f32, y: f32, r: f32, width: f32, c: Color) {
        if let Some(b) = self.brush(c) {
            unsafe { self.rt.DrawEllipse(&ellipse(x, y, r), &b, width, None) }
        }
    }

    fn rounded(x: f32, y: f32, w: f32, h: f32, r: f32) -> D2D1_ROUNDED_RECT {
        D2D1_ROUNDED_RECT {
            rect: D2D_RECT_F { left: x, top: y, right: x + w, bottom: y + h },
            radiusX: r,
            radiusY: r,
        }
    }

    pub fn fill_rounded(&self, x: f32, y: f32, w: f32, h: f32, r: f32, c: Color) {
        if let Some(b) = self.brush(c) {
            unsafe { self.rt.FillRoundedRectangle(&Self::rounded(x, y, w, h, r), &b) }
        }
    }

    pub fn stroke_rounded(&self, x: f32, y: f32, w: f32, h: f32, r: f32, width: f32, c: Color) {
        if let Some(b) = self.brush(c) {
            unsafe { self.rt.DrawRoundedRectangle(&Self::rounded(x, y, w, h, r), &b, width, None) }
        }
    }

    pub fn stroke_rounded_dashed(&self, x: f32, y: f32, w: f32, h: f32, r: f32, width: f32, c: Color) {
        let props = D2D1_STROKE_STYLE_PROPERTIES {
            dashStyle: D2D1_DASH_STYLE_CUSTOM,
            dashCap: D2D1_CAP_STYLE_FLAT,
            ..Default::default()
        };
        unsafe {
            let Ok(factory) = self.rt.GetFactory() else { return };
            let Ok(style) = factory.CreateStrokeStyle(&props, Some(&[3.0, 2.0])) else { return };
            if let Some(b) = self.brush(c) {
                self.rt.DrawRoundedRectangle(&Self::rounded(x, y, w, h, r), &b, width, &style);
            }
        }
    }

    /// The Cairn mark (brand/cairn-mark.svg) in a `size` square at x, y.
    pub fn cairn_mark(&self, x: f32, y: f32, size: f32, base: Color, top: Color) {
        // Each stone is a start point and four cubic segments on the 64 grid, copied from the SVG.
        const STONES: [[(f32, f32); 13]; 3] = [
            [(29.50, 40.00), (46.60, 40.00), (58.00, 43.71), (58.00, 49.28), (58.00, 55.06), (54.49, 56.00), (32.94, 56.00), (9.77, 56.00), (6.00, 55.06), (6.00, 49.28), (6.00, 43.71), (15.40, 40.00), (29.50, 40.00)],
            [(32.16, 23.50), (41.67, 23.50), (48.00, 26.63), (48.00, 31.33), (48.00, 36.21), (45.27, 37.00), (28.50, 37.00), (13.45, 37.00), (11.00, 36.21), (11.00, 31.33), (11.00, 26.63), (19.47, 23.50), (32.16, 23.50)],
            [(35.00, 9.59), (41.21, 8.49), (45.81, 10.39), (46.50, 14.33), (47.22, 18.42), (45.89, 19.34), (37.00, 20.91), (28.11, 22.48), (26.54, 22.07), (25.82, 17.98), (25.12, 14.04), (28.80, 10.68), (35.00, 9.59)],
        ];
        let k = size / 64.0;
        let pt = |(px, py): (f32, f32)| D2D_POINT_2F { x: x + px * k, y: y + py * k };
        unsafe {
            let Ok(factory) = self.rt.GetFactory() else { return };
            for (i, s) in STONES.iter().enumerate() {
                let Ok(geo) = factory.CreatePathGeometry() else { return };
                let Ok(sink) = geo.Open() else { return };
                sink.BeginFigure(pt(s[0]), D2D1_FIGURE_BEGIN_FILLED);
                for c in s[1..].chunks(3) {
                    sink.AddBezier(&D2D1_BEZIER_SEGMENT { point1: pt(c[0]), point2: pt(c[1]), point3: pt(c[2]) });
                }
                sink.EndFigure(D2D1_FIGURE_END_CLOSED);
                if sink.Close().is_err() {
                    return;
                }
                if let Some(b) = self.brush(if i == 2 { top } else { base }) {
                    self.rt.FillGeometry(&geo, &b, None);
                }
            }
        }
    }

    pub fn fill_rect(&self, x: f32, y: f32, w: f32, h: f32, c: Color) {
        if let Some(b) = self.brush(c) {
            unsafe { self.rt.FillRectangle(&D2D_RECT_F { left: x, top: y, right: x + w, bottom: y + h }, &b) }
        }
    }

    /// Soft shadow under a rounded card, built from stacked translucent
    /// rounded rectangles that grow outward.
    pub fn rect_shadow(&self, x: f32, y: f32, w: f32, h: f32, r: f32, blur: f32, alpha: f32) {
        const STEPS: usize = 10;
        for i in 0..STEPS {
            let grow = blur * (i as f32 + 1.0) / STEPS as f32;
            let a = alpha * 2.0 / STEPS as f32 * (1.0 - i as f32 / STEPS as f32);
            self.fill_rounded(x - grow, y - grow, w + grow * 2.0, h + grow * 2.0, r + grow, Color(0.0, 0.0, 0.0, a));
        }
    }

    /// Draws premultiplied BGRA pixels into the given box.
    pub fn image(&self, bgra: &[u8], w: u32, h: u32, x: f32, y: f32, dw: f32, dh: f32) {
        let props = D2D1_BITMAP_PROPERTIES {
            pixelFormat: D2D1_PIXEL_FORMAT {
                format: DXGI_FORMAT_B8G8R8A8_UNORM,
                alphaMode: D2D1_ALPHA_MODE_PREMULTIPLIED,
            },
            dpiX: 96.0,
            dpiY: 96.0,
        };
        unsafe {
            let size = D2D_SIZE_U { width: w, height: h };
            let Ok(bitmap) = self.rt.CreateBitmap(size, Some(bgra.as_ptr() as *const _), w * 4, &props) else { return };
            let dest = D2D_RECT_F { left: x, top: y, right: x + dw, bottom: y + dh };
            self.rt.DrawBitmap(&bitmap, Some(&dest), self.opacity.get(), D2D1_BITMAP_INTERPOLATION_MODE_LINEAR, None);
        }
    }

    /// Single line of text centered in the given box.
    pub fn text(&self, s: &str, format: &IDWriteTextFormat, x: f32, y: f32, w: f32, h: f32, c: Color) {
        let wide: Vec<u16> = s.encode_utf16().collect();
        if let Some(b) = self.brush(c) {
            unsafe {
                self.rt.DrawText(
                    &wide,
                    format,
                    &D2D_RECT_F { left: x, top: y, right: x + w, bottom: y + h },
                    &b,
                    D2D1_DRAW_TEXT_OPTIONS_NONE,
                    DWRITE_MEASURING_MODE_NATURAL,
                )
            }
        }
    }
}

/// A value that eases to its target on the brand curve, `cubic-bezier(0,0,0,1)`.
#[derive(Clone, Copy)]
pub struct Tween {
    from: f32,
    to: f32,
    start: Instant,
    ms: f32,
}

impl Tween {
    pub fn new(v: f32) -> Tween {
        Tween { from: v, to: v, start: Instant::now(), ms: 0.0 }
    }

    /// Heads for `to` over `ms` from wherever it is now; 0 ms jumps.
    pub fn set(&mut self, to: f32, ms: f32) {
        if to != self.to {
            *self = Tween { from: self.value(), to, start: Instant::now(), ms };
        }
    }

    /// Starts over from `from`, for one-off effects.
    pub fn run(&mut self, from: f32, to: f32, ms: f32) {
        *self = Tween { from, to, start: Instant::now(), ms };
    }

    fn k(&self) -> f32 {
        if self.ms <= 0.0 { 1.0 } else { (self.start.elapsed().as_secs_f32() * 1000.0 / self.ms).min(1.0) }
    }

    pub fn value(&self) -> f32 {
        self.from + (self.to - self.from) * bezier(0.0, 0.0, 0.0, 1.0, self.k())
    }

    pub fn target(&self) -> f32 {
        self.to
    }

    pub fn done(&self) -> bool {
        self.k() >= 1.0
    }
}

/// CSS-style cubic-bezier easing.
pub fn bezier(x1: f32, y1: f32, x2: f32, y2: f32, t: f32) -> f32 {
    let t = t.clamp(0.0, 1.0);
    let curve = |a: f32, b: f32, u: f32| 3.0 * a * u * (1.0 - u).powi(2) + 3.0 * b * u * u * (1.0 - u) + u.powi(3);
    let (mut lo, mut hi) = (0.0f32, 1.0f32);
    for _ in 0..24 {
        let mid = (lo + hi) / 2.0;
        if curve(x1, x2, mid) < t { lo = mid } else { hi = mid }
    }
    curve(y1, y2, (lo + hi) / 2.0)
}

    pub near: nearzone::NearZone,
    pub settings: Settings,
    /// What "Clear stack" took, while it can still be undone.
    cleared: Vec<shelf::ShelfItem>,
    hotkey: Option<&'static str>,
    /// Whether Ctrl+Alt+V (add clipboard) could be registered.
    clip_hotkey: bool,
    /// Where the zip being made will be saved.
    pub zip_out: Option<std::path::PathBuf>,
    tray_icon: HICON,
    taskbar_created: u32,
}

thread_local! {
    static APP: Cell<*mut App> = const { Cell::new(std::ptr::null_mut()) };
}

/// The app lives for the whole message loop on the UI thread.
fn app() -> Option<&'static mut App> {
    let p = APP.with(|a| a.get());
    unsafe { p.as_mut() }
}

/// For COM callbacks (screen readers), which run on the UI thread too.
pub fn app_ref() -> Option<&'static App> {
    app().map(|a| &*a)
}

pub fn app_mut() -> Option<&'static mut App> {
    app()
}

pub fn dpi_at(pt: POINT) -> u32 {
    unsafe {
        let mon = MonitorFromPoint(pt, MONITOR_DEFAULTTONEAREST);
        let (mut x, mut y) = (96u32, 96u32);
        let _ = GetDpiForMonitor(mon, MDT_EFFECTIVE_DPI, &mut x, &mut y);
        x
    }
}

fn animations_enabled() -> bool {
    let mut on = TRUE;
    unsafe {
        let _ = SystemParametersInfoW(
            SPI_GETCLIENTAREAANIMATION,
            0,
            Some(&mut on as *mut BOOL as *mut _),
            SYSTEM_PARAMETERS_INFO_UPDATE_FLAGS(0),
        );
    }
    on.as_bool()
}

fn wide<const N: usize>(s: &str) -> [u16; N] {
    let mut out = [0u16; N];
    for (o, c) in out.iter_mut().zip(s.encode_utf16().take(N - 1)) {
        *o = c;
    }
    out
}

fn tray_data(app: &App) -> NOTIFYICONDATAW {
    NOTIFYICONDATAW { cbSize: size_of::<NOTIFYICONDATAW>() as u32, hWnd: app.host, uID: 1, ..Default::default() }
}

fn tray_tip(app: &App) -> [u16; 128] {
    wide(&match app.hotkey {
        Some(k) => format!("Cairn ({k})"),
        None => "Cairn".to_string(),
    })
}

fn tray_add(app: &App) {
    let mut nid = tray_data(app);
    nid.uFlags = NIF_ICON | NIF_MESSAGE | NIF_TIP;
    nid.uCallbackMessage = WM_TRAY;
    nid.hIcon = app.tray_icon;
    nid.szTip = tray_tip(app);
    unsafe {
        let _ = Shell_NotifyIconW(NIM_ADD, &nid);
    }
}

fn tray_update(app: &App) {
    let mut nid = tray_data(app);
    nid.uFlags = NIF_ICON | NIF_TIP;
    nid.hIcon = app.tray_icon;
    nid.szTip = tray_tip(app);
    unsafe {
        let _ = Shell_NotifyIconW(NIM_MODIFY, &nid);
    }
}

pub fn notify(app: &App, title: &str, text: &str) {
    let mut nid = tray_data(app);
    nid.uFlags = NIF_INFO;
    nid.szInfoTitle = wide(title);
    nid.szInfo = wide(text);
    nid.dwInfoFlags = NIIF_NONE | NIIF_NOSOUND;
    unsafe {
        let _ = Shell_NotifyIconW(NIM_MODIFY, &nid);
    }
}

/// The one-color mark (brand/cairn-mono.svg), white on a dark taskbar and
/// near-black on a light one.
fn make_tray_icon(gfx: &Gfx) -> HICON {
    let dpi = dpi_at(POINT::default());
    let side = unsafe { GetSystemMetricsForDpi(SM_CXSMICON, dpi) };
    let Ok(surf) = Surface::new(gfx, side, side) else { return HICON::default() };
    let ink = if gfx::taskbar_is_light() { Color::hex(0x191b1d, 0xff) } else { Color::hex(0xffffff, 0xff) };
    let k = side as f32 / 16.0;
    let _ = surf.draw(96, |p| {
        for (x, y, w) in [(1.0, 11.0, 14.0), (2.0, 6.0, 10.0), (7.0, 1.0, 6.0)] {
            p.fill_rounded(x * k, y * k, w * k, 4.0 * k, 2.0 * k, ink);
        }
    });
    surf.to_icon().unwrap_or_default()
}

pub fn toggle_stack(app: &mut App) {
    if app.stack_view.open {
        stackview::close(app, true);
    } else {
        // Open beside the tab on the monitor the user is working on.
        app.active_tab = edgetab::under_cursor(app);
        let target = unsafe { GetForegroundWindow() };
        stackview::open(app, target);
    }
}

fn tab_index(app: &App, hwnd: HWND) -> Option<usize> {
    app.tabs.iter().position(|t| t.hwnd == hwnd)
}

/// A drag carrying something Cairn accepts is over a tab or the stack, or
/// has left it.
pub fn on_drag_hover(zone: dnd::Zone, over: bool) {
    let Some(app) = app() else { return };
    match zone {
        dnd::Zone::Tab(hwnd) => {
            if let Some(i) = tab_index(app, hwnd) {
                if over {
                    app.active_tab = i;
                }
                edgetab::set_drag_over(app, i, over);
            }
            unsafe {
                if over {
                    // Resting on the tab opens the stack, so the drop can go
                    // into a stack of the user's choice.
                    SetTimer(app.host, SPRING_TIMER, SPRING_MS, None);
                    let _ = KillTimer(app.host, DRAG_CLOSE_TIMER);
                } else {
                    let _ = KillTimer(app.host, SPRING_TIMER);
                    close_drag_stack_soon(app, 400);
                }
            }
        }
        dnd::Zone::Stack => {
            stackview::set_drop_hover(app, over);
            if over {
                unsafe {
                    let _ = KillTimer(app.host, DRAG_CLOSE_TIMER);
                }
            } else {
                close_drag_stack_soon(app, 400);
            }
        }
        dnd::Zone::Near => nearzone::set_hover(app, over),
    }
}

/// The drag moved: over the stack, pointing at a stack name switches to it.
pub fn on_drag_move(zone: dnd::Zone, pt: POINT) {
    let Some(app) = app() else { return };
    if zone == dnd::Zone::Stack && app.stack_view.open {
        stackview::drag_over(app, pt);
    }
}

/// The stack a drop would land in, for the label under the drag image.
pub fn drop_label() -> String {
    app().map(|a| a.stack.names[a.stack.current].clone()).unwrap_or_default()
}

/// A stack opened by a drag closes by itself once the drag is gone.
fn close_drag_stack_soon(app: &App, ms: u32) {
    if app.stack_view.dragging_in {
        unsafe {
            SetTimer(app.host, DRAG_CLOSE_TIMER, ms, None);
        }
    }
}

pub fn on_drop(zone: dnd::Zone, dropped: shelf::Dropped) {
    let Some(app) = app() else { return };
    unsafe {
        let _ = KillTimer(app.host, SPRING_TIMER);
    }
    let added = app.stack.add(dropped);
    if added > 0 {
        app.stack.save();
        copy_from_drives(app, added);
    }
    match zone {
        dnd::Zone::Tab(hwnd) => {
            if let Some(i) = tab_index(app, hwnd) {
                edgetab::dropped(app, i, added);
            }
        }
        dnd::Zone::Near => nearzone::dropped(app, added),
        dnd::Zone::Stack => {}
    }
    if app.stack_view.open {
        stackview::added(app, added);
    }
    // Leave the new item in view for a moment before closing.
    close_drag_stack_soon(app, 900);
}

/// USB sticks, card readers, discs and network shares: files there may be
/// gone the next time the stack is used.
fn on_unreliable_drive(path: &std::path::Path) -> bool {
    let s = path.to_string_lossy();
    if s.starts_with("\\\\") {
        return true;
    }
    let Some(root) = s.get(..3).filter(|r| r.ends_with(":\\")) else { return false };
    let kind = unsafe { windows::Win32::Storage::FileSystem::GetDriveTypeW(&HSTRING::from(root)) };
    matches!(kind, 2 | 4 | 5) // DRIVE_REMOVABLE, DRIVE_REMOTE, DRIVE_CDROM
}

/// With the setting on, copies newly added files from removable and network
/// drives into Cairn's folder in the background, then points the items at
/// the copies.
fn copy_from_drives(app: &App, added: usize) {
    if !app.settings.drive_copies {
        return;
    }
    let host = app.host.0 as isize;
    for item in &app.stack.items[..added.min(app.stack.items.len())] {
        if let shelf::Kind::File(p) = &item.kind
            && p.is_file()
            && on_unreliable_drive(p)
        {
            let (id, src) = (item.id, p.clone());
            std::thread::spawn(move || {
                let copy = Box::new(shelf::copy_in(&src));
                unsafe {
                    let _ = PostMessageW(HWND(host as *mut _), WM_COPY_DONE, WPARAM(id as usize), LPARAM(Box::into_raw(copy) as isize));
                }
            });
        }
    }
}

/// Sends Ctrl+V once focus has settled back in the target app.
pub fn schedule_paste(app: &App) {
    unsafe {
        SetTimer(app.host, PASTE_TIMER, 60, None);
    }
}

/// Empties the stack and offers undo for 10 seconds: click the notification
/// or choose "Undo clear" from the tray menu.
fn clear_stack(app: &mut App) {
    let cleared = app.stack.clear();
    if cleared.is_empty() {
        return;
    }
    app.stack.save();
    stackview::close(app, false);
    offer_undo(app, cleared, "Stack cleared");
}

/// Keeps removed items for 10 seconds: clicking the notification or "Undo"
/// in the tray menu puts them back in the current stack.
pub fn offer_undo(app: &mut App, items: Vec<shelf::ShelfItem>, title: &str) {
    let n = items.len();
    app.cleared = items;
    let what = if n == 1 { "1 item".to_string() } else { format!("{n} items") };
    notify(app, title, &format!("Removed {what}. Click here within 10 seconds to undo."));
    unsafe {
        SetTimer(app.host, UNDO_TIMER, UNDO_MS, None);
    }
}

fn undo_clear(app: &mut App) {
    unsafe {
        let _ = KillTimer(app.host, UNDO_TIMER);
    }
    if app.cleared.is_empty() {
        return;
    }
    let back = app.stack.restore(std::mem::take(&mut app.cleared));
    app.stack.save();
    if app.stack_view.open {
        stackview::refresh(app);
    }
    let what = if back == 1 { "1 item is".to_string() } else { format!("{back} items are") };
    let name = &app.stack.names[app.stack.current];
    notify(app, "Undone", &format!("{what} back in {name}."));
}

/// Adds whatever is on the clipboard (files, a link, text or an image) to the
/// current stack, and shows the result on the tab under the cursor.
fn add_clipboard(app: &mut App) {
    let dropped = unsafe { OleGetClipboard() }.ok().and_then(|data| dnd::read(&data));
    let Some(dropped) = dropped else {
        notify(app, "Nothing to add", "The clipboard holds no files, links, text or images.");
        return;
    };
    let added = app.stack.add(dropped);
    if added > 0 {
        app.stack.save();
        copy_from_drives(app, added);
    }
    if app.stack_view.open {
        stackview::added(app, added);
    } else {
        let i = edgetab::under_cursor(app);
        edgetab::dropped(app, i, added);
    }
}

fn register_hotkey(host: HWND, i: usize) -> bool {
    let h = &settings::HOTKEYS[i];
    unsafe { RegisterHotKey(host, HOTKEY_ID, h.mods | MOD_NOREPEAT, h.vk.0 as u32).is_ok() }
}

/// Switches the global hotkey, keeping the old one if the new one is taken.
fn choose_hotkey(app: &mut App, i: usize) {
    let label = settings::HOTKEYS[i].label;
    if app.hotkey == Some(label) {
        return;
    }
    unsafe {
        let _ = UnregisterHotKey(app.host, HOTKEY_ID);
    }
    if register_hotkey(app.host, i) {
        app.hotkey = Some(label);
        app.settings.hotkey = i;
        app.settings.save();
        tray_update(app);
    } else {
        if let Some(o) = app.hotkey.and_then(|l| settings::HOTKEYS.iter().position(|h| h.label == l)) {
            register_hotkey(app.host, o);
        }
        notify(app, "Hotkey unavailable", &format!("{label} is in use by another app. Cairn kept {}.", app.hotkey.unwrap_or("no hotkey")));
    }
}

fn set_edge(app: &mut App, edge: Edge) {
    if app.settings.edge != edge {
        app.settings.edge = edge;
        app.settings.save();
        stackview::close(app, false);
        edgetab::sync(app);
    }
}

fn center_tab(app: &mut App) {
    app.settings.tab_pos = 0.5;
    app.settings.save();
    stackview::close(app, false);
    edgetab::sync(app);
}

/// Turns the tab on or off for one monitor, keeping at least one on.
fn toggle_monitor(app: &mut App, device: String) {
    let off = &mut app.settings.monitors_off;
    if let Some(at) = off.iter().position(|d| *d == device) {
        off.remove(at);
    } else {
        let on = edgetab::monitors().iter().filter(|m| !off.contains(&m.device)).count();
        if on <= 1 {
            notify(app, "Monitors", "Cairn needs the tab on at least one monitor.");
            return;
        }
        off.push(device);
    }
    app.settings.save();
    stackview::close(app, false);
    edgetab::sync(app);
}

pub fn show_context_menu(app: &mut App) {
    stackview::close(app, false);
    unsafe {
        let Ok(popup) = CreatePopupMenu() else { return };
        let open = HSTRING::from(format!("Open stack\t{}", app.hotkey.unwrap_or_default()));
        let _ = AppendMenuW(popup, MF_STRING, CMD_OPEN as usize, &open);
        let clip = HSTRING::from(match app.clip_hotkey {
            true => format!("Add clipboard to stack\t{CLIP_HOTKEY}"),
            false => "Add clipboard to stack".to_string(),
        });
        let _ = AppendMenuW(popup, MF_STRING, CMD_ADD_CLIPBOARD as usize, &clip);
        if app.cleared.is_empty() {
            let empty = if app.stack.items.is_empty() { MF_GRAYED } else { MF_ENABLED };
            let _ = AppendMenuW(popup, MF_STRING | empty, CMD_CLEAR as usize, w!("Clear stack"));
        } else {
            let _ = AppendMenuW(popup, MF_STRING, CMD_UNDO_CLEAR as usize, w!("Undo"));
        }
        let _ = AppendMenuW(popup, MF_SEPARATOR, 0, None);
        if let Ok(keys) = CreatePopupMenu() {
            for (i, h) in settings::HOTKEYS.iter().enumerate() {
                let checked = if app.hotkey == Some(h.label) { MF_CHECKED } else { MF_UNCHECKED };
                let _ = AppendMenuW(keys, MF_STRING | checked, CMD_HOTKEY_BASE as usize + i, &HSTRING::from(h.label));
            }
            let _ = AppendMenuW(popup, MF_POPUP, keys.0 as usize, w!("Hotkey"));
        }
        if let Ok(edges) = CreatePopupMenu() {
            let check = |e: Edge| if app.settings.edge == e { MF_CHECKED } else { MF_UNCHECKED };
            let _ = AppendMenuW(edges, MF_STRING | check(Edge::Left), CMD_EDGE_LEFT as usize, w!("Left"));
            let _ = AppendMenuW(edges, MF_STRING | check(Edge::Right), CMD_EDGE_RIGHT as usize, w!("Right"));
            let _ = AppendMenuW(edges, MF_SEPARATOR, 0, None);
            let _ = AppendMenuW(edges, MF_STRING, CMD_CENTER_TAB as usize, w!("Center the tab"));
            let _ = AppendMenuW(popup, MF_POPUP, edges.0 as usize, w!("Screen edge"));
        }
        let monitors = edgetab::monitors();
        if monitors.len() > 1
            && let Ok(mons) = CreatePopupMenu()
        {
            for (i, m) in monitors.iter().enumerate() {
                let checked = if app.settings.monitors_off.contains(&m.device) { MF_UNCHECKED } else { MF_CHECKED };
                let (w, h) = (m.work.right - m.work.left, m.work.bottom - m.work.top);
                let main = if m.primary { ", main" } else { "" };
                let label = HSTRING::from(format!("Display {} ({w} × {h}{main})", i + 1));
                let _ = AppendMenuW(mons, MF_STRING | checked, CMD_MONITOR_BASE as usize + i, &label);
            }
            let _ = AppendMenuW(popup, MF_POPUP, mons.0 as usize, w!("Show tab on"));
        }
        let login = if settings::run_at_login() { MF_CHECKED } else { MF_UNCHECKED };
        let check = |on: bool| if on { MF_CHECKED } else { MF_UNCHECKED };
        let _ = AppendMenuW(popup, MF_STRING | check(app.settings.shake), CMD_SHAKE as usize, w!("Shake while dragging to show a drop zone"));
        let _ = AppendMenuW(popup, MF_STRING | check(app.settings.drive_copies), CMD_DRIVE_COPIES as usize, w!("Keep copies of files from USB and network drives"));
        let _ = AppendMenuW(popup, MF_STRING | login, CMD_RUN_AT_LOGIN as usize, w!("Run at login"));
        let _ = AppendMenuW(popup, MF_SEPARATOR, 0, None);
        let _ = AppendMenuW(popup, MF_STRING, CMD_QUIT as usize, w!("Quit Cairn"));
        let mut pt = POINT::default();
        let _ = GetCursorPos(&mut pt);
        let _ = SetForegroundWindow(app.host);
        let cmd = TrackPopupMenu(popup, TPM_RETURNCMD | TPM_RIGHTBUTTON | TPM_NONOTIFY, pt.x, pt.y, 0, app.host, None);
        let _ = PostMessageW(app.host, WM_NULL, WPARAM(0), LPARAM(0));
        let _ = DestroyMenu(popup);
        match cmd.0 as u32 {
            CMD_OPEN => toggle_stack(app),
            CMD_ADD_CLIPBOARD => add_clipboard(app),
            CMD_CLEAR => clear_stack(app),
            CMD_UNDO_CLEAR => undo_clear(app),
            CMD_EDGE_LEFT => set_edge(app, Edge::Left),
            CMD_EDGE_RIGHT => set_edge(app, Edge::Right),
            CMD_CENTER_TAB => center_tab(app),
            c if (CMD_MONITOR_BASE..CMD_MONITOR_BASE + monitors.len() as u32).contains(&c) => {
                toggle_monitor(app, monitors[(c - CMD_MONITOR_BASE) as usize].device.clone())
            }
            CMD_DRIVE_COPIES => {
                app.settings.drive_copies = !app.settings.drive_copies;
                app.settings.save();
            }
            CMD_SHAKE => {
                app.settings.shake = !app.settings.shake;
                app.settings.save();
                nearzone::set_enabled(app, app.settings.shake);
            }
            CMD_RUN_AT_LOGIN => {
                let on = !settings::run_at_login();
                if !settings::set_run_at_login(on) {
                    notify(app, "Run at login", "Windows did not accept the change. Try again.");
                }
            }
            c if (CMD_HOTKEY_BASE..CMD_HOTKEY_BASE + settings::HOTKEYS.len() as u32).contains(&c) => {
                choose_hotkey(app, (c - CMD_HOTKEY_BASE) as usize)
            }
            CMD_QUIT => quit(app),
            _ => {}
        }
    }
}

fn quit(app: &mut App) {
    unsafe {
        let _ = UnregisterHotKey(app.host, HOTKEY_ID);
        let _ = UnregisterHotKey(app.host, CLIP_HOTKEY_ID);
        let _ = Shell_NotifyIconW(NIM_DELETE, &tray_data(app));
        app.stack.prune_owned(&[]);
        PostQuitMessage(0);
    }
}

/// The monitor of the foreground window, if that window is a real app.
fn foreground_monitor() -> Option<(HWND, HMONITOR)> {
    unsafe {
        let fg = GetForegroundWindow();
        if fg.is_invalid() || fg == GetShellWindow() || fg == GetDesktopWindow() {
            return None;
        }
        // Click-through overlays (recorders, assistants) are not apps.
        if GetWindowLongW(fg, GWL_EXSTYLE) as u32 & WS_EX_TRANSPARENT.0 != 0 {
            return None;
        }
        Some((fg, MonitorFromWindow(fg, MONITOR_DEFAULTTONEAREST)))
    }
}

fn covers_monitor(fg: HWND, mon: HMONITOR) -> bool {
    unsafe {
        let mut rect = RECT::default();
        if GetWindowRect(fg, &mut rect).is_err() {
            return false;
        }
        let mut info = MONITORINFO { cbSize: size_of::<MONITORINFO>() as u32, ..Default::default() };
        let _ = GetMonitorInfoW(mon, &mut info);
        let m = info.rcMonitor;
        rect.left <= m.left && rect.top <= m.top && rect.right >= m.right && rect.bottom >= m.bottom
    }
}

/// Hides the tab on the monitor where a fullscreen app, game or
/// presentation is up, and keeps the tabs above other topmost windows.
fn poll_fullscreen(app: &mut App) {
    let fg = foreground_monitor();
    let busy = match unsafe { SHQueryUserNotificationState() } {
        Ok(QUNS_RUNNING_D3D_FULL_SCREEN | QUNS_PRESENTATION_MODE) => fg.map(|(_, m)| m),
        // QUNS_BUSY also fires for click-through overlays, so require the
        // foreground window itself to cover its monitor.
        Ok(QUNS_BUSY) => fg.filter(|&(w, m)| covers_monitor(w, m)).map(|(_, m)| m),
        _ => None,
    };
    for i in 0..app.tabs.len() {
        let hide = busy == Some(app.tabs[i].monitor);
        if hide != app.tabs[i].fullscreen_hidden {
            app.tabs[i].fullscreen_hidden = hide;
            if hide && app.active_tab == i {
                stackview::close(app, false);
            }
            edgetab::show(app, i);
        }
    }
    edgetab::raise(app);
}

fn host_handle(app: &mut App, msg: u32, wparam: WPARAM, lparam: LPARAM) -> Option<LRESULT> {
    match msg {
        WM_HOTKEY if wparam.0 as i32 == HOTKEY_ID => toggle_stack(app),
        WM_HOTKEY if wparam.0 as i32 == CLIP_HOTKEY_ID => add_clipboard(app),
        nearzone::WM_SHAKE if !app.near.shown && !app.stack_view.open => {
            let pt = POINT { x: (lparam.0 & 0xffff_ffff) as u32 as i32, y: (lparam.0 >> 32) as i32 };
            nearzone::show_at(app, pt);
        }
        nearzone::WM_SHAKE_END => nearzone::hide_soon(app, 500),
        WM_COPY_DONE => {
            let copy = unsafe { Box::from_raw(lparam.0 as *mut Option<std::path::PathBuf>) };
            match *copy {
                Some(path) if app.stack.replace_path(wparam.0 as u64, path.clone()) => {
                    app.stack.save();
                    if app.stack_view.open {
                        stackview::refresh(app);
                    }
                }
                // The item was removed while copying.
                Some(path) => {
                    let _ = std::fs::remove_file(path);
                }
                None => notify(app, "Copy failed", "Cairn could not keep a copy of a file from a removable or network drive. The item still points at the drive."),
            }
        }
        actions::WM_ZIP_DONE => match wparam.0 {
            1 => {
                let note = app.zip_out.take().map_or("The zip is ready.".to_string(), |p| format!("Saved {}", p.display()));
                notify(app, "Zip saved", &note)
            }
            _ => notify(app, "Zip failed", "Windows could not create the zip. Check the folder can be written to."),
        },
        WM_TRAY => match lparam.0 as u32 {
            WM_LBUTTONUP => toggle_stack(app),
            WM_RBUTTONUP => show_context_menu(app),
            NIN_BALLOONUSERCLICK if !app.cleared.is_empty() => undo_clear(app),
            _ => {}
        },
        WM_TIMER if wparam.0 == SPRING_TIMER => {
            unsafe {
                let _ = KillTimer(app.host, SPRING_TIMER);
            }
            if edgetab::is_drag_over(app, app.active_tab) {
                stackview::open_for_drag(app);
            }
        }
        WM_TIMER if wparam.0 == DRAG_CLOSE_TIMER => {
            unsafe {
                let _ = KillTimer(app.host, DRAG_CLOSE_TIMER);
            }
            let still_dragging = app.stack_view.drop_hover || (0..app.tabs.len()).any(|i| edgetab::is_drag_over(app, i));
            if app.stack_view.dragging_in && !still_dragging {
                stackview::close(app, false);
            }
        }
        WM_TIMER if wparam.0 == UNDO_TIMER => {
            unsafe {
                let _ = KillTimer(app.host, UNDO_TIMER);
            }
            app.cleared.clear();
            app.stack.prune_owned(&[]);
        }
        WM_TIMER if wparam.0 == FULLSCREEN_TIMER => poll_fullscreen(app),
        WM_TIMER if wparam.0 == PASTE_TIMER => {
            unsafe {
                let _ = KillTimer(app.host, PASTE_TIMER);
            }
            clipboard::send_paste();
        }
        WM_SETTINGCHANGE => {
            app.theme = Theme::system();
            app.animations = animations_enabled();
            let old = app.tray_icon;
            app.tray_icon = make_tray_icon(&app.gfx);
            tray_update(app);
            unsafe {
                let _ = DestroyIcon(old);
            }
            // The taskbar may have moved, changing the work areas.
            edgetab::sync(app);
        }
        WM_DISPLAYCHANGE => {
            stackview::close(app, false);
            edgetab::sync(app);
        }
        m if m == app.taskbar_created && m != 0 => tray_add(app),
        _ => return None,
    }
    Some(LRESULT(0))
}

extern "system" fn wndproc(hwnd: HWND, msg: u32, wparam: WPARAM, lparam: LPARAM) -> LRESULT {
    if let Some(app) = app() {
        let handled = if let Some(i) = tab_index(app, hwnd) {
            edgetab::handle(app, i, hwnd, msg, wparam, lparam)
        } else if hwnd == app.near.hwnd {
            nearzone::handle(app, msg, wparam)
        } else if hwnd == app.stack_view.hwnd {
            stackview::handle(app, hwnd, msg, wparam, lparam)
        } else if hwnd == app.host {
            host_handle(app, msg, wparam, lparam)
        } else {
            None
        };
        if let Some(r) = handled {
            return r;
        }
    }
    unsafe { DefWindowProcW(hwnd, msg, wparam, lparam) }
}

pub fn create_window(class: PCWSTR, ex: WINDOW_EX_STYLE) -> Result<HWND> {
    unsafe {
        let instance = GetModuleHandleW(None)?;
        let wc = WNDCLASSW {
            lpfnWndProc: Some(wndproc),
            hInstance: instance.into(),
            lpszClassName: class,
            hCursor: LoadCursorW(None, IDC_ARROW)?,
            ..Default::default()
        };
        RegisterClassW(&wc);
        CreateWindowExW(ex, class, w!("Cairn"), WS_POPUP, 0, 0, 0, 0, None, None, instance, None)
    }
}

fn main() -> Result<()> {
    unsafe {
        let _mutex = CreateMutexW(None, true, w!("Local\\Cairn.SingleInstance"))?;
        if GetLastError() == ERROR_ALREADY_EXISTS {
            return Ok(());
        }
        let _ = SetProcessDpiAwarenessContext(DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2);
        // Drag and drop needs OLE on this (the UI) thread.
        OleInitialize(None)?;

        let floating = WS_EX_LAYERED | WS_EX_TOPMOST | WS_EX_TOOLWINDOW;
        let host = create_window(w!("CairnHost"), WS_EX_TOOLWINDOW)?;
        let stack_hwnd = create_window(w!("CairnStack"), floating)?;
        let near_hwnd = create_window(w!("CairnNear"), floating | WS_EX_NOACTIVATE)?;

        let gfx = Gfx::new()?;
        let tray_icon = make_tray_icon(&gfx);
        let app = Box::leak(Box::new(App {
            gfx,
            theme: Theme::system(),
            animations: animations_enabled(),
            host,
            tabs: Vec::new(),
            active_tab: 0,
            stack: shelf::Shelf::load(),
            stack_view: stackview::StackView::new(stack_hwnd),
            near: nearzone::NearZone::new(near_hwnd),
            settings: Settings::load(),
            cleared: Vec::new(),
            hotkey: None,
            clip_hotkey: false,
            zip_out: None,
            tray_icon,
            taskbar_created: RegisterWindowMessageW(w!("TaskbarCreated")),
        }));
        APP.with(|a| a.set(app as *mut App));

        // The saved hotkey first, then the others. A fallback becomes the
        // saved choice so the notice shows only once.
        let preferred = app.settings.hotkey;
        let chosen = std::iter::once(preferred)
            .chain((0..settings::HOTKEYS.len()).filter(|&i| i != preferred))
            .find(|&i| register_hotkey(host, i));
        app.hotkey = chosen.map(|i| settings::HOTKEYS[i].label);
        app.clip_hotkey = RegisterHotKey(host, CLIP_HOTKEY_ID, MOD_CONTROL | MOD_ALT | MOD_NOREPEAT, VK_V.0 as u32).is_ok();

        tray_add(app);
        edgetab::sync(app);
        let _ = dnd::register(stack_hwnd, dnd::Zone::Stack);
        nearzone::register(near_hwnd);
        nearzone::set_enabled(app, app.settings.shake);
        // Copies nothing refers to any more, from removals since the last run.
        app.stack.prune_owned(&[]);
        SetTimer(host, FULLSCREEN_TIMER, 1000, None);
        match chosen {
            None => notify(app, "Hotkey unavailable", "Every Cairn hotkey is in use by another app. Use the edge tab or the tray icon."),
            Some(i) if i != preferred => {
                let (was, now) = (settings::HOTKEYS[preferred].label, settings::HOTKEYS[i].label);
                notify(app, "Hotkey changed", &format!("{was} is in use by another app, so Cairn uses {now}. Change it from the tray icon."));
                app.settings.hotkey = i;
                app.settings.save();
            }
            Some(_) => {}
        }

        let mut msg = MSG::default();
        while GetMessageW(&mut msg, None, 0, 0).as_bool() {
            let _ = TranslateMessage(&msg);
            DispatchMessageW(&msg);
        }
    }
    Ok(())
}
//! Shake to show (opt-in): while the left button is held, shaking the mouse
//! left and right shows a drop zone next to the cursor, so you can drop
//! without travelling to the screen edge.
//!
//! Seeing a shake needs a low-level mouse hook, which only runs while the
//! setting is on. The hook only watches the button and horizontal movement;
//! it never blocks or changes input.

use std::cell::Cell;
use std::time::Instant;

use windows::Win32::Foundation::*;
use windows::Win32::Graphics::DirectWrite::*;
use windows::Win32::Graphics::Gdi::*;
use windows::Win32::System::LibraryLoader::GetModuleHandleW;
use windows::Win32::UI::WindowsAndMessaging::*;

use crate::gfx::Surface;
use crate::{App, dnd, dpi_at};

const WIN_W: f32 = 120.0;
const WIN_H: f32 = 180.0;
const CARD_W: f32 = 104.0;
const CARD_H: f32 = 164.0;
/// How far right of the cursor the zone opens, in DIPs.
const OFFSET: f32 = 56.0;
const HIDE_TIMER: usize = 1;

/// Posted to the host by the hook: a shake was seen (lParam holds the
/// cursor), or the button was released.
pub const WM_SHAKE: u32 = WM_APP + 3;
pub const WM_SHAKE_END: u32 = WM_APP + 4;

/// Direction changes, each after at least this much travel...
const MIN_SWING: i32 = 24;
/// ...needed within this window to count as a shake.
const REVERSALS: usize = 4;
const WINDOW_MS: u128 = 700;

/// Recognises a shake from a stream of x positions while the button is held.
#[derive(Default)]
pub struct Shake {
    /// Furthest x in the current direction, and the direction (+1 or -1).
    edge: Option<(i32, i32)>,
    turns: Vec<Instant>,
}

impl Shake {
    pub fn reset(&mut self) {
        self.edge = None;
        self.turns.clear();
    }

    /// Feeds one x position; true when this move completes a shake.
    pub fn feed(&mut self, x: i32, now: Instant) -> bool {
        let Some((far, dir)) = self.edge else {
            self.edge = Some((x, 0));
            return false;
        };
        let d = x - far;
        if dir == 0 {
            if d.abs() >= MIN_SWING {
                self.edge = Some((x, d.signum()));
            }
            return false;
        }
        if d.signum() == dir {
            self.edge = Some((x, dir));
        } else if d.abs() >= MIN_SWING {
            // Swung back far enough: a reversal.
            self.edge = Some((x, -dir));
            self.turns.retain(|t| now.duration_since(*t).as_millis() <= WINDOW_MS);
            self.turns.push(now);
            if self.turns.len() >= REVERSALS {
                self.turns.clear();
                return true;
            }
        }
        false
    }
}

thread_local! {
    static HOOK: Cell<HHOOK> = const { Cell::new(HHOOK(std::ptr::null_mut())) };
    static HOST: Cell<isize> = const { Cell::new(0) };
    static DOWN: Cell<bool> = const { Cell::new(false) };
    static SHAKE: std::cell::RefCell<Shake> = std::cell::RefCell::new(Shake::default());
}

unsafe extern "system" fn mouse_hook(code: i32, wparam: WPARAM, lparam: LPARAM) -> LRESULT {
    if code == HC_ACTION as i32 {
        let info = unsafe { &*(lparam.0 as *const MSLLHOOKSTRUCT) };
        let host = HWND(HOST.get() as *mut _);
        match wparam.0 as u32 {
            WM_LBUTTONDOWN => {
                DOWN.set(true);
                SHAKE.with_borrow_mut(Shake::reset);
            }
            WM_LBUTTONUP => {
                DOWN.set(false);
                unsafe {
                    let _ = PostMessageW(host, WM_SHAKE_END, WPARAM(0), LPARAM(0));
                }
            }
            WM_MOUSEMOVE if DOWN.get() => {
                if SHAKE.with_borrow_mut(|s| s.feed(info.pt.x, Instant::now())) {
                    let packed = ((info.pt.y as u32 as isize) << 32) | info.pt.x as u32 as isize;
                    unsafe {
                        let _ = PostMessageW(host, WM_SHAKE, WPARAM(0), LPARAM(packed));
                    }
                }
            }
            _ => {}
        }
    }
    unsafe { CallNextHookEx(None, code, wparam, lparam) }
}

pub struct NearZone {
    pub hwnd: HWND,
    surface: Option<Surface>,
    dpi: u32,
    origin: (i32, i32),
    pub shown: bool,
    hover: bool,
    flash: Option<String>,
}

impl NearZone {
    pub fn new(hwnd: HWND) -> NearZone {
        NearZone { hwnd, surface: None, dpi: 96, origin: (0, 0), shown: false, hover: false, flash: None }
    }
}

/// Turns the hook on or off to match the setting.
pub fn set_enabled(app: &App, on: bool) {
    HOST.set(app.host.0 as isize);
    let current = HOOK.get();
    unsafe {
        if on && current.is_invalid() {
            let module = GetModuleHandleW(None).unwrap_or_default();
            if let Ok(h) = SetWindowsHookExW(WH_MOUSE_LL, Some(mouse_hook), module, 0) {
                HOOK.set(h);
            }
        } else if !on && !current.is_invalid() {
            let _ = UnhookWindowsHookEx(current);
            HOOK.set(HHOOK(std::ptr::null_mut()));
        }
    }
}

/// Opens the zone beside the cursor, inside the work area.
pub fn show_at(app: &mut App, pt: POINT) {
    let dpi = dpi_at(pt);
    let s = dpi as f32 / 96.0;
    let (w, h) = ((WIN_W * s).ceil() as i32, (WIN_H * s).ceil() as i32);
    let wa = unsafe {
        let mut info = MONITORINFO { cbSize: size_of::<MONITORINFO>() as u32, ..Default::default() };
        let _ = GetMonitorInfoW(MonitorFromPoint(pt, MONITOR_DEFAULTTONEAREST), &mut info);
        info.rcWork
    };
    let mut x = pt.x + (OFFSET * s) as i32;
    if x + w > wa.right {
        // No room on the right: open on the left of the cursor.
        x = pt.x - (OFFSET * s) as i32 - w;
    }
    let y = (pt.y - h / 2).clamp(wa.top, (wa.bottom - h).max(wa.top));
    let z = &mut app.near;
    z.dpi = dpi;
    z.origin = (x, y);
    z.shown = true;
    z.flash = None;
    z.hover = false;
    if z.surface.as_ref().is_none_or(|surf| surf.width != w || surf.height != h) {
        z.surface = Surface::new(&app.gfx, w, h).ok();
    }
    unsafe {
        let _ = KillTimer(app.near.hwnd, HIDE_TIMER);
    }
    render(app);
    unsafe {
        let _ = ShowWindow(app.near.hwnd, SW_SHOWNOACTIVATE);
        let _ = SetWindowPos(app.near.hwnd, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE);
    }
}

/// Hides the zone after `ms`, leaving time for a drop that is under way.
pub fn hide_soon(app: &App, ms: u32) {
    if app.near.shown {
        unsafe {
            SetTimer(app.near.hwnd, HIDE_TIMER, ms, None);
        }
    }
}

fn hide(app: &mut App) {
    app.near.shown = false;
    unsafe {
        let _ = KillTimer(app.near.hwnd, HIDE_TIMER);
        let _ = ShowWindow(app.near.hwnd, SW_HIDE);
    }
}

pub fn set_hover(app: &mut App, over: bool) {
    app.near.hover = over;
    render(app);
}

/// Shows what a drop did, then hides.
pub fn dropped(app: &mut App, added: usize) {
    app.near.flash = Some(match added {
        0 => "Already here".to_string(),
        n => format!("Added {n}"),
    });
    app.near.hover = false;
    render(app);
    hide_soon(app, 900);
}

fn render(app: &mut App) {
    let z = &app.near;
    let Some(surf) = &z.surface else { return };
    let (g, t) = (&app.gfx, &app.theme);
    let caption = g.text_format(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_NORMAL).ok();
    let (x, y) = ((WIN_W - CARD_W) / 2.0, (WIN_H - CARD_H) / 2.0);
    let _ = surf.draw(z.dpi, |d| {
        d.rect_shadow(x, y + 6.0, CARD_W, CARD_H, 8.0, 12.0, 0.45 * t.shadow);
        d.fill_rounded(x, y, CARD_W, CARD_H, 8.0, t.surface_flyout);
        let fill = if z.hover { t.primary.alpha(0.28) } else { t.primary_subtle };
        d.fill_rounded(x, y, CARD_W, CARD_H, 8.0, fill);
        d.stroke_rounded_dashed(x + 0.75, y + 0.75, CARD_W - 1.5, CARD_H - 1.5, 8.0, 1.5, t.primary);
        d.cairn_mark(x + (CARD_W - 32.0) / 2.0, y + 46.0, 32.0, t.stone, t.primary);
        if let Some(f) = &caption {
            let (text, color) = match &z.flash {
                Some(s) => (s.as_str(), t.text),
                None => ("Drop here", t.text_secondary),
            };
            d.text(text, f, x, y + 96.0, CARD_W, 20.0, color);
        }
    });
    let _ = surf.present(z.hwnd, z.origin.0, z.origin.1, 1.0);
}

pub fn handle(app: &mut App, msg: u32, wparam: WPARAM) -> Option<LRESULT> {
    match msg {
        WM_TIMER if wparam.0 == HIDE_TIMER => hide(app),
        WM_MOUSEACTIVATE => return Some(LRESULT(MA_NOACTIVATE as isize)),
        _ => return None,
    }
    Some(LRESULT(0))
}

/// Registers the zone window as a drop target.
pub fn register(hwnd: HWND) {
    let _ = dnd::register(hwnd, dnd::Zone::Near);
}

//! Which tiles in the stack are selected (C4): click, Ctrl click and Shift
//! click, as in Explorer. Selection is kept by item id so it survives items
//! being added in front.

#[derive(Default)]
pub struct Selection {
    ids: Vec<u64>,
    /// Where a Shift range starts.
    anchor: usize,
}

impl Selection {
    pub fn contains(&self, id: u64) -> bool {
        self.ids.contains(&id)
    }

    pub fn len(&self) -> usize {
        self.ids.len()
    }

    /// Plain click or arrow key: just this one.
    pub fn only(&mut self, ids: &[u64], i: usize) {
        self.ids = ids.get(i).copied().into_iter().collect();
        self.anchor = i;
    }

    /// Ctrl click: add or drop one, keeping the rest.
    pub fn toggle(&mut self, ids: &[u64], i: usize) {
        let Some(&id) = ids.get(i) else { return };
        match self.ids.iter().position(|&s| s == id) {
            Some(at) => {
                self.ids.remove(at);
            }
            None => self.ids.push(id),
        }
        self.anchor = i;
    }

    /// Shift click or Shift arrow: everything from the anchor to here.
    pub fn extend(&mut self, ids: &[u64], i: usize) {
        if ids.is_empty() {
            return;
        }
        let i = i.min(ids.len() - 1);
        let a = self.anchor.min(ids.len() - 1);
        self.ids = ids[a.min(i)..=a.max(i)].to_vec();
    }

    pub fn all(&mut self, ids: &[u64]) {
        self.ids = ids.to_vec();
        self.anchor = 0;
    }

    /// Forgets items that are no longer in the stack.
    pub fn retain(&mut self, ids: &[u64]) {
        self.ids.retain(|s| ids.contains(s));
        self.anchor = self.anchor.min(ids.len().saturating_sub(1));
    }

    /// Selected ids in stack order.
    pub fn ordered(&self, ids: &[u64]) -> Vec<u64> {
        ids.iter().copied().filter(|&id| self.contains(id)).collect()
    }
}

//! User choices: the global hotkey, which screen edge holds the tab, which
//! monitors show it and how high, and whether Cairn starts at login. Stored
//! in %APPDATA%\Cairn\settings.txt, except run-at-login, which lives in the
//! registry Run key.

use std::fs;

use windows::Win32::Foundation::ERROR_SUCCESS;
use windows::Win32::System::Registry::*;
use windows::Win32::UI::Input::KeyboardAndMouse::*;
use windows::core::*;

use crate::store;

pub struct Hotkey {
    pub mods: HOT_KEY_MODIFIERS,
    pub vk: VIRTUAL_KEY,
    pub label: &'static str,
}

const CTRL_ALT: HOT_KEY_MODIFIERS = HOT_KEY_MODIFIERS(MOD_CONTROL.0 | MOD_ALT.0);

/// Offered in the tray menu. The first is the default.
pub const HOTKEYS: [Hotkey; 3] = [
    Hotkey { mods: CTRL_ALT, vk: VK_C, label: "Ctrl+Alt+C" },
    Hotkey { mods: CTRL_ALT, vk: VK_K, label: "Ctrl+Alt+K" },
    Hotkey { mods: HOT_KEY_MODIFIERS(MOD_CONTROL.0 | MOD_ALT.0 | MOD_SHIFT.0), vk: VK_C, label: "Ctrl+Alt+Shift+C" },
];

const RUN_KEY: PCWSTR = w!("Software\\Microsoft\\Windows\\CurrentVersion\\Run");
const RUN_VALUE: PCWSTR = w!("Cairn");

#[derive(Clone, Copy, PartialEq, Debug)]
pub enum Edge {
    Left,
    Right,
}

pub struct Settings {
    /// Index into HOTKEYS the user picked.
    pub hotkey: usize,
    pub edge: Edge,
    /// Device names (`\\.\DISPLAY2`) of monitors with the tab turned off.
    pub monitors_off: Vec<String>,
    /// Where the tab's center sits along the edge: 0 top, 1 bottom.
    pub tab_pos: f32,
    /// Keep Cairn's own copy of files dropped from USB and network drives.
    pub drive_copies: bool,
    /// Shaking the mouse during a drag shows a drop zone near the cursor.
    pub shake: bool,
}

impl Settings {
    pub fn load() -> Settings {
        let text = store::dir().and_then(|d| fs::read_to_string(d.join("settings.txt")).ok()).unwrap_or_default();
        Settings::parse(&text)
    }

    fn parse(text: &str) -> Settings {
        let mut s = Settings { hotkey: 0, edge: Edge::Right, monitors_off: Vec::new(), tab_pos: 0.5, drive_copies: false, shake: false };
        for line in text.lines() {
            let Some((key, value)) = line.split_once('=') else { continue };
            match (key.trim(), value.trim()) {
                ("hotkey", v) => {
                    if let Some(i) = HOTKEYS.iter().position(|h| h.label == v) {
                        s.hotkey = i;
                    }
                }
                ("edge", "left") => s.edge = Edge::Left,
                ("edge", "right") => s.edge = Edge::Right,
                ("monitors_off", v) => s.monitors_off = v.split(';').filter(|d| !d.is_empty()).map(String::from).collect(),
                ("tab_pos", v) => {
                    if let Ok(p) = v.parse::<f32>() {
                        s.tab_pos = p.clamp(0.0, 1.0);
                    }
                }
                ("drive_copies", v) => s.drive_copies = v == "on",
                ("shake", v) => s.shake = v == "on",
                _ => {}
            }
        }
        s
    }

    fn serialize(&self) -> String {
        let edge = match self.edge {
            Edge::Left => "left",
            Edge::Right => "right",
        };
        let on = |b: bool| if b { "on" } else { "off" };
        format!(
            "hotkey={}\nedge={edge}\nmonitors_off={}\ntab_pos={:.3}\ndrive_copies={}\nshake={}\n",
            HOTKEYS[self.hotkey].label,
            self.monitors_off.join(";"),
            self.tab_pos,
            on(self.drive_copies),
            on(self.shake)
        )
    }

    pub fn save(&self) {
        if let Some(d) = store::dir() {
            let _ = store::write_atomic(&d.join("settings.txt"), self.serialize().as_bytes());
        }
    }
}

pub fn run_at_login() -> bool {
    unsafe { RegGetValueW(HKEY_CURRENT_USER, RUN_KEY, RUN_VALUE, RRF_RT_REG_SZ, None, None, None) == ERROR_SUCCESS }
}

pub fn set_run_at_login(on: bool) -> bool {
    unsafe {
        if !on {
            let r = RegDeleteKeyValueW(HKEY_CURRENT_USER, RUN_KEY, RUN_VALUE);
            return r == ERROR_SUCCESS || !run_at_login();
        }
        let Ok(exe) = std::env::current_exe() else { return false };
        let value: Vec<u16> = format!("\"{}\"", exe.display()).encode_utf16().chain(Some(0)).collect();
        RegSetKeyValueW(
            HKEY_CURRENT_USER,
            RUN_KEY,
            RUN_VALUE,
            REG_SZ.0,
            Some(value.as_ptr() as *const _),
            (value.len() * 2) as u32,
        ) == ERROR_SUCCESS
    }
}

//! The stack: items parked in Cairn (file references, text and
//! links), kept until removed and encrypted at rest.

use std::fs;
use std::path::{Path, PathBuf};

use windows::Win32::Foundation::SIZE;
use windows::Win32::Graphics::Gdi::*;
use windows::Win32::System::Com::*;
use windows::Win32::UI::Shell::*;
use windows::core::{HSTRING, Interface};


use crate::store;

const MAGIC: &[u8; 8] = b"CAIRNST3";
/// 0.3 to 0.5: named stacks, items with no extra fields.
const MAGIC_V2: &[u8; 8] = b"CAIRNST2";
/// 0.1 and 0.2 kept one unnamed stack.
const MAGIC_V1: &[u8; 8] = b"CAIRNST1";
/// Extra item fields, saved as (key, length, bytes) so unknown keys can be
/// skipped and new ones added.
const FIELD_TRACK: u8 = 1;
const ICON_PX: i32 = 48;

/// Premultiplied BGRA pixels, ready for Direct2D.
pub struct Thumb {
    pub w: u32,
    pub h: u32,
    pub bgra: Vec<u8>,
}

/// One line of text for a tile: the first 300 characters, whitespace
/// collapsed.
pub fn preview(text: &str) -> String {
    let head: String = text.chars().take(300).collect();
    head.split_whitespace().collect::<Vec<_>>().join(" ")
}

pub enum Kind {
    /// A reference to a file or folder; the shelf never copies or moves it.
    File(PathBuf),
    Text(String),
    Link(String),
}

pub struct ShelfItem {
    pub id: u64,
    pub kind: Kind,
    /// Shell icon or thumbnail, loaded on first draw. `Some(None)` means
    /// loading failed and a glyph is drawn instead.
    pub icon: Option<Option<Thumb>>,
    /// The shell's link data for a file, saved when it was added, so
    /// Windows can find the file again after it moves on the same drive.
    pub track: Option<Vec<u8>>,
}

impl ShelfItem {
    fn new(id: u64, kind: Kind) -> ShelfItem {
        let track = match &kind {
            Kind::File(p) if p.exists() => track(p),
            _ => None,
        };
        ShelfItem { id, kind, icon: None, track }
    }
}

/// Every named stack. The current one's items live in `items`, so most code
/// only ever sees one stack; switching swaps lists in and out of `others`.
pub struct Shelf {
    /// The current stack's items, newest first.
    pub items: Vec<ShelfItem>,
    pub names: Vec<String>,
    pub current: usize,
    /// The other stacks' items. The slot at `current` is kept empty.
    others: Vec<Vec<ShelfItem>>,
    next_id: u64,
}

pub const DEFAULT_NAME: &str = "Stack";

pub enum Dropped {
    Paths(Vec<PathBuf>),
    Text(String),
    /// Image data with no file behind it, such as a screenshot.
    Image { bytes: Vec<u8>, ext: &'static str },
}

fn file() -> Option<PathBuf> {
    Some(store::dir()?.join("stack.bin"))
}

/// Where Cairn keeps its own copies: images that had no file, and files
/// copied from removable or network drives.
pub fn owned_dir() -> Option<PathBuf> {
    let d = store::dir()?.join("files");
    fs::create_dir_all(&d).ok()?;
    Some(d)
}

/// "Image 2026-11-03 142501.png", or with " (2)" and so on if taken.
fn free_name(dir: &Path, stem: &str, ext: &str) -> PathBuf {
    let dot = if ext.is_empty() { String::new() } else { format!(".{ext}") };
    (1..)
        .map(|n| if n == 1 { dir.join(format!("{stem}{dot}")) } else { dir.join(format!("{stem} ({n}){dot}")) })
        .find(|p| !p.exists())
        .unwrap()
}

/// A free path in `dir` for a file or folder name, keeping its extension.
pub fn free_path(dir: &Path, name: &str) -> PathBuf {
    let p = Path::new(name);
    let stem = p.file_stem().map_or(name.into(), |s| s.to_string_lossy());
    let ext = p.extension().map(|e| e.to_string_lossy().into_owned()).unwrap_or_default();
    free_name(dir, &stem, &ext)
}

fn stamp() -> String {
    let t = unsafe { windows::Win32::System::SystemInformation::GetLocalTime() };
    format!("{:04}-{:02}-{:02} {:02}{:02}{:02}", t.wYear, t.wMonth, t.wDay, t.wHour, t.wMinute, t.wSecond)
}

fn save_image(dir: &Path, bytes: &[u8], ext: &str, stamp: &str) -> Option<PathBuf> {
    let path = free_name(dir, &format!("Image {stamp}"), ext);
    fs::write(&path, bytes).ok()?;
    Some(path)
}

/// Copies a file into Cairn's folder, keeping its name.
pub fn copy_in(src: &Path) -> Option<PathBuf> {
    let dir = owned_dir()?;
    let stem = src.file_stem()?.to_string_lossy().into_owned();
    let ext = src.extension().map(|e| e.to_string_lossy().into_owned()).unwrap_or_default();
    let dest = free_name(&dir, &stem, &ext);
    fs::copy(src, &dest).ok()?;
    Some(dest)
}

pub fn looks_like_link(s: &str) -> bool {
    let s = s.trim();
    !s.contains(char::is_whitespace) && (s.starts_with("http://") || s.starts_with("https://") || s.starts_with("www."))
}

impl Shelf {
    pub fn load() -> Shelf {
        let saved = file()
            .and_then(|p| fs::read(p).ok())
            .and_then(|enc| store::unprotect(&enc))
            .and_then(|raw| parse(&raw));
        Shelf::from_stacks(saved.unwrap_or_default())
    }

    fn from_stacks(saved: Saved) -> Shelf {
        let (mut stacks, current) = (saved.stacks, saved.current);
        if stacks.is_empty() {
            stacks.push((DEFAULT_NAME.to_string(), Vec::new()));
        }
        let current = current.min(stacks.len() - 1);
        let next_id = stacks.iter().flat_map(|(_, items)| items).map(|i| i.id).max().unwrap_or(0) + 1;
        let (names, mut others): (Vec<String>, Vec<Vec<ShelfItem>>) = stacks.into_iter().unzip();
        let items = std::mem::take(&mut others[current]);
        Shelf { items, names, current, others, next_id }
    }

    pub fn save(&self) {
        if let (Some(p), Some(enc)) = (file(), store::protect(&self.encode())) {
            let _ = store::write_atomic(&p, &enc);
        }
    }

    fn encode(&self) -> Vec<u8> {
        let mut out = MAGIC.to_vec();
        out.extend_from_slice(&(self.names.len() as u32).to_le_bytes());
        out.extend_from_slice(&(self.current as u32).to_le_bytes());
        for (i, name) in self.names.iter().enumerate() {
            put_str(&mut out, name);
            let items = if i == self.current { &self.items } else { &self.others[i] };
            out.extend_from_slice(&(items.len() as u32).to_le_bytes());
            for item in items {
                let (tag, body) = match &item.kind {
                    Kind::File(p) => (0u8, p.to_string_lossy().into_owned()),
                    Kind::Text(t) => (1, t.clone()),
                    Kind::Link(l) => (2, l.clone()),
                };
                out.extend_from_slice(&item.id.to_le_bytes());
                out.push(tag);
                put_str(&mut out, &body);
                let fields: Vec<(u8, &[u8])> = item.track.iter().map(|t| (FIELD_TRACK, t.as_slice())).collect();
                out.push(fields.len() as u8);
                for (key, bytes) in fields {
                    out.push(key);
                    out.extend_from_slice(&(bytes.len() as u32).to_le_bytes());
                    out.extend_from_slice(bytes);
                }
            }
        }
        out
    }

    /// How many items a stack holds.
    pub fn count(&self, stack: usize) -> usize {
        if stack == self.current { self.items.len() } else { self.others.get(stack).map_or(0, Vec::len) }
    }

    /// Makes another stack the current one.
    pub fn switch(&mut self, stack: usize) {
        if stack < self.names.len() && stack != self.current {
            std::mem::swap(&mut self.items, &mut self.others[self.current]);
            std::mem::swap(&mut self.items, &mut self.others[stack]);
            self.current = stack;
        }
    }

    /// Adds an empty stack with a free "Stack N" name and switches to it.
    pub fn new_stack(&mut self) -> usize {
        let name = (2..).map(|n| format!("{DEFAULT_NAME} {n}")).find(|n| !self.names.contains(n)).unwrap();
        self.names.push(name);
        self.others.push(Vec::new());
        let i = self.names.len() - 1;
        self.switch(i);
        i
    }

    pub fn rename(&mut self, stack: usize, name: &str) {
        let name = name.trim();
        if let Some(n) = self.names.get_mut(stack)
            && !name.is_empty()
        {
            *n = name.chars().take(40).collect();
        }
    }

    /// Points an item, in whichever stack holds it, at a new path.
    pub fn replace_path(&mut self, id: u64, path: PathBuf) -> bool {
        let all = self.items.iter_mut().chain(self.others.iter_mut().flatten());
        match all.into_iter().find(|i| i.id == id) {
            Some(item) => {
                item.track = track(&path);
                item.kind = Kind::File(path);
                item.icon = None;
                true
            }
            None => false,
        }
    }

    /// Asks Windows where a missing file in the current stack went, using
    /// the link data saved when it was added, and points the item there.
    pub fn find_again(&mut self, id: u64) -> Option<PathBuf> {
        let item = self.items.iter().find(|i| i.id == id)?;
        let found = resolve(item.track.as_deref()?)?;
        self.replace_path(id, found.clone());
        Some(found)
    }

    /// Deletes Cairn's own copies that no stack refers to any more. Items in
    /// `keep` (waiting for undo) count as referred to.
    pub fn prune_owned(&self, keep: &[ShelfItem]) {
        let Some(dir) = owned_dir() else { return };
        prune(&dir, self.items.iter().chain(self.others.iter().flatten()).chain(keep));
    }

    /// Moves items from the current stack to the front of another, skipping
    /// any it already holds. Returns how many moved.
    pub fn move_to(&mut self, ids: &[u64], stack: usize) -> usize {
        if stack == self.current || stack >= self.names.len() {
            return 0;
        }
        let (moving, keep): (Vec<ShelfItem>, Vec<ShelfItem>) = std::mem::take(&mut self.items).into_iter().partition(|i| ids.contains(&i.id));
        self.items = keep;
        let dest = &mut self.others[stack];
        let mut moved = 0;
        for item in moving {
            if !dest.iter().any(|d| same(&d.kind, &item.kind)) {
                dest.insert(moved, item);
                moved += 1;
            }
        }
        moved
    }

    /// Deletes a stack and returns its items, unless it is the last one.
    pub fn delete_stack(&mut self, stack: usize) -> Option<Vec<ShelfItem>> {
        if self.names.len() < 2 || stack >= self.names.len() {
            return None;
        }
        if stack == self.current {
            self.switch(if stack == 0 { 1 } else { stack - 1 });
        }
        self.names.remove(stack);
        let items = self.others.remove(stack);
        if self.current > stack {
            self.current -= 1;
        }
        Some(items)
    }

    fn contains(&self, kind: &Kind) -> bool {
        self.items.iter().any(|i| same(&i.kind, kind))
    }

    /// Adds dropped content, skipping duplicates. Returns how many were new.
    pub fn add(&mut self, dropped: Dropped) -> usize {
        let kinds: Vec<Kind> = match dropped {
            Dropped::Paths(paths) => paths.into_iter().map(Kind::File).collect(),
            Dropped::Image { bytes, ext } => owned_dir().and_then(|d| save_image(&d, &bytes, ext, &stamp())).map(Kind::File).into_iter().collect(),
            Dropped::Text(t) if t.trim().is_empty() => Vec::new(),
            Dropped::Text(t) if looks_like_link(&t) => vec![Kind::Link(t.trim().to_string())],
            Dropped::Text(t) => vec![Kind::Text(t)],
        };
        let mut added = 0;
        for kind in kinds {
            if !self.contains(&kind) {
                self.items.insert(added, ShelfItem::new(self.next_id, kind));
                self.next_id += 1;
                added += 1;
            }
        }
        added
    }

    pub fn remove(&mut self, id: u64) {
        self.items.retain(|i| i.id != id);
    }

    /// Empties the stack, handing back what was in it for undo.
    pub fn clear(&mut self) -> Vec<ShelfItem> {
        std::mem::take(&mut self.items)
    }

    /// Puts cleared items back behind anything added since, skipping ones
    /// that were dropped in again. Returns how many came back.
    pub fn restore(&mut self, items: Vec<ShelfItem>) -> usize {
        let before = self.items.len();
        for item in items {
            if !self.contains(&item.kind) {
                self.items.push(item);
            }
        }
        self.items.len() - before
    }
}

fn prune<'a>(dir: &Path, items: impl Iterator<Item = &'a ShelfItem>) {
    let used: Vec<&Path> = items.filter_map(|i| if let Kind::File(p) = &i.kind { Some(p.as_path()) } else { None }).collect();
    let Ok(entries) = fs::read_dir(dir) else { return };
    for e in entries.flatten() {
        let p = e.path();
        if used.contains(&p.as_path()) {
            continue;
        }
        // Folders come from dropped folders that had no path, like one
        // dragged out of a zip.
        if p.is_dir() {
            let _ = fs::remove_dir_all(p);
        } else {
            let _ = fs::remove_file(p);
        }
    }
}

fn same(a: &Kind, b: &Kind) -> bool {
    match (a, b) {
        (Kind::File(a), Kind::File(b)) => a == b,
        (Kind::Text(a), Kind::Text(b)) | (Kind::Link(a), Kind::Link(b)) => a == b,
        _ => false,
    }
}

fn put_str(out: &mut Vec<u8>, s: &str) {
    out.extend_from_slice(&(s.len() as u32).to_le_bytes());
    out.extend_from_slice(s.as_bytes());
}

/// What the stack file holds: named stacks and which one is current.
#[derive(Default)]
struct Saved {
    stacks: Vec<(String, Vec<ShelfItem>)>,
    current: usize,
}

/// Reads the current format, or the single-stack format of 0.1 and 0.2.
fn parse(raw: &[u8]) -> Option<Saved> {
    let mut at = 0usize;
    let mut take = |n: usize| -> Option<&[u8]> {
        let s = raw.get(at..at + n)?;
        at += n;
        Some(s)
    };
    let magic = take(8)?;
    let fields = magic == MAGIC;
    let named = fields || magic == MAGIC_V2;
    if !named && magic != MAGIC_V1 {
        return None;
    }
    let u32_at = |b: &[u8]| u32::from_le_bytes(b.try_into().unwrap());
    let (stack_count, current) = if named { (u32_at(take(4)?), u32_at(take(4)?) as usize) } else { (1, 0) };
    let mut saved = Saved { stacks: Vec::new(), current };
    for _ in 0..stack_count {
        let name = if named {
            let len = u32_at(take(4)?) as usize;
            String::from_utf8(take(len)?.to_vec()).ok()?
        } else {
            DEFAULT_NAME.to_string()
        };
        let count = u32_at(take(4)?);
        let mut items = Vec::new();
        for _ in 0..count {
            let id = u64::from_le_bytes(take(8)?.try_into().ok()?);
            let tag = take(1)?[0];
            let len = u32_at(take(4)?) as usize;
            let body = String::from_utf8(take(len)?.to_vec()).ok()?;
            let kind = match tag {
                0 => Kind::File(PathBuf::from(body)),
                1 => Kind::Text(body),
                2 => Kind::Link(body),
                _ => return None,
            };
            let mut item = ShelfItem { id, kind, icon: None, track: None };
            for _ in 0..if fields { take(1)?[0] } else { 0 } {
                let key = take(1)?[0];
                let len = u32_at(take(4)?) as usize;
                let bytes = take(len)?.to_vec();
                if key == FIELD_TRACK {
                    item.track = Some(bytes);
                }
            }
            items.push(item);
        }
        saved.stacks.push((name, items));
    }
    Some(saved)
}

/// Short display name for a tile.
pub fn name(kind: &Kind) -> String {
    match kind {
        Kind::File(p) => p.file_name().map(|n| n.to_string_lossy().into_owned()).unwrap_or_else(|| p.to_string_lossy().into_owned()),
        Kind::Text(t) => preview(t),
        Kind::Link(l) => l.trim_start_matches("https://").trim_start_matches("http://").trim_start_matches("www.").to_string(),
    }
}

/// The kind word shown under the name.
pub fn kind_label(kind: &Kind) -> &'static str {
    match kind {
        Kind::File(p) if p.is_dir() => "Folder",
        Kind::File(p) if p.exists() => "File",
        Kind::File(_) => "Missing",
        Kind::Text(_) => "Text",
        Kind::Link(_) => "Link",
    }
}

/// Shell link data for a file: what a .lnk file would hold, including the
/// IDs Windows uses to track a file moved within its drive.
fn track(path: &Path) -> Option<Vec<u8>> {
    unsafe {
        let link: IShellLinkW = CoCreateInstance(&ShellLink, None, CLSCTX_INPROC_SERVER).ok()?;
        link.SetPath(&HSTRING::from(path.as_os_str())).ok()?;
        let stream = SHCreateMemStream(None)?;
        link.cast::<IPersistStream>().ok()?.Save(&stream, true).ok()?;
        let mut len = 0u64;
        stream.Seek(0, STREAM_SEEK_CUR, Some(&mut len)).ok()?;
        stream.Seek(0, STREAM_SEEK_SET, None).ok()?;
        let mut out = vec![0u8; len as usize];
        let mut read = 0u32;
        stream.Read(out.as_mut_ptr() as *mut _, len as u32, Some(&mut read)).ok().ok()?;
        out.truncate(read as usize);
        Some(out)
    }
}

/// Where tracked link data points now, if that file exists. Never shows UI
/// and never searches folders by name; gives up after half a second.
fn resolve(data: &[u8]) -> Option<PathBuf> {
    unsafe {
        let link: IShellLinkW = CoCreateInstance(&ShellLink, None, CLSCTX_INPROC_SERVER).ok()?;
        let stream = SHCreateMemStream(Some(data))?;
        link.cast::<IPersistStream>().ok()?.Load(&stream).ok()?;
        let flags = (SLR_NO_UI.0 | SLR_NOSEARCH.0) as u32 | (500 << 16);
        link.Resolve(None, flags).ok()?;
        let mut buf = vec![0u16; 32768];
        link.GetPath(&mut buf, std::ptr::null_mut(), 0).ok()?;
        let end = buf.iter().position(|&u| u == 0).unwrap_or(buf.len());
        let path = PathBuf::from(String::from_utf16_lossy(&buf[..end]));
        path.exists().then_some(path)
    }
}

/// The shell's icon or thumbnail for a path, as premultiplied BGRA.
pub fn shell_icon(path: &Path) -> Option<Thumb> {
    unsafe {
        let name = HSTRING::from(path.to_string_lossy().as_ref());
        let factory: IShellItemImageFactory = SHCreateItemFromParsingName(&name, None).ok()?;
        let bitmap = factory.GetImage(SIZE { cx: ICON_PX, cy: ICON_PX }, SIIGBF_BIGGERSIZEOK).ok()?;
        let mut bm = BITMAP::default();
        GetObjectW(bitmap, size_of::<BITMAP>() as i32, Some(&mut bm as *mut BITMAP as *mut _));
        let (w, h) = (bm.bmWidth, bm.bmHeight.abs());
        let mut info = BITMAPINFO {
            bmiHeader: BITMAPINFOHEADER {
                biSize: size_of::<BITMAPINFOHEADER>() as u32,
                biWidth: w,
                biHeight: -h,
                biPlanes: 1,
                biBitCount: 32,
                biCompression: BI_RGB.0,
                ..Default::default()
            },
            ..Default::default()
        };
        let mut bgra = vec![0u8; (w * h * 4) as usize];
        let dc = GetDC(None);
        let lines = GetDIBits(dc, bitmap, 0, h as u32, Some(bgra.as_mut_ptr() as *mut _), &mut info, DIB_RGB_COLORS);
        ReleaseDC(None, dc);
        let _ = DeleteObject(bitmap);
        if lines == 0 || w <= 0 {
            return None;
        }
        // Some thumbnails come without alpha; show them opaque.
        if bgra.chunks_exact(4).all(|p| p[3] == 0) {
            bgra.chunks_exact_mut(4).for_each(|p| p[3] = 255);
        }
        Some(Thumb { w: w as u32, h: h as u32, bgra })
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn shelf() -> Shelf {
        Shelf::from_stacks(Saved::default())
    }

    #[test]
    fn classifies_links_and_text() {
        let mut s = shelf();
        s.add(Dropped::Text("https://example.com/a".into()));
        s.add(Dropped::Text("two words".into()));
        s.add(Dropped::Text("   ".into()));
        assert!(matches!(s.items[1].kind, Kind::Link(_)));
        assert!(matches!(s.items[0].kind, Kind::Text(_)));
        assert_eq!(s.items.len(), 2);
    }

    #[test]
    fn newest_first_and_no_duplicates() {
        let mut s = shelf();
        assert_eq!(s.add(Dropped::Paths(vec!["C:\\a.txt".into(), "C:\\b.txt".into()])), 2);
        assert_eq!(s.add(Dropped::Paths(vec!["C:\\b.txt".into(), "C:\\c.txt".into()])), 1);
        let names: Vec<String> = s.items.iter().map(|i| name(&i.kind)).collect();
        assert_eq!(names, ["c.txt", "a.txt", "b.txt"]);
    }

    #[test]
    fn storage_round_trips() {
        let mut s = shelf();
        s.add(Dropped::Paths(vec!["C:\\x\\ü.txt".into()]));
        s.add(Dropped::Text("note\nline".into()));
        s.add(Dropped::Text("https://example.com".into()));
        assert!(parse(&shelf().encode()).unwrap().stacks[0].1.is_empty());
        s.new_stack();

const COLS: usize = 4;
const ROWS: usize = 3;
const TILE_W: f32 = 96.0;
const TILE_H: f32 = 88.0;
const GAP: f32 = 8.0;
const M: f32 = 12.0;
const PAD: f32 = 32.0;
const GRID_Y: f32 = M + 28.0 + 8.0;
const GRID_H: f32 = ROWS as f32 * TILE_H + (ROWS - 1) as f32 * GAP;
const FOOTER_Y: f32 = GRID_Y + GRID_H + 12.0;
const CARD_W: f32 = M * 2.0 + COLS as f32 * TILE_W + (COLS - 1) as f32 * GAP;

// 600, and new tiles settle in from 4 px above over 160.
const OPEN_MS: f32 = 140.0;
const CLOSE_MS: f32 = 120.0;
const HOVER_MS: f32 = 120.0;
const FLASH_MS: f32 = 600.0;
const SETTLE_MS: f32 = 160.0;
/// How far the card slides in from the tab while it opens, in DIPs.
const SLIDE: f32 = 8.0;
const SETTLE_DROP: f32 = 4.0;

const GLYPH_TEXT: &str = "\u{E8A5}";
const GLYPH_LINK: &str = "\u{E71B}";
const GLYPH_MISSING: &str = "\u{E7BA}";
const GLYPH_ADD: &str = "\u{E710}";
const GLYPH_REMOVE: &str = "\u{E711}";

pub struct StackView {

    /// render, and the x of the "+" button.
    heads: Vec<(f32, f32)>,
    add_x: f32,
    last_head_click: Option<(usize, Instant)>,
    /// The current stack's name while it is being renamed. `fresh` means the
    /// first key typed replaces it.
    editing: Option<(String, bool)>,

    show: Tween,

    drop_t: Tween,

const HEAD_GAP: f32 = 4.0;
const ADD_W: f32 = 28.0;

impl StackView {
    pub fn new(hwnd: HWND) -> StackView {
        StackView {
            hwnd,
            open: false,
            drop_hover: false,
            dragging_in: false,

            scroll: 0,
            hover: None,
            hover_remove: false,

        }

/// A transition's length, or none when Windows animations are off.
fn ms(app: &App, ms: f32) -> f32 {
    if app.animations { ms } else { 0.0 }
}

/// Runs the animation timer until every transition has finished.
fn animate(app: &mut App) {
    if app.animations {
        unsafe {
            SetTimer(app.stack_view.hwnd, ANIM_TIMER, 10, None);
        }
    }
    tick(app);
}

fn tick(app: &mut App) {
    render(app);
    let v = &mut app.stack_view;
    if v.settle.done() {
        v.fresh.clear();
    }
    if v.lift.done() && v.lift.target() == 0.0 {
        v.lifted.clear();
    }
    let all = [v.show, v.drop_t, v.flash, v.settle, v.lift];
    if all.iter().all(Tween::done) {
        unsafe {
            let _ = KillTimer(v.hwnd, ANIM_TIMER);
        }
    }
    if !v.open && v.show.done() {
        unsafe {
            let _ = ShowWindow(v.hwnd, SW_HIDE);
        }
    }
}

/// A drag carrying something Cairn accepts is over the stack, or left it.
pub fn set_drop_hover(app: &mut App, over: bool) {
    let d = ms(app, HOVER_MS);
    let v = &mut app.stack_view;
    v.drop_hover = over;
    v.drop_t.set(if over { 1.0 } else { 0.0 }, d);
    animate(app);
}

/// After a drop or paste while the stack is open: the `n` new items at the
/// front settle in and the stack flashes once.
pub fn added(app: &mut App, n: usize) {
    refresh(app);
    if n == 0 {
        return;
    }
    let fresh = app.stack.items.iter().take(n).map(|i| i.id).collect();
    let (flash, settle) = (ms(app, FLASH_MS), ms(app, SETTLE_MS));
    let v = &mut app.stack_view;
    v.fresh = fresh;
    v.flash.run(1.0, 0.0, flash);
    v.settle.run(0.0, 1.0, settle);
    animate(app);
}

struct Fonts {
    name: Option<IDWriteTextFormat>,
    caption: Option<IDWriteTextFormat>,
    icon32: Option<IDWriteTextFormat>,
    icon12: Option<IDWriteTextFormat>,
}

fn fonts(g: &Gfx) -> Fonts {
    let fmt = |family, size, weight| g.text_format_aligned(family, size, weight, DWRITE_TEXT_ALIGNMENT_CENTER).ok();
    Fonts {
        name: fmt(&g.ui_font, 14.0, DWRITE_FONT_WEIGHT_SEMI_BOLD),
        caption: fmt(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_NORMAL),
        icon32: fmt(&g.icon_font, 28.0, DWRITE_FONT_WEIGHT_NORMAL),
        icon12: fmt(&g.icon_font, 10.0, DWRITE_FONT_WEIGHT_NORMAL),
    }
}

#[derive(Default)]
struct Look {
    hovered: bool,
    selected: bool,
    focused: bool,
}

/// One tile: its card, icon or thumbnail, name and kind.
fn draw_tile(p: &Painter, t: &Theme, f: &Fonts, item: &ShelfItem, tx: f32, ty: f32, look: Look) {
    p.rect_shadow(tx, ty + 1.0, TILE_W, TILE_H, 8.0, 2.0, 0.4 * t.shadow);
    p.fill_rounded(tx, ty, TILE_W, TILE_H, 8.0, if look.hovered { t.surface_hover } else { t.surface });
    if look.selected {
        p.fill_rounded(tx, ty, TILE_W, TILE_H, 8.0, t.primary_subtle);
    }
    if look.focused {
        p.stroke_rounded(tx + 0.5, ty + 0.5, TILE_W - 1.0, TILE_H - 1.0, 8.0, 1.0, t.primary);
    }
    // A missing file's icon and name are dimmed; its "Missing" label is not,
    // so it stays readable.
    let missing = matches!(&item.kind, Kind::File(p) if !p.exists());
    p.layer(if missing { 0.5 } else { 1.0 }, 0.0, 0.0, |p| {
        match item.icon.as_ref().and_then(|i| i.as_ref()) {
            Some(icon) => {
                let scale = (32.0 / icon.w.max(icon.h) as f32).min(1.0);
                let (iw, ih) = (icon.w as f32 * scale, icon.h as f32 * scale);
                p.image(&icon.bgra, icon.w, icon.h, tx + (TILE_W - iw) / 2.0, ty + 10.0 + (32.0 - ih) / 2.0, iw, ih);
            }
            None => {
                let glyph = match &item.kind {
                    Kind::File(_) => GLYPH_MISSING,
                    Kind::Text(_) => GLYPH_TEXT,
                    Kind::Link(_) => GLYPH_LINK,
                };
                if let Some(fm) = &f.icon32 {
                    p.text(glyph, fm, tx, ty + 8.0, TILE_W, 36.0, t.text);
                }
            }
        }
        if let Some(fm) = &f.name {
            p.text(&shelf::name(&item.kind), fm, tx + 6.0, ty + 46.0, TILE_W - 12.0, 20.0, t.text);
        }
    });
    if let Some(fm) = &f.caption {
        p.text(shelf::kind_label(&item.kind), fm, tx + 6.0, ty + 64.0, TILE_W - 12.0, 16.0, t.text_secondary);
    }
}

/// What a drag out shows under the cursor: the grabbed tile, the rest of
/// the selection stacked behind it, and a count when there are several.
fn drag_image(app: &App, index: usize, count: usize, grab: POINT) -> Option<SHDRAGIMAGE> {
    const X: f32 = 6.0;
    const Y: f32 = 14.0;
    let item = app.stack.items.get(index)?;
    let rect = tile_screen_rect(app, index)?;
    let dpi = app.stack_view.dpi;
    let s = dpi as f32 / 96.0;
    let (w, h) = (((X + TILE_W + 22.0) * s).ceil() as i32, ((Y + TILE_H + 8.0) * s).ceil() as i32);
    let surf = Surface::new(&app.gfx, w, h).ok()?;
    let (g, t, f) = (&app.gfx, &app.theme, fonts(&app.gfx));
    let badge = g.text_format_aligned(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_SEMI_BOLD, DWRITE_TEXT_ALIGNMENT_CENTER).ok();
    surf.draw(dpi, |p| {
        p.rect_shadow(X, Y + 3.0, TILE_W, TILE_H, 8.0, 6.0, 0.5 * t.shadow);
        for k in (1..count.min(3)).rev() {
            let o = 4.0 * k as f32;
            p.fill_rounded(X + o, Y - o, TILE_W, TILE_H, 8.0, t.surface);
            p.stroke_rounded(X + o + 0.5, Y - o + 0.5, TILE_W - 1.0, TILE_H - 1.0, 8.0, 1.0, t.outline);
        }
        draw_tile(p, t, &f, item, X, Y, Look::default());
        p.stroke_rounded(X + 0.5, Y + 0.5, TILE_W - 1.0, TILE_H - 1.0, 8.0, 1.0, t.outline);
        if count > 1
            && let Some(fm) = &badge
        {
            let label = count.to_string();
            let bw = (g.text_width(&label, fm) + 12.0).max(20.0);
            let (bx, by) = (X + TILE_W + 2.0 - bw / 2.0, Y - 10.0);
            p.fill_rounded(bx, by, bw, 20.0, 10.0, t.primary);
            p.text(&label, fm, bx, by, bw, 20.0, t.surface);
        }
    })
    .ok()?;
    let offset = POINT { x: grab.x - rect.left + (X * s) as i32, y: grab.y - rect.top + (Y * s) as i32 };
    Some(SHDRAGIMAGE {
        sizeDragImage: SIZE { cx: w, cy: h },
        ptOffset: offset,
        hbmpDragImage: surf.into_bitmap(),
        // No color key: the bitmap's own alpha shapes it.
        crColorKey: COLORREF(0xffff_ffff),
    })
}

/// Grid cells: one per item.
fn cells(app: &App) -> usize {
    app.stack.items.len()
}

fn ids(app: &App) -> Vec<u64> {
    app.stack.items.iter().map(|i| i.id).collect()
}

fn cell_pos(index: usize, scroll: usize) -> Option<(f32, f32)> {
    let row = index / COLS;
    if row < scroll || row >= scroll + ROWS {
        return None;
    }
    let col = index % COLS;
    Some((M + col as f32 * (TILE_W + GAP), GRID_Y + (row - scroll) as f32 * (TILE_H + GAP)))
}

/// A stack tab's text: its name, or the name being typed with a caret.
fn head_label(app: &App, i: usize, name: &str) -> String {
    match &app.stack_view.editing {
        Some((text, _)) if i == app.stack.current => format!("{text}|"),
        _ => name.to_string(),
    }
}

/// Opens beside the edge tab, on the screen side of it.
fn place(app: &mut App) {
    let (anchor, wa) = edgetab::anchor(app);
    let dpi = dpi_at(anchor);
    let s = dpi as f32 / 96.0;
    let (cw, ch) = ((CARD_W * s) as i32, (CARD_H * s) as i32);
    let gap = (GAP_FROM_TAB * s) as i32;
    let margin = (8.0 * s) as i32;
    let x = match app.settings.edge {
        Edge::Left => anchor.x + gap,
        Edge::Right => anchor.x - gap - cw,
    };
    let x = x.clamp(wa.left + margin, (wa.right - margin - cw).max(wa.left + margin));
    let y = (anchor.y - ch / 2).clamp(wa.top + margin, (wa.bottom - margin - ch).max(wa.top + margin));
    let pad = (PAD * s) as i32;
    let v = &mut app.stack_view;
    v.dpi = dpi;
    v.origin = (x - pad, y - pad);
    let (pw, ph) = (((CARD_W + PAD * 2.0) * s).ceil() as i32, ((CARD_H + PAD * 2.0) * s).ceil() as i32);
    if v.surface.as_ref().is_none_or(|surf| surf.width != pw || surf.height != ph) {
        v.surface = Surface::new(&app.gfx, pw, ph).ok();
    }
}

pub fn render(app: &mut App) {
    // Keeps drawing while the close fades out.
    if !app.stack_view.open && app.stack_view.show.done() {
        return;
    }
    // Load icons for visible tiles that have none yet.
    let scroll = app.stack_view.scroll;
    for (i, item) in app.stack.items.iter_mut().enumerate() {
        if item.icon.is_none() && cell_pos(i, scroll).is_some() {
            item.icon = Some(match &item.kind {
                Kind::File(p) if p.exists() => shelf::shell_icon(p),
                _ => None,
            });
        }
    }

    // Lay out the stack tabs: each name in a pill, then "+", leaving room
    // for the item count. Tabs that do not fit are reached with Ctrl+Tab.
    let head_fmt = app.gfx.text_format_aligned(&app.gfx.ui_font, 14.0, DWRITE_FONT_WEIGHT_SEMI_BOLD, DWRITE_TEXT_ALIGNMENT_CENTER).ok();
    // A tab that does not fit gets zero width, so indexes match names.
    let mut heads = Vec::new();
    let mut x = M;
    let room = CARD_W - M - COUNT_W - ADD_W - HEAD_GAP;
    for (i, name) in app.stack.names.iter().enumerate() {
        let w = head_fmt.as_ref().map_or(60.0, |f| app.gfx.text_width(&head_label(app, i, name), f)).min(160.0) + HEAD_PAD * 2.0;
        if x + w > room && i != app.stack.current {
            heads.push((0.0, 0.0));
            continue;
        }
        heads.push((x, w));
        x += w + HEAD_GAP;
    }
    app.stack_view.heads = heads;
    app.stack_view.add_x = x;

    let v = &app.stack_view;
    let Some(surf) = &v.surface else { return };
    let (g, t) = (&app.gfx, &app.theme);
    let fmt = |family, size, weight, align| g.text_format_aligned(family, size, weight, align).ok();
    let f = fonts(g);
    let caption_lead = fmt(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_NORMAL, DWRITE_TEXT_ALIGNMENT_LEADING);
    let caption_end = fmt(&g.ui_font, 12.0, DWRITE_FONT_WEIGHT_NORMAL, DWRITE_TEXT_ALIGNMENT_TRAILING);
    let items = &app.stack.items;
    let (drop_t, flash, settle, lift) = (v.drop_t.value(), v.flash.value(), v.settle.value(), v.lift.value());

    let _ = surf.draw(v.dpi, |p| {
        let (cx, cy) = (PAD, PAD);
        p.rect_shadow(cx, cy + 8.0, CARD_W, CARD_H, 8.0, 24.0, 0.55 * t.shadow);
        p.fill_rounded(cx, cy, CARD_W, CARD_H, 8.0, t.surface_flyout);
        if drop_t < 1.0 {
            p.layer(1.0 - drop_t, 0.0, 0.0, |p| p.stroke_rounded(cx + 0.5, cy + 0.5, CARD_W - 1.0, CARD_H - 1.0, 8.0, 1.0, t.outline));
        }
        if drop_t > 0.0 {
            // The whole stack is the drop target. An opaque base keeps the
            // window behind from showing through the tint.
            p.layer(drop_t, 0.0, 0.0, |p| {
                p.fill_rounded(cx, cy, CARD_W, CARD_H, 8.0, t.surface);
                p.fill_rounded(cx, cy, CARD_W, CARD_H, 8.0, t.primary_subtle);
                p.stroke_rounded_dashed(cx + 0.75, cy + 0.75, CARD_W - 1.5, CARD_H - 1.5, 8.0, 1.5, t.primary);
            });
        }
        if flash > 0.0 {
            p.fill_rounded(cx, cy, CARD_W, CARD_H, 8.0, t.primary.alpha(0.22 * flash));
        }
        if items.is_empty() {
            if let Some(f) = &f.name {
                p.text("Nothing here yet", f, cx, cy + GRID_Y + GRID_H / 2.0 - 22.0, CARD_W, 20.0, t.text);
            }
            if let Some(f) = &f.caption {
                p.text("Drag files, text or links onto the edge tab or here", f, cx, cy + GRID_Y + GRID_H / 2.0, CARD_W, 16.0, t.text_secondary);
            }
        }
        for (i, &(hx, hw)) in v.heads.iter().enumerate().filter(|(_, h)| h.1 > 0.0) {
            let current = i == app.stack.current;
            if current {
                p.fill_rounded(cx + hx, cy + M, hw, 28.0, 6.0, t.surface);
            }
            if current && v.editing.is_some() {
                p.stroke_rounded(cx + hx + 0.5, cy + M + 0.5, hw - 1.0, 27.0, 6.0, 1.0, t.primary);
            }
            if let Some(f) = &head_fmt {
                let color = if current { t.text } else { t.text_secondary };
                p.text(&head_label(app, i, &app.stack.names[i]), f, cx + hx + HEAD_PAD, cy + M + 4.0, hw - HEAD_PAD * 2.0, 20.0, color);
            }
        }
        if let Some(f) = &f.icon12 {
            p.text(GLYPH_ADD, f, cx + v.add_x, cy + M + 8.0, ADD_W, 14.0, t.text_secondary);
        }
        if let Some(f) = &caption_end {
            let n = items.len();
            let count = if n == 1 { "1 item".to_string() } else { format!("{n} items") };
            p.text(&count, f, cx + CARD_W - M - COUNT_W, cy + M, COUNT_W, 28.0, t.text_secondary);
        }

        for i in 0..items.len() {
            let Some((tx, ty)) = cell_pos(i, v.scroll) else { continue };
            let (tx, ty) = (cx + tx, cy + ty);
            let item = &items[i];
            let hovered = v.hover == Some(i);
            let look = Look { hovered, selected: v.sel.contains(item.id), focused: v.focus == i };
            if v.lifted.contains(&item.id) {
                // Lifted out by a drag: a faded tile over its empty slot.
                p.layer(lift, 0.0, 0.0, |p| p.stroke_rounded_dashed(tx + 0.75, ty + 0.75, TILE_W - 1.5, TILE_H - 1.5, 8.0, 1.5, t.outline_control));
                p.layer(1.0 - 0.65 * lift, 0.0, 0.0, |p| draw_tile(p, t, &f, item, tx, ty, look));
            } else if v.fresh.contains(&item.id) {
                p.layer(settle, 0.0, -SETTLE_DROP * (1.0 - settle), |p| {
                    draw_tile(p, t, &f, item, tx, ty, look);
                    p.fill_rounded(tx, ty, TILE_W, TILE_H, 8.0, t.primary.alpha(0.2 * flash));
                });
            } else {
                draw_tile(p, t, &f, item, tx, ty, look);
            }
            if hovered {
                let (rx, ry) = (tx + TILE_W - REMOVE - 4.0, ty + 4.0);
                let bg = if v.hover_remove { t.surface } else { t.surface_flyout };
                p.fill_circle(rx + REMOVE / 2.0, ry + REMOVE / 2.0, REMOVE / 2.0, bg);
                p.stroke_circle(rx + REMOVE / 2.0, ry + REMOVE / 2.0, REMOVE / 2.0 - 0.5, 1.0, t.outline_control);
                if let Some(f) = &f.icon12 {
                    p.text(GLYPH_REMOVE, f, rx, ry, REMOVE, REMOVE, t.text);
                }
            }
        }

        let total_rows = items.len().div_ceil(COLS);
        if total_rows > ROWS {
            let thumb = (GRID_H * ROWS as f32 / total_rows as f32).max(16.0);
            let top = (GRID_H - thumb) * v.scroll as f32 / (total_rows - ROWS) as f32;
            p.fill_rounded(cx + CARD_W - 5.0, cy + GRID_Y + top, 2.0, thumb, 1.0, t.text_secondary);
        }
        p.fill_rect(cx, cy + FOOTER_Y - 4.0, CARD_W, 1.0, t.divider);
        if let Some(f) = &caption_lead {
            let hint = match v.sel.len() {
                _ if v.drop_hover && app.stack.names.len() > 1 => format!("Drop to add to {}     Point at another name to switch", app.stack.names[app.stack.current]),
                _ if v.drop_hover => format!("Drop to add to {}", app.stack.names[app.stack.current]),
                _ if v.dragging_in => "Move onto the stack to drop here".to_string(),
                n if n > 1 => format!("{n} selected     Drag out to use     Del remove     Right-click for more"),
                _ => "Drag out to use     Enter open     Del remove     Right-click for more".to_string(),
            };
            p.text(&hint, f, cx + M, cy + FOOTER_Y, CARD_W - M * 2.0, 16.0, t.text_secondary);
        }
    });
    // Fades in while sliding out from the tab, and back on close.
    let shown = v.show.value();
    let toward_tab = match app.settings.edge {
        Edge::Left => -1.0,
        Edge::Right => 1.0,
    };
    let dx = ((1.0 - shown) * SLIDE * v.dpi as f32 / 96.0 * toward_tab).round() as i32;
    let _ = surf.present(v.hwnd, v.origin.0 + dx, v.origin.1, shown);
}

pub fn open(app: &mut App, target: HWND) {
    app.stack_view.target = target;
    app.stack_view.editing = None;
    app.stack_view.open = true;
    app.stack_view.focus = 0;
    let ids = ids(app);
    app.stack_view.sel.only(&ids, 0);
    app.stack_view.scroll = 0;
    app.stack_view.hover = None;
    place(app);
    app.stack_view.show.set(1.0, ms(app, OPEN_MS));
    render(app);
    unsafe {
        let _ = ShowWindow(app.stack_view.hwnd, SW_SHOW);
        let _ = SetForegroundWindow(app.stack_view.hwnd);
    }
    animate(app);
    announce(app);
}

/// Back to the first tile, selected, after the stack's contents change.
fn reset_view(app: &mut App) {
    let ids = ids(app);
    let v = &mut app.stack_view;
    v.focus = 0;
    v.sel.only(&ids, 0);
    v.scroll = 0;
    v.hover = None;
}

pub fn switch_stack(app: &mut App, stack: usize) {
    if stack == app.stack.current || stack >= app.stack.names.len() {
        return;
    }
    finish_rename(app, true);
    app.stack.switch(stack);
    app.stack.save();
    reset_view(app);
    render(app);
    announce(app);
}

fn cycle_stack(app: &mut App, delta: isize) {
    let n = app.stack.names.len() as isize;
    let next = (app.stack.current as isize + delta).rem_euclid(n) as usize;
    switch_stack(app, next);
}

fn new_stack(app: &mut App) {
    finish_rename(app, true);
    app.stack.new_stack();
    app.stack.save();
    reset_view(app);
    start_rename(app);
}

fn start_rename(app: &mut App) {
    let name = app.stack.names[app.stack.current].clone();
    app.stack_view.editing = Some((name, true));
    render(app);
}

fn finish_rename(app: &mut App, commit: bool) {
    if let Some((text, _)) = app.stack_view.editing.take() {
        if commit {
            app.stack.rename(app.stack.current, &text);
            app.stack.save();
        }
        render(app);
    }
}

/// Typing while a stack is being renamed. Returns false for keys it leaves
/// alone.
fn edit_key(app: &mut App, vk: VIRTUAL_KEY) -> bool {
    match vk {
        VK_RETURN => finish_rename(app, true),
        VK_ESCAPE => finish_rename(app, false),
        VK_BACK => {
            if let Some((text, fresh)) = &mut app.stack_view.editing {
                if *fresh {
                    text.clear();
                } else {
                    text.pop();
                }
                *fresh = false;
            }
            render(app);
        }
        _ => return false,
    }
    true
}

fn edit_char(app: &mut App, c: char) {
    if let Some((text, fresh)) = &mut app.stack_view.editing
        && !c.is_control()
    {
        if *fresh {
            text.clear();
            *fresh = false;
        }
        if text.chars().count() < 40 {
            text.push(c);
        }
        render(app);
    }
}

/// Deletes a stack after the user picked it from the menu. Its items can be
/// brought back for 10 seconds, into whichever stack is current then.
fn delete_stack(app: &mut App, stack: usize) {
    finish_rename(app, false);
    let name = app.stack.names[stack].clone();
    let Some(items) = app.stack.delete_stack(stack) else { return };
    app.stack.save();
    reset_view(app);
    render(app);
    announce(app);
    if !items.is_empty() {
        crate::offer_undo(app, items, &format!("Deleted \"{name}\""));
    }
}

/// The selected items, or the focused one if nothing is selected.
fn chosen(app: &App) -> Vec<u64> {
    let mut ids = app.stack_view.sel.ordered(&ids(app));
    if ids.is_empty() {
        ids.extend(app.stack.items.get(app.stack_view.focus).map(|i| i.id));
    }
    ids
}

fn popup(owner: HWND, menu: HMENU, at: POINT) -> u32 {
    unsafe {
        let cmd = TrackPopupMenu(menu, TPM_RETURNCMD | TPM_RIGHTBUTTON | TPM_NONOTIFY, at.x, at.y, 0, owner, None);
        let _ = DestroyMenu(menu);
        cmd.0 as u32
    }
}

/// The actions menu for the selection.
fn tile_menu(app: &mut App, at: POINT) {
    let ids = chosen(app);
    if ids.is_empty() {
        return;
    }
    // Copy what the actions need now: the menu and dialogs below can close
    // the flyout or change the stack.
    let (text_paths, text_md, files, refs, first_is_text, first_missing) = {
        let kinds: Vec<&Kind> = app.stack.items.iter().filter(|i| ids.contains(&i.id)).map(|i| &i.kind).collect();
        let refs = kinds.iter().any(|k| !matches!(k, Kind::Text(_)));
        let missing = match kinds[0] {
            Kind::File(p) if !p.exists() => Some(p.clone()),
            _ => None,
        };
        (actions::paths_text(&kinds), actions::markdown(&kinds), actions::existing_files(&kinds), refs, matches!(kinds[0], Kind::Text(_)), missing)
    };
    let one = ids.len() == 1;
    let find = first_missing.filter(|_| one);
    let has_files = !files.is_empty();
    let menu = unsafe {
        let Ok(menu) = CreatePopupMenu() else { return };
        let on = |b: bool| if b { MF_ENABLED } else { MF_GRAYED };
        if find.is_some() {
            let _ = AppendMenuW(menu, MF_STRING, A_FIND as usize, w!("Find file…"));
            let _ = AppendMenuW(menu, MF_SEPARATOR, 0, None);
        } else if one {
            let open = if first_is_text { w!("Paste") } else { w!("Open") };
            let _ = AppendMenuW(menu, MF_STRING, A_OPEN as usize, open);
            let _ = AppendMenuW(menu, MF_SEPARATOR, 0, None);
        }
        let _ = AppendMenuW(menu, MF_STRING | on(refs), A_COPY_PATHS as usize, if one { w!("Copy path") } else { w!("Copy paths") });
        let _ = AppendMenuW(menu, MF_STRING | on(refs), A_COPY_MD as usize, w!("Copy as Markdown link"));
        let _ = AppendMenuW(menu, MF_STRING | on(has_files), A_ZIP as usize, w!("Save as zip…"));
        let _ = AppendMenuW(menu, MF_STRING | on(has_files), A_COPY_TO as usize, w!("Copy to folder…"));
        if app.stack.names.len() > 1
            && let Ok(stacks) = CreatePopupMenu()
        {
            for (i, name) in app.stack.names.iter().enumerate().filter(|&(i, _)| i != app.stack.current) {
                let _ = AppendMenuW(stacks, MF_STRING, A_MOVE_BASE as usize + i, &HSTRING::from(name.as_str()));
            }
            let _ = AppendMenuW(menu, MF_POPUP, stacks.0 as usize, w!("Move to stack"));
        }
        let _ = AppendMenuW(menu, MF_SEPARATOR, 0, None);
        let _ = AppendMenuW(menu, MF_STRING, A_REMOVE as usize, w!("Remove"));
        menu
    };
    match popup(app.stack_view.hwnd, menu, at) {
        A_OPEN => activate(app, app.stack.items.iter().position(|i| i.id == ids[0]).unwrap_or(0)),
        A_COPY_PATHS => copy_text(app, &text_paths, if one { "Path copied" } else { "Paths copied" }),
        A_COPY_MD => copy_text(app, &text_md, "Markdown copied"),
        A_ZIP => {
            close(app, false);
            if let Some(out) = actions::pick_zip(HWND::default(), &files[0]) {
                app.zip_out = Some(out.clone());
                actions::zip(files, out, app.host);
            }
        }
        A_COPY_TO => {
            close(app, false);
            if let Some(folder) = actions::pick_folder(HWND::default()) {
                actions::copy_to(HWND::default(), &files, &folder);
            }
        }
        A_FIND => {
            let Some(old) = find else { return };
            if app.stack.find_again(ids[0]).is_some() {
                app.stack.save();
                refresh(app);
                return;
            }
            close(app, false);
            if let Some(path) = actions::pick_file(HWND::default(), &old)
                && app.stack.replace_path(ids[0], path)
            {
                app.stack.save();
            }
        }
        A_REMOVE => remove_selected(app),
        c if c >= A_MOVE_BASE && ((c - A_MOVE_BASE) as usize) < app.stack.names.len() => {
            let dest = (c - A_MOVE_BASE) as usize;
            let moved = app.stack.move_to(&ids, dest);
            app.stack.save();
            let n = ids.len();
            refresh(app);
            reset_view(app);
            render(app);
            announce(app);
            let name = &app.stack.names[dest];
            let note = match (moved, n) {
                (1, 1) => format!("Moved to {name}"),
                (m, n) if m == n => format!("Moved {m} items to {name}"),
                (m, _) => format!("Moved {m} to {name}; the rest were already there"),
            };
            crate::notify(app, "Cairn", &note);
        }
        _ => {}
    }
}

fn copy_text(app: &mut App, text: &str, done: &str) {
    if clipboard::write_text(app.host, text) {
        crate::notify(app, done, &shelf::preview(text));
    }
}

/// The menu for a stack's tab.
fn head_menu(app: &mut App, stack: usize, at: POINT) {
    switch_stack(app, stack);
    let n = app.stack.count(stack);
    let menu = unsafe {
        let Ok(menu) = CreatePopupMenu() else { return };
        let _ = AppendMenuW(menu, MF_STRING, H_RENAME as usize, w!("Rename\tF2"));
        let _ = AppendMenuW(menu, MF_STRING, H_NEW as usize, w!("New stack\tCtrl+N"));
        let delete = match n {
            0 => "Delete stack".to_string(),
            1 => "Delete stack and its item".to_string(),
            n => format!("Delete stack and its {n} items"),
        };
        let last = if app.stack.names.len() < 2 { MF_GRAYED } else { MF_ENABLED };
        let _ = AppendMenuW(menu, MF_SEPARATOR, 0, None);
        let _ = AppendMenuW(menu, MF_STRING | last, H_DELETE as usize, &HSTRING::from(delete));
        menu
    };
    match popup(app.stack_view.hwnd, menu, at) {
        H_RENAME => start_rename(app),
        H_NEW => new_stack(app),
        H_DELETE => delete_stack(app, stack),
        _ => {}
    }
}

fn hit_head(app: &App, lparam: LPARAM) -> Option<Head> {
    let s = app.stack_view.dpi as f32 / 96.0;
    let x = (lparam.0 & 0xffff) as i16 as f32 / s - PAD;
    let y = ((lparam.0 >> 16) & 0xffff) as i16 as f32 / s - PAD;
    head_at(app, x, y)
}

/// The header control at a point in card DIPs.
fn head_at(app: &App, x: f32, y: f32) -> Option<Head> {
    let v = &app.stack_view;
    if !(M..M + 28.0).contains(&y) {
        return None;
    }
    if (v.add_x..v.add_x + ADD_W).contains(&x) {
        return Some(Head::Add);
    }
    v.heads.iter().position(|&(hx, hw)| hw > 0.0 && (hx..hx + hw).contains(&x)).map(Head::Stack)
}

/// Opens the stack while something is being dragged onto the edge tab,
/// without taking focus from the app the drag came from.
pub fn open_for_drag(app: &mut App) {
    if app.stack_view.open {
        return;
    }
    app.stack_view.target = unsafe { GetForegroundWindow() };
    app.stack_view.editing = None;
    app.stack_view.open = true;
    app.stack_view.dragging_in = true;
    reset_view(app);
    place(app);
    app.stack_view.show.set(1.0, ms(app, OPEN_MS));
    render(app);
    unsafe {
        let _ = ShowWindow(app.stack_view.hwnd, SW_SHOWNOACTIVATE);
        let _ = SetWindowPos(app.stack_view.hwnd, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE | SWP_NOSIZE | SWP_NOACTIVATE);
    }
    animate(app);
}

/// A drag moving over the stack: pointing at another stack's name switches
/// to it, so the drop lands there.
pub fn drag_over(app: &mut App, pt: POINT) {
    let v = &app.stack_view;
    let s = v.dpi as f32 / 96.0;
    let (x, y) = ((pt.x - v.origin.0) as f32 / s - PAD, (pt.y - v.origin.1) as f32 / s - PAD);
    if let Some(Head::Stack(i)) = head_at(app, x, y) {
        switch_stack(app, i);
    }
}

/// The actions menu from the keyboard, at the focused tile.
fn focused_menu(app: &mut App) {
    let focus = app.stack_view.focus;
    if let Some(r) = tile_screen_rect(app, focus) {
        if !is_selected(app, focus) {
            take_focus(app, focus);
        }
        tile_menu(app, POINT { x: r.left + 16, y: r.bottom - 16 });
    }
}

fn cursor() -> POINT {
    let mut pt = POINT::default();
    unsafe {
        let _ = GetCursorPos(&mut pt);
    }
    pt
}

/// Tells screen readers which tile has the keyboard focus.
fn announce(app: &App) {
    let v = &app.stack_view;
    a11y::focus_changed(v.hwnd, (!app.stack.items.is_empty()).then_some(v.focus));
}

pub fn focus_index(app: &App) -> usize {
    app.stack_view.focus
}

pub fn is_selected(app: &App, index: usize) -> bool {
    app.stack.items.get(index).is_some_and(|i| app.stack_view.sel.contains(i.id))
}

/// A tile's rectangle in screen pixels, if it is scrolled into view.
pub fn tile_screen_rect(app: &App, index: usize) -> Option<RECT> {
    let v = &app.stack_view;
    let (tx, ty) = cell_pos(index, v.scroll)?;
    let s = v.dpi as f32 / 96.0;
    let (x, y) = (v.origin.0 + ((PAD + tx) * s) as i32, v.origin.1 + ((PAD + ty) * s) as i32);
    Some(RECT { left: x, top: y, right: x + (TILE_W * s) as i32, bottom: y + (TILE_H * s) as i32 })
}

/// The tile under a screen point.
pub fn hit_screen(app: &App, pt: POINT) -> Option<usize> {
    (0..app.stack.items.len()).find(|&i| {
        tile_screen_rect(app, i).is_some_and(|r| pt.x >= r.left && pt.x < r.right && pt.y >= r.top && pt.y < r.bottom)
    })
}

/// Moves the focus to a tile and selects only it, as a screen reader asks.
pub fn take_focus(app: &mut App, index: usize) {
    if index >= app.stack.items.len() {
        return;
    }
    let ids = ids(app);
    let v = &mut app.stack_view;
    v.focus = index;
    v.sel.only(&ids, index);
    keep_visible(v);
    render(app);
    announce(app);
}

pub fn close(app: &mut App, restore: bool) {
    if !app.stack_view.open {
        return;
    }
    // Clicking away keeps a name that was being typed.
    finish_rename(app, true);
    let d = ms(app, CLOSE_MS);
    let v = &mut app.stack_view;
    v.open = false;
    v.dragging_in = false;
    v.drop_hover = false;
    v.press = None;
    v.drop_t.set(0.0, d);
    // Fades out; the timer hides the window at the end.
    v.show.set(0.0, d);
    unsafe {
        if restore && !v.target.is_invalid() {
            let _ = SetForegroundWindow(v.target);
        }
    }
    animate(app);
}

/// Called after the stack changes from outside (a drop on the edge tab).
pub fn refresh(app: &mut App) {
    let n = cells(app);
    let ids = ids(app);
    let v = &mut app.stack_view;
    v.sel.retain(&ids);
    v.focus = v.focus.min(n.saturating_sub(1));
    let rows = n.div_ceil(COLS);
    v.scroll = v.scroll.min(rows.saturating_sub(ROWS));
    render(app);
}

fn keep_visible(v: &mut StackView) {
    let row = v.focus / COLS;
    if row < v.scroll {
        v.scroll = row;
    } else if row >= v.scroll + ROWS {
        v.scroll = row + 1 - ROWS;
    }
}

/// Files and links open in their default app; text is pasted into the app
/// the stack was opened from.
pub fn activate(app: &mut App, index: usize) {
    enum Act {
        Missing,
        Open(String),
        Paste(String),
    }
    // A file moved on the same drive is found again before it counts as missing.
    if let Some(item) = app.stack.items.get(index)
        && matches!(&item.kind, Kind::File(p) if !p.exists())
        && app.stack.find_again(item.id).is_some()
    {
        app.stack.save();
    }
    let act = match app.stack.items.get(index).map(|i| &i.kind) {
        None => return,
        Some(Kind::File(p)) if !p.exists() => Act::Missing,
        Some(Kind::File(p)) => Act::Open(p.to_string_lossy().into_owned()),
        Some(Kind::Link(l)) if l.starts_with("www.") => Act::Open(format!("https://{l}")),
        Some(Kind::Link(l)) => Act::Open(l.clone()),
        Some(Kind::Text(t)) => Act::Paste(t.clone()),
    };
    match act {
        Act::Missing => crate::notify(app, "File not found", "It was moved or deleted. Right-click it to find it, or press Del to remove it."),
        Act::Open(target) => {
            close(app, false);
            unsafe {
                ShellExecuteW(None, w!("open"), &HSTRING::from(target), None, None, SW_SHOWNORMAL);
            }
        }
        Act::Paste(text) => {
            if clipboard::write_text(app.host, &text) {
                close(app, true);
                if !app.stack_view.target.is_invalid() {
                    crate::schedule_paste(app);
                }
            }
        }
    }
}

fn remove(app: &mut App, gone: &[u64]) {
    if gone.is_empty() {
        return;
    }
    for &id in gone {
        app.stack.remove(id);
    }
    app.stack.save();
    app.stack_view.hover = None;
    refresh(app);
}

/// Del removes the selection, or the focused tile if nothing is selected,
/// then selects whatever moved into the focused spot.
fn remove_selected(app: &mut App) {
    let mut gone = app.stack_view.sel.ordered(&ids(app));
    if gone.is_empty() {
        gone.extend(app.stack.items.get(app.stack_view.focus).map(|i| i.id));
    }
    remove(app, &gone);
    let (ids, focus) = (ids(app), app.stack_view.focus);
    app.stack_view.sel.only(&ids, focus);
    render(app);
    announce(app);
}

fn hit(app: &App, lparam: LPARAM) -> Option<(usize, bool)> {
    let v = &app.stack_view;
    let s = v.dpi as f32 / 96.0;
    let x = (lparam.0 & 0xffff) as i16 as f32 / s - PAD;
    let y = ((lparam.0 >> 16) & 0xffff) as i16 as f32 / s - PAD;
    (0..cells(app)).find_map(|i| {
        let (tx, ty) = cell_pos(i, v.scroll)?;
        if x < tx || x >= tx + TILE_W || y < ty || y >= ty + TILE_H {
            return None;
        }
        let on_remove = i < app.stack.items.len() && x >= tx + TILE_W - REMOVE - 4.0 && y < ty + REMOVE + 4.0;
        Some((i, on_remove))
    })
}

fn key_down(vk: VIRTUAL_KEY) -> bool {
    unsafe { GetKeyState(vk.0 as i32) < 0 }
}

/// Arrow keys move the focus and select it; with Shift they extend the
/// selection from the anchor.
fn move_focus(app: &mut App, delta: isize) {
    let n = app.stack.items.len();
    if n == 0 {
        return;
    }
    let ids = ids(app);
    let v = &mut app.stack_view;
    v.focus = (v.focus as isize + delta).clamp(0, n as isize - 1) as usize;
    if key_down(VK_SHIFT) {
        v.sel.extend(&ids, v.focus);
    } else {
        v.sel.only(&ids, v.focus);
    }
    keep_visible(v);
    render(app);
    announce(app);
}

/// Pressing on a selected tile drags the whole selection; pressing on any
/// other tile selects it alone and drags just that one. `grab` is where the
/// press was, so the drag image stays under the same spot of the tile.
fn start_drag(app: &mut App, index: usize, grab: POINT) {
    app.stack_view.press = None;
    let ids = ids(app);
    let Some(&id) = ids.get(index) else { return };
    if !app.stack_view.sel.contains(id) {
        app.stack_view.sel.only(&ids, index);
        app.stack_view.focus = index;
        render(app);
    }
    let chosen = app.stack_view.sel.ordered(&ids);
    let data = {
        let kinds: Vec<&Kind> = app.stack.items.iter().filter(|i| chosen.contains(&i.id)).map(|i| &i.kind).collect();
        dnd::data_object_many(&kinds)
    };
    let Some(data) = data else { return };
    let image = drag_image(app, index, chosen.len(), grab);
    let d = ms(app, HOVER_MS);
    let v = &mut app.stack_view;
    v.lifted = chosen;
    v.hover = None;
    v.lift.set(1.0, d);
    animate(app);
    // Messages keep flowing during the drag; hold no app state across it.
    dnd::drag_out(&data, image);
    if let Some(app) = crate::app_mut() {
        app.stack_view.lift.set(0.0, d);
        animate(app);
    }
}

pub fn handle(app: &mut App, hwnd: HWND, msg: u32, wparam: WPARAM, lparam: LPARAM) -> Option<LRESULT> {
    if msg == WM_GETOBJECT {
        return a11y::get_object(hwnd, wparam, lparam);
    }
    if msg == WM_TIMER && wparam.0 == ANIM_TIMER {
        tick(app);
        return Some(LRESULT(0));
    }
    if !app.stack_view.open {
        return None;
    }
    let vk = VIRTUAL_KEY(wparam.0 as u16);
    if msg == WM_KEYDOWN && app.stack_view.editing.is_some() && edit_key(app, vk) {
        return Some(LRESULT(0));
    }
    match msg {
        WM_CHAR => {
            if let Some(c) = char::from_u32(wparam.0 as u32) {
                edit_char(app, c);
            }
        }
        WM_KEYDOWN if app.stack_view.editing.is_some() => {}
        WM_KEYDOWN => match vk {
            VK_TAB if key_down(VK_CONTROL) => cycle_stack(app, if key_down(VK_SHIFT) { -1 } else { 1 }),
            VK_N if key_down(VK_CONTROL) => new_stack(app),
            VK_F2 => start_rename(app),
            VK_APPS => focused_menu(app),
            VK_F10 if key_down(VK_SHIFT) => focused_menu(app),
            VK_ESCAPE => close(app, true),
            VK_RIGHT => move_focus(app, 1),
            VK_LEFT => move_focus(app, -1),
            VK_DOWN => move_focus(app, COLS as isize),
            VK_UP => move_focus(app, -(COLS as isize)),
            VK_RETURN => activate(app, app.stack_view.focus),
            VK_DELETE => remove_selected(app),
            VK_A if key_down(VK_CONTROL) => {
                let ids = ids(app);
                app.stack_view.sel.all(&ids);
                render(app);
                announce(app);
            }
            _ => {}
        },
        WM_MOUSEMOVE => {
            if !app.stack_view.tracked {
                app.stack_view.tracked = true;
                let mut tme = TRACKMOUSEEVENT { cbSize: size_of::<TRACKMOUSEEVENT>() as u32, dwFlags: TME_LEAVE, hwndTrack: hwnd, dwHoverTime: 0 };
                unsafe {
                    let _ = TrackMouseEvent(&mut tme);
                }
            }
            if let Some((index, from)) = app.stack_view.press {
                let mut now = POINT::default();
                unsafe {
                    let _ = GetCursorPos(&mut now);
                }
                let (tx, ty) = unsafe { (GetSystemMetrics(SM_CXDRAG), GetSystemMetrics(SM_CYDRAG)) };
                if (now.x - from.x).abs() > tx || (now.y - from.y).abs() > ty {
                    start_drag(app, index, from);
                    return Some(LRESULT(0));
                }
            }
            let h = hit(app, lparam).filter(|&(i, _)| i < app.stack.items.len());
            let (hover, rm) = (h.map(|(i, _)| i), h.is_some_and(|(_, r)| r));
            if hover != app.stack_view.hover || rm != app.stack_view.hover_remove {
                app.stack_view.hover = hover;
                app.stack_view.hover_remove = rm;
                render(app);
            }
        }
        WM_MOUSELEAVE => {
            app.stack_view.tracked = false;
            app.stack_view.hover = None;
            render(app);
        }
        WM_LBUTTONDOWN => {
            if let Some((i, false)) = hit(app, lparam).filter(|&(i, _)| i < app.stack.items.len()) {
                let mut pt = POINT::default();
                unsafe {
                    let _ = GetCursorPos(&mut pt);
                }
                app.stack_view.press = Some((i, pt));
            }
        }
        WM_RBUTTONUP => match (hit_head(app, lparam), hit(app, lparam)) {
            (Some(Head::Stack(i)), _) => head_menu(app, i, cursor()),
            (_, Some((i, _))) if i < app.stack.items.len() => {
                // Right-clicking outside the selection acts on that tile alone.
                if !is_selected(app, i) {
                    take_focus(app, i);
                }
                tile_menu(app, cursor());
            }
            _ => {}
        },
        WM_LBUTTONUP if hit_head(app, lparam).is_some() => {
            app.stack_view.press = None;
            match hit_head(app, lparam) {
                Some(Head::Add) => new_stack(app),
                Some(Head::Stack(i)) => {
                    let double = app.stack_view.last_head_click.is_some_and(|(j, at)| {
                        j == i && at.elapsed().as_millis() <= unsafe { GetDoubleClickTime() } as u128
                    });
                    app.stack_view.last_head_click = Some((i, Instant::now()));
                    if double && i == app.stack.current {
                        start_rename(app);
                    } else {
                        switch_stack(app, i);
                    }
                }
                None => {}
            }
        }
        WM_LBUTTONUP => {
            app.stack_view.press = None;
            match hit(app, lparam) {
                Some((i, true)) => {
                    let id = app.stack.items[i].id;
                    remove(app, &[id]);
                }
                Some((i, false)) if i < app.stack.items.len() => {
                    let (ctrl, shift) = (wparam.0 as u32 & MK_CONTROL.0 != 0, wparam.0 as u32 & MK_SHIFT.0 != 0);
                    let double = !ctrl && !shift && app.stack_view.last_click.is_some_and(|(j, at)| {
                        j == i && at.elapsed().as_millis() <= unsafe { GetDoubleClickTime() } as u128
                    });
                    let ids = ids(app);
                    let v = &mut app.stack_view;
                    v.focus = i;
                    match (ctrl, shift) {
                        (_, true) => v.sel.extend(&ids, i),
                        (true, false) => v.sel.toggle(&ids, i),
                        _ => v.sel.only(&ids, i),
                    }
                    v.last_click = Some((i, Instant::now()));
                    if double {
                        app.stack_view.last_click = None;
                        activate(app, i);
                    } else {
                        render(app);
                        announce(app);
                    }
                }
                _ => {}
            }
        }
        WM_MOUSEWHEEL => {
            let delta = (wparam.0 >> 16) as i16 as isize;
            let rows = cells(app).div_ceil(COLS);
            let v = &mut app.stack_view;
            v.scroll = (v.scroll as isize - delta.signum()).clamp(0, rows.saturating_sub(ROWS) as isize) as usize;
            v.hover = None;
            render(app);
        }
        WM_ACTIVATE => {
            if (wparam.0 & 0xffff) as u32 == WA_INACTIVE && app.stack_view.press.is_none() {
                close(app, false);
            }
        }
        _ => return None,
    }
    Some(LRESULT(0))
}
//! Small local state under %APPDATA%\Cairn, plus DPAPI encryption for
//! anything that may hold user content.

use std::fs;
use std::path::{Path, PathBuf};

use windows::Win32::Foundation::{HLOCAL, LocalFree};
use windows::Win32::Security::Cryptography::*;
use windows::core::w;

pub fn dir() -> Option<PathBuf> {
    let base = std::env::var_os("APPDATA")?;
    let dir = PathBuf::from(base).join("Cairn");
    fs::create_dir_all(&dir).ok()?;
    Some(dir)
}

/// Writes through a temp file so a crash never leaves half a file.
pub fn write_atomic(path: &Path, data: &[u8]) -> std::io::Result<()> {
    let tmp = path.with_extension("tmp");
    fs::write(&tmp, data)?;
    fs::rename(&tmp, path)
}

fn blob(data: &[u8]) -> CRYPT_INTEGER_BLOB {
    CRYPT_INTEGER_BLOB { cbData: data.len() as u32, pbData: data.as_ptr() as *mut u8 }
}

fn take(out: CRYPT_INTEGER_BLOB) -> Vec<u8> {
    unsafe {
        let v = std::slice::from_raw_parts(out.pbData, out.cbData as usize).to_vec();
        let _ = LocalFree(HLOCAL(out.pbData as _));
        v
    }
}

/// Encrypts for the current Windows user (DPAPI).
pub fn protect(data: &[u8]) -> Option<Vec<u8>> {
    let mut out = CRYPT_INTEGER_BLOB::default();
    unsafe {
        CryptProtectData(&blob(data), w!("Cairn"), None, None, None, CRYPTPROTECT_UI_FORBIDDEN, &mut out).ok()?;
    }
    Some(take(out))
}

pub fn unprotect(data: &[u8]) -> Option<Vec<u8>> {
    let mut out = CRYPT_INTEGER_BLOB::default();
    unsafe {
        CryptUnprotectData(&blob(data), None, None, None, None, CRYPTPROTECT_UI_FORBIDDEN, &mut out).ok()?;
    }
    Some(take(out))
}
